* <SIGNATURE>---------------------------------------------------------------------------------------+
* | Instance Public Method ZCL_LOGGER->ADD_EXCEPTION
* +-------------------------------------------------------------------------------------------------+
* | [--->] IV_TYPE                        TYPE        SYMSGTY
* | [--->] IO_EXCEPTION                   TYPE REF TO CX_ROOT
* +--------------------------------------------------------------------------------------</SIGNATURE>
  METHOD add_exception.
    DATA: ls_exception TYPE bal_s_exc.

    CLEAR ls_exception.

    IF iv_type EQ 'E'.
      ADD 1 TO me->gv_errors.
      ADD 1 TO me->gv_errors_real.
    ENDIF.

    ls_exception-msgty     = iv_type.
    ls_exception-exception = io_exception.

    CALL FUNCTION 'BAL_LOG_EXCEPTION_ADD'
      EXPORTING
        i_log_handle     = gs_log_handler
        i_s_exc          = ls_exception
*   IMPORTING
*       E_S_MSG_HANDLE   =
*       E_MSG_WAS_LOGGED =
*       E_MSG_WAS_DISPLAYED       =
      EXCEPTIONS
        log_not_found    = 1
        msg_inconsistent = 2
        log_is_full      = 3
        OTHERS           = 4.
    IF sy-subrc NE 0.
      MESSAGE ID sy-msgid TYPE sy-msgty NUMBER sy-msgno WITH sy-msgv1 sy-msgv2 sy-msgv3 sy-msgv4.
    ENDIF.

    CLEAR me->gv_empty.

  ENDMETHOD.

    CALL FUNCTION 'BAL_LOG_MSG_ADD'
      EXPORTING

*   IMPORTING

* <SIGNATURE>---------------------------------------------------------------------------------------+
* | Instance Public Method ZCL_LOGGER->CONSTRUCTOR
* +-------------------------------------------------------------------------------------------------+
* | [--->] IV_EXTNUMBER                   TYPE        VBELN
* | [--->] IV_OBJECT                      TYPE        BALOBJ_D
* | [--->] IV_SUBOBJECT                   TYPE        BALSUBOBJ
* | [--->] IV_REPID                       TYPE        SY-REPID
* +--------------------------------------------------------------------------------------</SIGNATURE>
METHOD constructor.

  " 🔹 Asignar número externo al atributo del objeto
  me->mv_extnumber = iv_extnumber.

  " 🔹 Inicializar estructuras
  CLEAR: me->gs_log_header,
         me->gs_log_handler.

  " 🔹 Inicializar tabla de handlers
  REFRESH: me->gt_log_handler.

  " 🔹 Preparar cabecera del log
  me->gs_log_header-extnumber = iv_extnumber.

  CALL FUNCTION 'CONVERSION_EXIT_ALPHA_OUTPUT'
    EXPORTING
      input  = me->gs_log_header-extnumber
    IMPORTING
      output = me->gs_log_header-extnumber.

  me->gs_log_header-object     = iv_object.
  me->gs_log_header-subobject  = iv_subobject.
  me->gs_log_header-aldate     = sy-datum.
  me->gs_log_header-altime     = sy-uzeit.
  me->gs_log_header-aluser     = sy-uname.
  me->gs_log_header-alprog     = iv_repid.

  " 🔹 Crear el log
  CALL FUNCTION 'BAL_LOG_CREATE'
    EXPORTING
      i_s_log                 = me->gs_log_header
    IMPORTING
      e_log_handle            = me->gs_log_handler
    EXCEPTIONS
      log_header_inconsistent = 1
      OTHERS                  = 2.

  IF sy-subrc NE 0.
    MESSAGE ID sy-msgid TYPE sy-msgty NUMBER sy-msgno
            WITH sy-msgv1 sy-msgv2 sy-msgv3 sy-msgv4.
  ENDIF.

  " 🔹 Guardar el handler en la tabla interna
  APPEND me->gs_log_handler TO me->gt_log_handler.

  " 🔹 Marcar logger como vacío
  me->gv_empty = abap_true.

ENDMETHOD.

* <SIGNATURE>---------------------------------------------------------------------------------------+
* | Instance Public Method ZCL_LOGGER->FLUSH_TO_NAST_PROTOCOL
* +-------------------------------------------------------------------------------------------------+
* | [--->] IS_NAST                        TYPE        NAST
* | [<-->] CV_SUBRC                       TYPE        SY-SUBRC
* +--------------------------------------------------------------------------------------</SIGNATURE>
METHOD flush_to_nast_protocol.

  DATA: lv_line        TYPE ty_log_entry,
        lv_text        TYPE string,
        lv_formatted   TYPE string,
        lv_timestamp_s TYPE char25.   " ✅ Compatible con CONCATENATE

  LOOP AT mt_log INTO lv_line.

    " 🔹 Convertir timestamp a texto legible
    WRITE lv_line-timestamp TO lv_timestamp_s.

    " 🔹 Formatear línea
    CONCATENATE '[ ' lv_timestamp_s ' ]' '[ ' lv_line-type ' ]' lv_line-text
      INTO lv_formatted SEPARATED BY space.

    CONCATENATE lv_text lv_formatted cl_abap_char_utilities=>newline
      INTO lv_text.
  ENDLOOP.

  " 🔹 Registrar en protocolo NAST
  IF lv_text IS NOT INITIAL.
    CALL FUNCTION 'NAST_PROTOCOL_UPDATE'
      EXPORTING
        msg_text    = lv_text
        nast        = is_nast
        return_code = cv_subrc.
  ELSE.
    cv_subrc = 0.
  ENDIF.

ENDMETHOD.


* <SIGNATURE>---------------------------------------------------------------------------------------+
* | Instance Public Method ZCL_LOGGER->FLUSH_TO_SLG1
* +-------------------------------------------------------------------------------------------------+
* | [--->] IV_VBELN                       TYPE        VBELN
* +--------------------------------------------------------------------------------------</SIGNATURE>
METHOD flush_to_slg1.

  DATA: ls_msg TYPE bal_s_msg.

  " 🔹 Validar que el log handle esté disponible
  IF me->gs_log_handler IS INITIAL.
    RETURN.
  ENDIF.

  " 🔹 Agregar cada mensaje al log real
  LOOP AT me->mt_messages INTO ls_msg.
    CALL FUNCTION 'BAL_LOG_MSG_ADD'
      EXPORTING
        i_log_handle = me->gs_log_handler
        i_s_msg      = ls_msg.
  ENDLOOP.

  " 🔹 Grabar en SLG1 solo si hay mensajes
  IF sy-binpt = space AND me->mt_messages IS NOT INITIAL.
    CALL FUNCTION 'BAL_DB_SAVE'
      EXPORTING
        i_log_handle      = me->gs_log_handler
        i_in_update_task  = ' '
      EXCEPTIONS
        OTHERS = 1.
  ENDIF.

ENDMETHOD.


* <SIGNATURE>---------------------------------------------------------------------------------------+
* | Instance Public Method ZCL_LOGGER->GET_ALL_TEXTS

  ENDLOOP.

ENDMETHOD.


* <SIGNATURE>---------------------------------------------------------------------------------------+
* | Instance Public Method ZCL_LOGGER->GET_AS_BAPIRET1
* +-------------------------------------------------------------------------------------------------+
* | [<-()] RT_MESSAGES                    TYPE        BAPIRET1_TAB
* +--------------------------------------------------------------------------------------</SIGNATURE>
  METHOD get_as_bapiret1.
    DATA: lt_msg_handler TYPE bal_t_msgh,
          ls_message     TYPE bal_s_msg,
          ls_bapiret1    TYPE bapiret1,
          lv_text        TYPE text1024.

    FIELD-SYMBOLS: <ls_msg_handler> TYPE balmsghndl. " Work area para LT_MSG_HANDLER

    IF me->gv_empty EQ abap_true.
      RETURN.
    ENDIF.

    REFRESH rt_messages.
    CALL FUNCTION 'BAL_DB_LOAD'
      EXPORTING
        i_t_log_handle     = gt_log_handler
      IMPORTING
        e_t_msg_handle     = lt_msg_handler
      EXCEPTIONS
        no_logs_specified  = 1
        log_not_found      = 2
        log_already_loaded = 3
        OTHERS             = 4.
    IF sy-subrc NE 0.
      MESSAGE ID sy-msgid TYPE sy-msgty NUMBER sy-msgno
              WITH sy-msgv1 sy-msgv2 sy-msgv3 sy-msgv4.
    ENDIF.

    LOOP AT lt_msg_handler ASSIGNING <ls_msg_handler>.

      CALL FUNCTION 'BAL_LOG_MSG_READ'
        EXPORTING
          i_s_msg_handle = <ls_msg_handler>
        IMPORTING
          e_s_msg        = ls_message
          e_txt_msg      = lv_text
        EXCEPTIONS
          log_not_found  = 1
          msg_not_found  = 2
          OTHERS         = 3.
      IF sy-subrc NE 0.
        MESSAGE ID sy-msgid TYPE sy-msgty NUMBER sy-msgno WITH sy-msgv1 sy-msgv2 sy-msgv3 sy-msgv4.
      ENDIF.

      CLEAR ls_bapiret1.
      MOVE ls_message-msgty TO ls_bapiret1-type.
      MOVE ls_message-msgid TO ls_bapiret1-id.
      MOVE ls_message-msgno TO ls_bapiret1-number.
      MOVE lv_text TO ls_bapiret1-message.
      MOVE ls_message-msgv1 TO ls_bapiret1-message_v1.
      MOVE ls_message-msgv2 TO ls_bapiret1-message_v2.
      MOVE ls_message-msgv3 TO ls_bapiret1-message_v3.
      MOVE ls_message-msgv4 TO ls_bapiret1-message_v4.
      APPEND ls_bapiret1 TO rt_messages.

    ENDLOOP.

  ENDMETHOD.

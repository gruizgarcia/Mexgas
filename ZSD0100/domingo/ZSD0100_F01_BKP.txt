FORM entry USING return_code us_screen.

  DATA: lo_logger      TYPE REF TO zcl_logger,
        lo_cfdi        TYPE REF TO zcl_zmfcfdi40_mgs,
        lx_root        TYPE REF TO cx_root,
        lv_vbeln       TYPE vbeln,

  FIELD-SYMBOLS: <ls_message> TYPE bapiret1.

  CONSTANTS:
    c_log_objt   TYPE balobj_d   VALUE 'ZCFDI',
    c_log_subobj TYPE balsubobj  VALUE 'TIMBRADO',
    c_msgid      TYPE symsgid    VALUE 'ZCFDI_40'.

  CLEAR return_code.

  " 🔹 Obtener número de factura desde NAST
  lv_vbeln = nast-objky.

  CALL FUNCTION 'CONVERSION_EXIT_ALPHA_INPUT'
    EXPORTING
      input  = lv_vbeln
    IMPORTING
      output = lv_vbeln.

  " 🔹 Instanciar logger
  TRY.
      CREATE OBJECT lo_logger
        EXPORTING
          iv_extnumber = lv_vbeln
          iv_object    = c_log_objt
          iv_subobject = c_log_subobj
          iv_repid     = sy-repid.

      lo_logger->add_message(
        EXPORTING
          iv_msgty = 'I'
          iv_msgid = c_msgid
          iv_msgno = '001'
          iv_msgv1 = 'Logger activo en ENTRY.'
      ).

    CATCH cx_root INTO lx_root.
      return_code = 0.
      RETURN.
  ENDTRY.

  " 🔹 Instanciar clase CFDI
  TRY.
      CREATE OBJECT lo_cfdi.
      lo_logger->add_message(
        EXPORTING
          iv_msgty = 'I'
          iv_msgid = c_msgid
          iv_msgno = '002'
          iv_msgv1 = 'Clase CFDI instanciada correctamente.'
      ).
    CATCH cx_root INTO lx_root.
      lo_logger->add_message(
        EXPORTING
          iv_msgty = 'E'
          iv_msgid = c_msgid
          iv_msgno = '003'
          iv_msgv1 = lx_root->get_text( )
      ).
      return_code = 0.
  ENDTRY.

  " 🔹 Finalizar trazas
  lo_logger->add_message(
    EXPORTING
      iv_msgty = 'I'
      iv_msgid = c_msgid
      iv_msgno = '004'
      iv_msgv1 = 'Fin de rutina ENTRY.'
  ).

  " 🔹 Grabar en SLG1 si hay contenido
  IF lo_logger->has_entries( ) = abap_true.
    CALL METHOD lo_logger->flush_to_slg1( EXPORTING iv_vbeln = lv_vbeln ).
  ENDIF.

  " 🔹 Convertir a mensajes estructurados
  CALL METHOD lo_logger->get_as_bapiret1
    RECEIVING
      rt_messages = lt_messages.

  lv_count = lines( lt_messages ).

  " 🔹 Procesar cada mensaje en NAST_PROTOCOL_UPDATE
  LOOP AT lt_messages ASSIGNING <ls_message>.
    CASE <ls_message>-type.
      WHEN 'E'. lv_has_error   = abap_true.
      WHEN 'W'. lv_has_warning = abap_true.
    ENDCASE.

    CALL FUNCTION 'NAST_PROTOCOL_UPDATE'
      EXPORTING
        msg_arbgb = <ls_message>-id
        msg_nr    = <ls_message>-number
        msg_ty    = <ls_message>-type
        msg_v1    = <ls_message>-message_v1
        msg_v2    = <ls_message>-message_v2
        msg_v3    = <ls_message>-message_v3
        msg_v4    = <ls_message>-message_v4
      EXCEPTIONS
        OTHERS = 1.

    " 🔹 Acumular texto para popup (modo clásico)
    CLEAR lv_line.
    CONCATENATE <ls_message>-type
                ' - '
                <ls_message>-message_v1
                cl_abap_char_utilities=>newline
           INTO lv_line.
    CONCATENATE lv_popup_text lv_line INTO lv_popup_text.
  ENDLOOP.

  " 🔹 Determinar estado final
  IF lv_has_error = abap_true.
    return_code = 1.
    nast-vstat  = '1'.
  ELSEIF lv_has_warning = abap_true.
    return_code = 0.
    nast-vstat  = '2'.
  ELSE.
    return_code = 0.
    nast-vstat  = '0'.
  ENDIF.

  " 🔹 Mostrar popup si hay mensajes (modo interactivo)
  IF lv_popup_text IS NOT INITIAL AND sy-binpt = space.
    CALL FUNCTION 'POPUP_TO_DISPLAY_TEXT'
      EXPORTING
        textline1 = 'Mensajes CFDI'
        textline2 = lv_popup_text
        titel     = 'Resultado de procesamiento'
      EXCEPTIONS
        OTHERS = 1.
  ENDIF.

ENDFORM.

FORM update_from_logger USING po_logger TYPE REF TO zcl_logger
                              return_code TYPE i
                              nast TYPE nast.

  DATA: lt_messages    TYPE STANDARD TABLE OF bapiret1,
        lv_has_error   TYPE abap_bool VALUE abap_false,
        lv_has_warning TYPE abap_bool VALUE abap_false,
        lv_count       TYPE i,
        lv_popup_text  TYPE string,
        lv_line        TYPE string,
        lv_vbeln       TYPE vbeln.

  FIELD-SYMBOLS: <ls_message> TYPE bapiret1.

  CLEAR return_code.

  " 🔹 Validar que el logger esté instanciado
  IF po_logger IS BOUND.

    " 🔹 Convertir NAST-OBJKY a VBELN con formato ALPHA
    lv_vbeln = nast-objky.
    CALL FUNCTION 'CONVERSION_EXIT_ALPHA_INPUT'
      EXPORTING
        input  = lv_vbeln
      IMPORTING
        output = lv_vbeln.

    " 🔹 Grabar en SLG1 si hay contenido
    IF po_logger->has_entries( ) = abap_true.
      CALL METHOD po_logger->flush_to_slg1( EXPORTING iv_vbeln = lv_vbeln ).
    ENDIF.

    " 🔹 Convertir a mensajes estructurados
    CALL METHOD po_logger->save.
    CALL METHOD po_logger->get_as_bapiret1
      RECEIVING
        rt_messages = lt_messages.

    lv_count = lines( lt_messages ).

    " 🔹 Procesar cada mensaje en NAST_PROTOCOL_UPDATE
    LOOP AT lt_messages ASSIGNING <ls_message>.
      CASE <ls_message>-type.
        WHEN 'E'. lv_has_error   = abap_true.
        WHEN 'W'. lv_has_warning = abap_true.
      ENDCASE.

      CALL FUNCTION 'NAST_PROTOCOL_UPDATE'
        EXPORTING
          msg_arbgb = <ls_message>-id
          msg_nr    = <ls_message>-number
          msg_ty    = <ls_message>-type
          msg_v1    = <ls_message>-message_v1
          msg_v2    = <ls_message>-message_v2
          msg_v3    = <ls_message>-message_v3
          msg_v4    = <ls_message>-message_v4
        EXCEPTIONS
          OTHERS = 1.

      " 🔹 Acumular texto para popup (modo clásico)
      CLEAR lv_line.
      CONCATENATE <ls_message>-type
                  ' - '
                  <ls_message>-message_v1
                  cl_abap_char_utilities=>newline
             INTO lv_line.
      CONCATENATE lv_popup_text lv_line INTO lv_popup_text.
    ENDLOOP.

    " 🔹 Determinar estado final
    IF lv_has_error = abap_true.
      return_code = 1.
      nast-vstat  = '1'.
    ELSEIF lv_has_warning = abap_true.
      return_code = 0.
      nast-vstat  = '2'.
    ELSE.
      return_code = 0.
      nast-vstat  = '0'.
    ENDIF.

    " 🔹 Mostrar popup si hay mensajes (modo interactivo)
    IF lv_popup_text IS NOT INITIAL AND sy-binpt = space.
      CALL FUNCTION 'POPUP_TO_DISPLAY_TEXT'
        EXPORTING
          textline1 = 'Mensajes CFDI'
          textline2 = lv_popup_text
          titel     = 'Resultado de procesamiento'
        EXCEPTIONS
          OTHERS = 1.
    ENDIF.

  ELSE.
    " 🔹 Logger no instanciado: error técnico
    return_code = 1.
    nast-vstat  = '1'.
  ENDIF.

ENDFORM.
FORM get_nast_data
  USING
    p_nast TYPE nast
  CHANGING
    p_vbeln TYPE vbeln
    p_error TYPE string
    p_logger TYPE REF TO zcl_logger.

  CONSTANTS: c_log_objt TYPE balobj_d VALUE 'ZCFDI',
             c_log_subobj TYPE balsubobj VALUE 'NAST_READ'.

  TRY.
      p_vbeln = p_nast-objky.

      IF p_vbeln IS INITIAL.
        p_error = 'OBJKEY vacío en NAST.'.
        RETURN.
      ENDIF.

      CREATE OBJECT p_logger
        EXPORTING
          iv_extnumber = p_vbeln
          iv_object    = c_log_objt
          iv_subobject = c_log_subobj
          iv_repid     = sy-repid.

      p_logger->add_text( EXPORTING iv_type = 'I' iv_text = |NAST leído correctamente: { p_vbeln }| ).

    CATCH cx_root INTO DATA(lx_root).
      p_error = lx_root->get_text( ).
  ENDTRY.

ENDFORM.
FORM validar_y_procesar_nast
  USING
    p_nast TYPE nast
  CHANGING
    p_return_code TYPE sy-subrc
    p_error       TYPE string.

  DATA: lv_text_popup  TYPE string,
        lv_anzal_text  TYPE string,
        lv_popup_line  TYPE string,
        lv_answer      TYPE char1,
        lv_vbeln       TYPE vbeln_vf,
        lv_factura     TYPE vbeln.

  CLEAR: p_error, p_return_code, lv_text_popup.

  " 🔹 Validar campos clave de NAST
  IF p_nast-objky IS INITIAL.
    CONCATENATE lv_text_popup 'NAST-OBJKEY está vacío.' cl_abap_char_utilities=>newline INTO lv_text_popup.
    p_error = 'NAST-OBJKEY está vacío.'.
  ENDIF.

  IF p_nast-kschl IS INITIAL.
    CONCATENATE lv_text_popup 'NAST-KSCHL (tipo de mensaje) está vacío.' cl_abap_char_utilities=>newline INTO lv_text_popup.
    IF p_error IS INITIAL.
      p_error = 'NAST-KSCHL está vacío.'.
    ENDIF.
  ENDIF.

  IF p_nast-spras IS INITIAL.
    CONCATENATE lv_text_popup 'NAST-SPRAS (idioma) está vacío.' cl_abap_char_utilities=>newline INTO lv_text_popup.
    IF p_error IS INITIAL.
      p_error = 'NAST-SPRAS está vacío.'.
    ENDIF.
  ENDIF.

  IF p_nast-parnr IS INITIAL.
    CONCATENATE lv_text_popup 'NAST-PARNR (socio comercial) está vacío.' cl_abap_char_utilities=>newline INTO lv_text_popup.
    IF p_error IS INITIAL.
      p_error = 'NAST-PARNR está vacío.'.
    ENDIF.
  ENDIF.

  IF p_nast-parvw IS INITIAL.
    CONCATENATE lv_text_popup 'NAST-PARVW (rol del socio) está vacío.' cl_abap_char_utilities=>newline INTO lv_text_popup.
    IF p_error IS INITIAL.
      p_error = 'NAST-PARVW está vacío.'.
    ENDIF.
  ENDIF.

  IF p_nast-vstat = '9'.
    CONCATENATE lv_text_popup 'NAST-VSTAT indica que el mensaje ya fue procesado (estado 9).' cl_abap_char_utilities=>newline INTO lv_text_popup.
    IF p_error IS INITIAL.
      p_error = 'Mensaje ya procesado (VSTAT = 9).'.
    ENDIF.
  ENDIF.

  IF p_nast-anzal > 3.
    lv_anzal_text = p_nast-anzal.
    CONCATENATE 'NAST-ANZAL excede el límite de intentos:' lv_anzal_text INTO lv_popup_line SEPARATED BY space.
    CONCATENATE lv_text_popup lv_popup_line cl_abap_char_utilities=>newline INTO lv_text_popup.
    IF p_error IS INITIAL.
      p_error = lv_popup_line.
    ENDIF.
  ENDIF.

  " 🔹 Mostrar popup si hay errores
  IF p_error IS NOT INITIAL.
    CALL FUNCTION 'POPUP_TO_CONFIRM'
      EXPORTING
        titlebar       = 'Errores en NAST'
        text_question  = lv_text_popup
        text_button_1  = 'Reintentar'
        text_button_2  = 'Cancelar'
        default_button = '2'
      IMPORTING
        answer         = lv_answer.

    CALL FUNCTION 'NAST_PROTOCOL_UPDATE'
      EXPORTING
        msg_arbgb = 'ZCFDI'
        msg_nr    = '000'
        msg_ty    = 'E'
        msg_v1    = p_error
        msg_v2    = p_nast-objky
      EXCEPTIONS
        OTHERS    = 1.

    IF lv_answer = '2'.
      p_return_code = 1.
      RETURN.
    ENDIF.
  ENDIF.

  " 🔹 Validar existencia en VBRK
  lv_vbeln = p_nast-objky.

  SELECT SINGLE vbeln FROM vbrk INTO lv_factura
    WHERE vbeln = lv_vbeln.

  IF sy-subrc <> 0.
    p_return_code = 1.
    p_error = |Factura { lv_vbeln } no existe en VBRK.|.

    CALL FUNCTION 'NAST_PROTOCOL_UPDATE'
      EXPORTING
        msg_arbgb = 'ZCFDI'
        msg_nr    = '001'
        msg_ty    = 'E'
        msg_v1    = p_error
        msg_v2    = lv_vbeln
      EXCEPTIONS
        OTHERS    = 1.

    RETURN.
  ENDIF.

  " 🔹 Procesar timbrado
  DATA(lo_cfdi) = NEW zcl_zmfcfdi40_mgs( ).

  CALL METHOD lo_cfdi->procesar_salida_mensaje
    EXPORTING
      iv_objkey = p_nast-objky
      iv_nacha  = p_nast-nacha
      iv_spras  = p_nast-spras
    IMPORTING
      ev_error  = p_error
      CHANGING
    po_logger = lo_logger.

  IF p_error IS INITIAL AND lo_cfdi->was_successful( ) = abap_true.
    p_return_code = 0.

    CALL FUNCTION 'NAST_PROTOCOL_UPDATE'
      EXPORTING
        msg_arbgb = 'ZCFDI'
        msg_nr    = '002'
        msg_ty    = 'S'
        msg_v1    = 'Timbrado exitoso'
        msg_v2    = lv_vbeln
      EXCEPTIONS
        OTHERS    = 1.

  ELSE.
    p_return_code = 1.
    IF p_error IS INITIAL.
      p_error = 'Proceso finalizó sin error explícito pero no se generó factura.'.
    ENDIF.

    CALL FUNCTION 'NAST_PROTOCOL_UPDATE'
      EXPORTING
        msg_arbgb = 'ZCFDI'
        msg_nr    = '003'
        msg_ty    = 'E'
        msg_v1    = p_error
        msg_v2    = lv_vbeln
      EXCEPTIONS
        OTHERS    = 1.
  ENDIF.

ENDFORM.

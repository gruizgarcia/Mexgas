* <SIGNATURE>---------------------------------------------------------------------------------------+
* | Instance Public Method ZCL_ZMFCFDI40_MGS->ANEXA_FACTURA_GOS
* +-------------------------------------------------------------------------------------------------+
* | [--->] IV_FACTURA                     TYPE        VBELN
* | [--->] IV_ARCHIVO                     TYPE        STRING
* | [--->] IV_DESCRIPCION                 TYPE        SO_OBJ_DES
* | [--->] IV_SERVIDOR                    TYPE        STRING
* | [--->] IT_PDF                         TYPE        SOLIX_TAB
* | [<-->] PO_LOGGER                      TYPE REF TO ZCL_LOGGER
* +--------------------------------------------------------------------------------------</SIGNATURE>
  METHOD anexa_factura_gos.
    DATA: p_type      TYPE swo_objtyp VALUE 'VBRK',
          p_key       TYPE swo_typeid,
          p_file      TYPE c LENGTH 255,
          p_desc      TYPE so_obj_des,
          ls_fol_id   TYPE soodk,
          ls_obj_id   TYPE soodk,
          ls_obj_data TYPE sood1,
          ls_folmem_k TYPE sofmk,
          ls_note     TYPE borident,
          ls_object   TYPE borident,
          lv_ep_note  TYPE borident-objkey,
          lv_offset   TYPE i,
          it_objhead  TYPE STANDARD TABLE OF soli,
          it_content  TYPE soli_tab,
          it_content2 TYPE solix_tab,
          wa_content  TYPE soli,
          lv_len      TYPE i.

    p_key  = iv_factura.
    p_file = iv_archivo.
    p_desc = iv_descripcion.

    ls_object-objkey  = p_key.
    ls_object-objtype = p_type.

    IF iv_servidor = 'X'.
      TRY.
          OPEN DATASET p_file FOR INPUT IN BINARY MODE.
          WHILE sy-subrc = 0.
            READ DATASET p_file INTO wa_content.
            IF sy-subrc = 0.
              APPEND wa_content TO it_content.
            ENDIF.
          ENDWHILE.
          CLOSE DATASET p_file.
        CATCH cx_sy_file_access_error.
          IF po_logger IS BOUND.
            po_logger->add_message(
              EXPORTING
                iv_msgty = 'E'          " Tipo de mensaje: E = Error, W = Warning, etc.
                iv_msgid = 'ZCFDI'      " ID del mensaje (catálogo)
                iv_msgno = '001'        " Número del mensaje
                iv_msgv1 = 'Eror archivo open datase' ). " Texto dinámico del mensaje
          ENDIF.
          RETURN.
      ENDTRY.
    ELSE.
      IF it_pdf IS INITIAL.
        IF po_logger IS BOUND.
          po_logger->add_message( EXPORTING iv_msgty = 'E' iv_msgid = 'ZCFDI_40' iv_msgno = '001'
                                  iv_msgv1 = 'IT_PDF vacío (se esperaba contenido PDF en SOLIX_TAB)' ).
        ENDIF.
        RETURN.
      ENDIF.
      it_content2 = it_pdf.
      CALL FUNCTION 'SO_SOLIXTAB_TO_SOLITAB' EXPORTING ip_solixtab = it_content2 IMPORTING ep_solitab = it_content.
    ENDIF.

    CALL FUNCTION 'SO_CONVERT_CONTENTS_BIN'
      EXPORTING
        it_contents_bin = it_content[]
      IMPORTING
        et_contents_bin = it_content[].

    CALL FUNCTION 'SO_FOLDER_ROOT_ID_GET' EXPORTING region = 'B' IMPORTING folder_id = ls_fol_id.
    IF sy-subrc <> 0.
      IF po_logger IS BOUND.
        po_logger->add_message( EXPORTING iv_msgty = 'E' iv_msgid ='ZCFDI_40' iv_msgno ='001'
                                iv_msgv1 ='No se pudo obtener carpeta Business Documents (SO_FOLDER_ROOT_ID_GET).' ).
      ENDIF.
      RETURN.
    ENDIF.

    ls_obj_data-objsns = 'O'.
    ls_obj_data-objla  = sy-langu.
    ls_obj_data-objdes = p_desc.
    IF p_file IS INITIAL OR strlen( p_file ) < 3.
      ls_obj_data-file_ext = 'PDF'.
    ELSE.
      lv_offset            = strlen( p_file ) - 3.
      ls_obj_data-file_ext = p_file+lv_offset(3).
    ENDIF.

    lv_len = lines( it_content ) * 510.
    ls_obj_data-objlen = lv_len.

    CALL FUNCTION 'SO_OBJECT_INSERT'
      EXPORTING
        folder_id        = ls_fol_id
        object_type      = 'EXT'
        object_hd_change = ls_obj_data
      IMPORTING
        object_id        = ls_obj_id
      TABLES
        objhead          = it_objhead
        objcont          = it_content.
    IF sy-subrc <> 0.
      IF po_logger IS BOUND.
        po_logger->add_message( EXPORTING iv_msgty ='E' iv_msgid ='ZCFDI_40' iv_msgno ='001'
                                iv_msgv1 ='Error al insertar objeto en SAPoffice (SO_OBJECT_INSERT).' ).
      ENDIF.
      RETURN.
    ENDIF.

    ls_folmem_k-foltp = ls_fol_id-objtp.
    ls_folmem_k-folyr = ls_fol_id-objyr.
    ls_folmem_k-folno = ls_fol_id-objno.
    ls_folmem_k-doctp = ls_obj_id-objtp.
    ls_folmem_k-docyr = ls_obj_id-objyr.
    ls_folmem_k-docno = ls_obj_id-objno.
    lv_ep_note        = ls_folmem_k.

    ls_note-objtype = 'MESSAGE'.
    ls_note-objkey  = lv_ep_note.

    CALL FUNCTION 'BINARY_RELATION_CREATE'
      EXPORTING
        obj_rolea    = ls_object
        obj_roleb    = ls_note
        relationtype = 'ATTA'.

    IF po_logger IS BOUND.
      IF sy-subrc = 0.
        po_logger->add_message( EXPORTING iv_msgty ='S' iv_msgid ='ZCFDI_40' iv_msgno ='001'
                                iv_msgv1 = |Archivo anexado exitosamente a factura { iv_factura }| ).
      ELSE.
        po_logger->add_message( EXPORTING iv_msgty = 'E' iv_msgid = 'ZCFDI_40' iv_msgno ='001'
                                iv_msgv1 = 'Error al crear relación ATTA (BINARY_RELATION_CREATE).' ).
      ENDIF.
    ENDIF.
  ENDMETHOD.

          ls_obj_data TYPE sood1,
          ls_folmem_k TYPE sofmk,
          ls_note     TYPE borident,
          ls_object   TYPE borident,
          lv_ep_note  TYPE borident-objkey,
          lv_offset   TYPE i,

    p_file = |Factura.xml|.

    DATA ls_xml TYPE zcfdi_40_xml.

      RETURN.
    ENDIF.

    CALL FUNCTION 'SO_STRING_TO_TAB' EXPORTING content_str = ls_xml-xml TABLES content_tab = it_content[].

    CALL FUNCTION 'SO_FOLDER_ROOT_ID_GET' EXPORTING region ='B' IMPORTING folder_id = ls_fol_id.
    IF sy-subrc <> 0.
      IF po_logger IS BOUND.
        po_logger->add_message( iv_msgty ='E' iv_msgid ='ZCFDI_40' iv_msgno ='001'

        object_hd_change = ls_obj_data
      IMPORTING
        object_id        = ls_obj_id
      TABLES
        objhead          = it_objhead
        objcont          = it_content.
    IF sy-subrc <> 0.
      IF po_logger IS BOUND.
        po_logger->add_message( EXPORTING iv_msgty ='E' iv_msgid ='ZCFDI_40' iv_msgno ='001'
                                iv_msgv1 ='Error al insertar objeto XML en SAPoffice (SO_OBJECT_INSERT).' ).
      ENDIF.
      RETURN.
    ENDIF.

    ls_object-objtype = p_type.
    ls_object-objkey  = iv_factura.

    ls_folmem_k-foltp = ls_fol_id-objtp.
    ls_folmem_k-folyr = ls_fol_id-objyr.

    ls_folmem_k-docno = ls_obj_id-objno.

* <SIGNATURE>---------------------------------------------------------------------------------------+
* | Instance Public Method ZCL_ZMFCFDI40_MGS->BUILD_CONCEPTOS
* +-------------------------------------------------------------------------------------------------+
* | [--->] IS_HEAD                        TYPE        LINE OF TT_HEAD40
* | [--->] IT_DET                         TYPE        TT_DET
* | [--->] IV_VER                         TYPE        ABAP_BOOL
* | [--->] IV_FIX_IMPORT                  TYPE        ABAP_BOOL
* | [<-->] CV_SUBTOTAL                    TYPE        TY_P14
* +--------------------------------------------------------------------------------------</SIGNATURE>
  METHOD build_conceptos.
    DATA: lr_concepts TYPE REF TO if_ixml_element,
          lv_dec      TYPE i,
          ls_head40   TYPE zst_head_40,    " <-- estructura explícita
          lv_waers    TYPE waers.

    FIELD-SYMBOLS: <ls_det> TYPE ty_det,
                   <f_cur>  TYPE any.

    " Copiar parámetro a estructura local (evita que el parser lo vea como tabla)
    ls_head40 = is_head.

    " Nodo raíz
    lr_concepts = mr_doc->create_simple_element(
                    name   = 'cfdi:Conceptos'
                    parent = mr_comp ).

    " --- Moneda y decimales (dinámico: MONEDAXML/WAERK/WAERS) ---
    CLEAR lv_waers.
    ASSIGN COMPONENT 'MONEDAXML' OF STRUCTURE ls_head40 TO <f_cur>.
    IF sy-subrc = 0 AND <f_cur> IS ASSIGNED AND <f_cur> IS NOT INITIAL.
      lv_waers = <f_cur>.
    ELSE.
      ASSIGN COMPONENT 'WAERK' OF STRUCTURE ls_head40 TO <f_cur>.
      IF sy-subrc = 0 AND <f_cur> IS ASSIGNED AND <f_cur> IS NOT INITIAL.
        lv_waers = <f_cur>.
      ELSE.
        ASSIGN COMPONENT 'WAERS' OF STRUCTURE ls_head40 TO <f_cur>.
        IF sy-subrc = 0 AND <f_cur> IS ASSIGNED AND <f_cur> IS NOT INITIAL.
          lv_waers = <f_cur>.
        ENDIF.
      ENDIF.
    ENDIF.
    IF lv_waers IS INITIAL.
      lv_waers = 'MXN'.
    ENDIF.

    lv_dec = me->currency_decimals( iv_waers = lv_waers ).
    IF lv_dec IS INITIAL OR lv_dec < 0.
      lv_dec = 2.
    ENDIF.

    " --- Conceptos ---
    LOOP AT it_det ASSIGNING <ls_det>.
      DATA: lr_c         TYPE REF TO if_ixml_element,
            lv_qty_10    TYPE ty_p10,
            lv_vu_10     TYPE ty_p10,
            lv_qty_p     TYPE ty_p14,
            lv_vu_p      TYPE ty_p14,
            lv_imp       TYPE ty_p14,
            lv_imp_iva_p TYPE ty_p14,
            lv_objimp    TYPE string,
            lv_tasa      TYPE ty_p14.

      lr_c = mr_doc->create_simple_element( name = 'cfdi:Concepto' parent = lr_concepts ).

      " Cantidad
      IF <ls_det>-cantidad > 0.
        lv_qty_10 = <ls_det>-cantidad.
      ELSE.
        lv_qty_10 = 1.
      ENDIF.

      " Valor unitario
      lv_vu_10 = <ls_det>-valor_unitario.

      lv_qty_p = lv_qty_10.
      lv_vu_p  = lv_vu_10.

      " Importe del concepto
      IF iv_fix_import = abap_true.
        lv_imp = me->round_half_up( iv_val = lv_qty_p * lv_vu_p iv_dec = lv_dec ).
      ELSE.
        lv_imp = <ls_det>-importe.
      ENDIF.

      " Subtotal
      cv_subtotal = cv_subtotal + lv_imp.

      " Atributos del concepto
      me->set_attr( node = lr_c name = 'ClaveProdServ'    value = <ls_det>-clave_prodserv ).
      me->set_attr( node = lr_c name = 'NoIdentificacion' value = <ls_det>-clave_prodserv ).
      me->set_attr( node = lr_c name = 'Cantidad'         value = me->to_str( iv_val = lv_qty_p iv_dec = lv_dec ) ).
      me->set_attr( node = lr_c name = 'ClaveUnidad'      value = <ls_det>-clave_unidad ).
      me->set_attr( node = lr_c name = 'Descripcion'      value = <ls_det>-descripcion ).
      me->set_attr( node = lr_c name = 'ValorUnitario'    value = me->to_str( iv_val = lv_vu_p  iv_dec = 6 ) ).
      me->set_attr( node = lr_c name = 'Importe'          value = me->to_str( iv_val = lv_imp   iv_dec = lv_dec ) ).

      " ObjetoImp
      IF <ls_det>-importe_impuesto > 0.
        lv_objimp = '02'.
      ELSE.
        lv_objimp = '01'.
      ENDIF.
      me->set_attr( node = lr_c name = 'ObjetoImp' value = lv_objimp ).

      " Impuestos del concepto (si aplica)
      IF <ls_det>-importe_impuesto > 0.
        DATA: lr_imps TYPE REF TO if_ixml_element,
              lr_tras TYPE REF TO if_ixml_element,
              lr_t    TYPE REF TO if_ixml_element.

        lr_imps = mr_doc->create_simple_element( name = 'cfdi:Impuestos' parent = lr_c ).
        lr_tras = mr_doc->create_simple_element( name = 'cfdi:Traslados' parent = lr_imps ).
        lr_t    = mr_doc->create_simple_element( name = 'cfdi:Traslado'  parent = lr_tras ).

        IF <ls_det>-tasa_iva IS INITIAL.
          lv_tasa = '0.160000'.
        ELSE.
          lv_tasa = <ls_det>-tasa_iva.
        ENDIF.
        lv_imp_iva_p = <ls_det>-importe_impuesto.

        me->set_attr( node = lr_t name = 'Base'       value = me->to_str( iv_val = lv_imp        iv_dec = lv_dec ) ).
        me->set_attr( node = lr_t name = 'Impuesto'   value = '002' ).
        me->set_attr( node = lr_t name = 'TipoFactor' value = 'Tasa' ).
        me->set_attr( node = lr_t name = 'TasaOCuota' value = me->to_str( iv_val = lv_tasa       iv_dec = 6 ) ).
        me->set_attr( node = lr_t name = 'Importe'    value = me->to_str( iv_val = lv_imp_iva_p  iv_dec = 2 ) ).
      ENDIF.
    ENDLOOP.
  ENDMETHOD.

* <SIGNATURE>---------------------------------------------------------------------------------------+
* | Instance Public Method ZCL_ZMFCFDI40_MGS->BUILD_EMISOR
* +-------------------------------------------------------------------------------------------------+
* | [--->] IS_HEAD                        TYPE        LINE OF TT_HEAD40
* +--------------------------------------------------------------------------------------</SIGNATURE>
  METHOD build_emisor.
    DATA lr_emisor TYPE REF TO if_ixml_element.

    FIELD-SYMBOLS:
      <fs_bukrs> TYPE any,
      <fs_rfc>   TYPE any,
      <fs_nom>   TYPE any,
      <fs_reg>   TYPE any.

    DATA:
      lv_bukrs TYPE bukrs,
      lv_rfc   TYPE stcd1,
      lv_nom   TYPE string,
      lv_reg   TYPE c LENGTH 3.

    "---- Leer BUKRS (para fallback)
    ASSIGN COMPONENT 'BUKRS' OF STRUCTURE is_head TO <fs_bukrs>.
    IF <fs_bukrs> IS ASSIGNED.
      lv_bukrs = <fs_bukrs>.
    ENDIF.

    "---- RFC emisor: probar distintos nombres; si no, T001Z party = 'MX_RFC'
    CLEAR lv_rfc.
    ASSIGN COMPONENT 'RFC_EMISOR' OF STRUCTURE is_head TO <fs_rfc>.
    IF sy-subrc <> 0 OR <fs_rfc> IS INITIAL.
      ASSIGN COMPONENT 'RFCEMISOR' OF STRUCTURE is_head TO <fs_rfc>.
    ENDIF.
    IF sy-subrc = 0 AND <fs_rfc> IS ASSIGNED AND <fs_rfc> IS NOT INITIAL.
      lv_rfc = <fs_rfc>.
    ELSEIF lv_bukrs IS NOT INITIAL.
      SELECT SINGLE paval INTO lv_rfc
        FROM t001z
        WHERE bukrs = lv_bukrs AND party = 'MX_RFC'.
    ENDIF.

    "---- Nombre emisor: probar distintos nombres; si no, T001-BUTXT
    CLEAR lv_nom.
    ASSIGN COMPONENT 'NOMBRE_EMISOR' OF STRUCTURE is_head TO <fs_nom>.
    IF sy-subrc <> 0 OR <fs_nom> IS INITIAL.
      ASSIGN COMPONENT 'NOM_EMISOR' OF STRUCTURE is_head TO <fs_nom>.
    ENDIF.
    IF sy-subrc = 0 AND <fs_nom> IS ASSIGNED AND <fs_nom> IS NOT INITIAL.
      lv_nom = <fs_nom>.
    ELSEIF lv_bukrs IS NOT INITIAL.
      SELECT SINGLE butxt INTO lv_nom FROM t001 WHERE bukrs = lv_bukrs.
    ENDIF.

    "---- Régimen fiscal: probar variantes; si no, default '601'
    CLEAR lv_reg.
    ASSIGN COMPONENT 'REGIMEN_FISCAL'      OF STRUCTURE is_head TO <fs_reg>.
    IF sy-subrc <> 0 OR <fs_reg> IS INITIAL.
      ASSIGN COMPONENT 'REGIMENFISCAL'     OF STRUCTURE is_head TO <fs_reg>.
    ENDIF.
    IF sy-subrc <> 0 OR <fs_reg> IS INITIAL.
      ASSIGN COMPONENT 'REGIMENFISCALRECEPTOR' OF STRUCTURE is_head TO <fs_reg>. "por si lo guardaste así
    ENDIF.
    IF sy-subrc = 0 AND <fs_reg> IS ASSIGNED AND <fs_reg> IS NOT INITIAL.
      lv_reg = <fs_reg>.
    ELSE.
      lv_reg = '601'.  " General de Ley Personas Morales (ajusta si aplica)
    ENDIF.

    "---- Crear nodo y setear solo lo que tengas
    lr_emisor = mr_doc->create_simple_element( name = 'cfdi:Emisor' parent = mr_comp ).

    IF lv_rfc IS NOT INITIAL.
      me->set_attr( node = lr_emisor name = 'Rfc' value = lv_rfc ).
    ENDIF.
    IF lv_nom IS NOT INITIAL.
      me->set_attr( node = lr_emisor name = 'Nombre' value = lv_nom ).
    ENDIF.
    IF lv_reg IS NOT INITIAL.
      me->set_attr( node = lr_emisor name = 'RegimenFiscal' value = lv_reg ).
    ENDIF.
  ENDMETHOD.


* <SIGNATURE>---------------------------------------------------------------------------------------+
* | Instance Public Method ZCL_ZMFCFDI40_MGS->BUILD_HEADER
* +-------------------------------------------------------------------------------------------------+
* | [--->] IS_HEAD                        TYPE        LINE OF TT_HEAD40
* | [--->] IV_FIX_FECHA                   TYPE        ABAP_BOOL
* | [<-->] CV_USO_CFDI                    TYPE        TY_CHAR3
* +--------------------------------------------------------------------------------------</SIGNATURE>
  METHOD build_header.
    TYPES ty_p14 TYPE p LENGTH 14 DECIMALS 6.

    FIELD-SYMBOLS:
      <f_fecha> TYPE any,
      <f_bldat> TYPE any,
      <f_uzeit> TYPE any,
      <f_cond>  TYPE any,
      <f_exp>   TYPE any,
      <f_cur>   TYPE any,
      <f_tc>    TYPE any,
      <f_lug>   TYPE any,
      <f_tip>   TYPE any,
      <f_met>   TYPE any,
      <f_for>   TYPE any,
      <f_uso>   TYPE any.

    DATA: lw_head  TYPE zst_head_40,   " estructura garantizada
          lv_fecha TYPE string,
          lv_mon   TYPE waers,
          lv_tmp_c TYPE c LENGTH 40,
          lv_tc_p  TYPE ty_p14,
          lv_tc_s  TYPE string,
          lv_exp   TYPE string,
          lv_d     TYPE dats,
          lv_t     TYPE tims.

    lw_head = is_head.

    "---- Fecha ---------------------------------------------------------------
    IF iv_fix_fecha = abap_true.
      CALL FUNCTION 'ZCFDI_GET_TIMESTAMP'
        IMPORTING
          ev_timestamp = lv_fecha.
    ELSE.
      ASSIGN COMPONENT 'FECHAFACT' OF STRUCTURE lw_head TO <f_fecha>.
      IF sy-subrc = 0 AND <f_fecha> IS ASSIGNED AND <f_fecha> IS NOT INITIAL.
        lv_fecha = <f_fecha>.
      ELSE.
        CLEAR: lv_d, lv_t.
        ASSIGN COMPONENT 'BLDAT' OF STRUCTURE lw_head TO <f_bldat>.
        IF sy-subrc = 0 AND <f_bldat> IS ASSIGNED AND <f_bldat> IS NOT INITIAL.
          lv_d = <f_bldat>.
        ELSE.
          ASSIGN COMPONENT 'ERDAT' OF STRUCTURE lw_head TO <f_bldat>.
          IF sy-subrc = 0 AND <f_bldat> IS ASSIGNED AND <f_bldat> IS NOT INITIAL.
            lv_d = <f_bldat>.
          ELSE.
            lv_d = sy-datum.
          ENDIF.
        ENDIF.

        ASSIGN COMPONENT 'UZEIT' OF STRUCTURE lw_head TO <f_uzeit>.
        IF sy-subrc = 0 AND <f_uzeit> IS ASSIGNED AND <f_uzeit> IS NOT INITIAL.
          lv_t = <f_uzeit>.
        ELSE.
          lv_t = sy-uzeit.
        ENDIF.

        lv_fecha = |{ lv_d+0(4) }-{ lv_d+4(2) }-{ lv_d+6(2) }T{ lv_t+0(2) }:{ lv_t+2(2) }:{ lv_t+4(2) }|.
      ENDIF.
    ENDIF.

    " Nodo Comprobante
    me->set_attr( node = mr_comp name = 'Version' value = '4.0' ).
    me->set_attr( node = mr_comp name = 'Fecha'   value = lv_fecha ).

    "---- Moneda (MONEDAXML/WAERK/WAERS) -------------------------------------
    CLEAR lv_mon.
    ASSIGN COMPONENT 'MONEDAXML' OF STRUCTURE lw_head TO <f_cur>.
    IF sy-subrc = 0 AND <f_cur> IS ASSIGNED AND <f_cur> IS NOT INITIAL.
      lv_mon = <f_cur>.
    ELSE.
      ASSIGN COMPONENT 'WAERK' OF STRUCTURE lw_head TO <f_cur>.
      IF sy-subrc = 0 AND <f_cur> IS ASSIGNED AND <f_cur> IS NOT INITIAL.
        lv_mon = <f_cur>.
      ELSE.
        ASSIGN COMPONENT 'WAERS' OF STRUCTURE lw_head TO <f_cur>.
        IF sy-subrc = 0 AND <f_cur> IS ASSIGNED AND <f_cur> IS NOT INITIAL.
          lv_mon = <f_cur>.
        ENDIF.
      ENDIF.
    ENDIF.
    IF lv_mon IS NOT INITIAL.
      me->set_attr( node = mr_comp name = 'Moneda' value = lv_mon ).
    ENDIF.

    "---- TipoCambio (si viene y moneda <> MXN) -------------------------------
    ASSIGN COMPONENT 'TIPO_CAMBIO' OF STRUCTURE lw_head TO <f_tc>.
    IF sy-subrc = 0 AND <f_tc> IS ASSIGNED AND <f_tc> IS NOT INITIAL AND lv_mon <> 'MXN'.
      lv_tmp_c = <f_tc>.
      REPLACE ALL OCCURRENCES OF ',' IN lv_tmp_c WITH '.'.
      CONDENSE lv_tmp_c NO-GAPS.
      lv_tc_p  = lv_tmp_c.
      lv_tc_s  = |{ lv_tc_p DECIMALS = 6 }|.
      REPLACE ALL OCCURRENCES OF ',' IN lv_tc_s WITH '.'.
      me->set_attr( node = mr_comp name = 'TipoCambio' value = lv_tc_s ).
    ENDIF.

    "---- LugarExpedicion / TipoDeComprobante --------------------------------
    ASSIGN COMPONENT 'LUGAR_EXP' OF STRUCTURE lw_head TO <f_lug>.
    IF sy-subrc = 0 AND <f_lug> IS ASSIGNED AND <f_lug> IS NOT INITIAL.
      me->set_attr( node = mr_comp name = 'LugarExpedicion' value = <f_lug> ).
    ENDIF.

    ASSIGN COMPONENT 'TIPO_COMP' OF STRUCTURE lw_head TO <f_tip>.
    IF sy-subrc = 0 AND <f_tip> IS ASSIGNED AND <f_tip> IS NOT INITIAL.
      me->set_attr( node = mr_comp name = 'TipoDeComprobante' value = <f_tip> ).
    ENDIF.

    "---- Exportacion (alias; default '01') -----------------------------------
    CLEAR lv_exp.
    ASSIGN COMPONENT 'EXPORTACION'         OF STRUCTURE lw_head TO <f_exp>.
    IF sy-subrc <> 0 OR <f_exp> IS INITIAL.
      ASSIGN COMPONENT 'IND_EXPORTACION'   OF STRUCTURE lw_head TO <f_exp>.
    ENDIF.
    IF sy-subrc <> 0 OR <f_exp> IS INITIAL.
      ASSIGN COMPONENT 'CLAVE_EXPORTACION' OF STRUCTURE lw_head TO <f_exp>.
    ENDIF.
    IF sy-subrc <> 0 OR <f_exp> IS INITIAL.
      ASSIGN COMPONENT 'ES_EXPORTACION'    OF STRUCTURE lw_head TO <f_exp>.
    ENDIF.
    IF sy-subrc = 0 AND <f_exp> IS ASSIGNED AND <f_exp> IS NOT INITIAL.
      lv_exp = <f_exp>.
    ELSE.
      lv_exp = '01'.
    ENDIF.
    me->set_attr( node = mr_comp name = 'Exportacion' value = lv_exp ).

    "---- Método/forma/condiciones -------------------------------------------
    ASSIGN COMPONENT 'METODO_PAGO'      OF STRUCTURE lw_head TO <f_met>.
    IF sy-subrc = 0 AND <f_met> IS ASSIGNED AND <f_met> IS NOT INITIAL.
      me->set_attr( node = mr_comp name = 'MetodoPago' value = <f_met> ).
    ENDIF.

    ASSIGN COMPONENT 'FORMA_PAGO'       OF STRUCTURE lw_head TO <f_for>.
    IF sy-subrc = 0 AND <f_for> IS ASSIGNED AND <f_for> IS NOT INITIAL.
      me->set_attr( node = mr_comp name = 'FormaPago' value = <f_for> ).
    ENDIF.

    ASSIGN COMPONENT 'CONDICIONES_PAGO' OF STRUCTURE lw_head TO <f_cond>.
    IF sy-subrc = 0 AND <f_cond> IS ASSIGNED AND <f_cond> IS NOT INITIAL.
      me->set_attr( node = mr_comp name = 'CondicionesDePago' value = <f_cond> ).
    ENDIF.

    "---- UsoCFDI (propagar) --------------------------------------------------
    ASSIGN COMPONENT 'USO_CFDI' OF STRUCTURE lw_head TO <f_uso>.
    IF sy-subrc = 0 AND <f_uso> IS ASSIGNED AND <f_uso> IS NOT INITIAL.
      cv_uso_cfdi = <f_uso>.
    ENDIF.
  ENDMETHOD.


* <SIGNATURE>---------------------------------------------------------------------------------------+
* | Instance Public Method ZCL_ZMFCFDI40_MGS->BUILD_IMPUESTOS
* +-------------------------------------------------------------------------------------------------+
* | [--->] IS_HEAD                        TYPE        LINE OF TT_HEAD40
* | [--->] IV_SUBTOTAL                    TYPE        TY_P14
* | [--->] IT_DET                         TYPE        TT_DET
* +--------------------------------------------------------------------------------------</SIGNATURE>
  METHOD build_impuestos.

    DATA: lr_impuestos TYPE REF TO if_ixml_element,
          lr_traslados TYPE REF TO if_ixml_element,
          lr_traslado  TYPE REF TO if_ixml_element,
          ls_det       TYPE ty_det,
          lv_total_iva TYPE ty_p14,
          lv_iva       TYPE ty_p14,
          lv_tasa      TYPE ty_p14.

    CONSTANTS: c_impuesto TYPE c LENGTH 3 VALUE '002',      " IVA
               c_tipo_fac TYPE c LENGTH 10 VALUE 'Tasa'.    " TipoFactor

    LOOP AT it_det INTO ls_det.
      IF ls_det-tasa_iva IS NOT INITIAL AND ls_det-importe IS NOT INITIAL.
        lv_iva = ls_det-importe * ls_det-tasa_iva.
        lv_total_iva = lv_total_iva + lv_iva.
      ENDIF.
    ENDLOOP.

    IF lv_total_iva > 0.
      lr_impuestos = mr_doc->create_simple_element( name = 'cfdi:Impuestos' parent = mr_comp ).
      me->set_attr( node = lr_impuestos name = 'TotalImpuestosTrasladados' value = me->to_str( iv_val = lv_total_iva iv_dec = 2 ) ).

      lr_traslados = mr_doc->create_simple_element( name = 'cfdi:Traslados' parent = lr_impuestos ).
      lr_traslado  = mr_doc->create_simple_element( name = 'cfdi:Traslado' parent = lr_traslados ).

      lv_tasa = ls_det-tasa_iva.

      me->set_attr( node = lr_traslado name = 'Base'        value = me->to_str( iv_val = lv_total_iva iv_dec = 2 ) ).
      me->set_attr( node = lr_traslado name = 'Impuesto'    value = c_impuesto ).
      me->set_attr( node = lr_traslado name = 'TipoFactor'  value = c_tipo_fac ).
      me->set_attr( node = lr_traslado name = 'TasaOCuota'  value = me->to_str( iv_val = lv_tasa iv_dec = 6 ) ).
      me->set_attr( node = lr_traslado name = 'Importe'     value = me->to_str( iv_val = lv_total_iva iv_dec = 2 ) ).
    ENDIF.

  ENDMETHOD.

* <SIGNATURE>---------------------------------------------------------------------------------------+
* | Instance Public Method ZCL_ZMFCFDI40_MGS->GET_DATA_FACTURA
* +-------------------------------------------------------------------------------------------------+
* | [--->] IV_VBELN                       TYPE        VBELN
* | [--->] IV_FECHADOC                    TYPE        SY-DATUM
* | [<---] EV_VGJAHR                      TYPE        GJAHR
* | [<---] EV_ERROR                       TYPE        STRING
* | [<---] EV_PEDIDO                      TYPE        VBELN
* | [<-->] CT_HEAD_CFDI                   TYPE        TT_HEAD
* | [<-->] CT_DET_CFDI                    TYPE        TT_DET
* +--------------------------------------------------------------------------------------</SIGNATURE>
  METHOD get_data_factura.

    CLEAR: ev_error, ev_vgjahr, ev_pedido.
    REFRESH: ct_head_cfdi, ct_det_cfdi.

    " Estructuras de trabajo
    DATA: ls_vbrk           TYPE vbrk,
          lt_header         TYPE TABLE OF vbrkvb,
          lt_positions      TYPE TABLE OF vbrpvb,
          lt_pos_price_cond TYPE TABLE OF komv,
          lt_partners       TYPE TABLE OF vbpavb,
          ls_header         TYPE vbrkvb,
          wa_t001           TYPE t001,
          wa_adrc           TYPE adrc,
          l_exch_rate       TYPE bapi1093_0,
          l_return          TYPE bapiret1,
          pi_fcurr_des      TYPE waers VALUE 'MXN',
          lv_rfc_rec        TYPE stcd1,
          lv_rfc_em         TYPE stcd1,
          lv_tc             TYPE ukurs,
          lv_name1          TYPE kna1-name1,
          ls_head           TYPE ty_head,
          ls_det            TYPE ty_det.

    " Leer documento de facturación
    ls_vbrk-vbeln = iv_vbeln.
    CALL FUNCTION 'RV_INVOICE_DOCUMENT_READ'
      EXPORTING
        konv_read = abap_true
        no_nast   = space
        vbrk_i    = ls_vbrk
      TABLES
        xkomv     = lt_pos_price_cond
        xvbpa     = lt_partners
        xvbrk     = lt_header
        xvbrp     = lt_positions.

    READ TABLE lt_header INTO ls_header WITH KEY vbeln = iv_vbeln.
    IF sy-subrc <> 0.
      ev_error = |No se encontró cabecera de factura { iv_vbeln }|.
      RETURN.
    ENDIF.

    " Datos de sociedad y dirección
    SELECT SINGLE * INTO wa_t001 FROM t001 WHERE bukrs = ls_header-bukrs.
    IF sy-subrc = 0 AND wa_t001-adrnr IS NOT INITIAL.
      SELECT SINGLE * INTO wa_adrc FROM adrc WHERE addrnumber = wa_t001-adrnr.
    ENDIF.

    " RFC receptor y nombre
    SELECT SINGLE stcd1 name1 INTO (lv_rfc_rec, lv_name1)
      FROM kna1
      WHERE kunnr = ls_header-kunag.

    " RFC emisor parametrizado
    SELECT SINGLE paval INTO lv_rfc_em
      FROM t001z
      WHERE bukrs = ls_header-bukrs
        AND party = 'MX_RFC'.

    " Tipo de cambio
    IF ls_header-waerk = 'MXN'.
      lv_tc = 1.
    ELSE.
      CALL FUNCTION 'BAPI_EXCHANGERATE_GETDETAIL'
        EXPORTING
          rate_type  = 'M'
          from_curr  = ls_header-waerk
          to_currncy = pi_fcurr_des
          date       = iv_fechadoc
        IMPORTING
          exch_rate  = l_exch_rate
          return     = l_return.
      lv_tc = COND #( WHEN sy-subrc = 0 AND l_exch_rate-exch_rate IS NOT INITIAL
                      THEN l_exch_rate-exch_rate
                      ELSE 1 ).
    ENDIF.

    " Construir HEAD CFDI
    CLEAR ls_head.
    ls_head-tipo_comp       = 'I'.
    ls_head-bukrs           = ls_header-bukrs.
    ls_head-belnr           = ls_header-vbeln.
    ls_head-bldat           = ls_header-fkdat.
    ls_head-gjahr           = ls_header-fkdat(4).
    ls_head-tipo_doc        = 'FF'.
    ls_head-uzeit           = sy-uzeit.
    ls_head-lugar_exp       = wa_adrc-post_code1.
    ls_head-total           = ls_header-netwr + ls_header-mwsbk.
    ls_head-waers           = ls_header-waerk.
    ls_head-total_iva       = ls_header-mwsbk.
    ls_head-subtotal        = ls_header-netwr.
    ls_head-version         = '4.0'.
    ls_head-regimen_fiscal  = '601'.
    ls_head-kunnr           = ls_header-kunag.
    ls_head-nombre_emisor   = wa_t001-butxt.
    ls_head-rfc_emisor      = lv_rfc_em.
    ls_head-nombre_receptor = lv_name1.
    ls_head-rfc_receptor    = lv_rfc_rec.
    ls_head-factura         = ls_header-belnr.
    ls_head-uso_cfdi        = 'G03'.
    ls_head-vkorg           = ls_header-vkorg.
    ls_head-vtweg           = ls_header-vtweg.
    ls_head-spart           = ls_header-spart.
    ls_head-tipo_cambio     = lv_tc.

    APPEND ls_head TO ct_head_cfdi.
    ev_vgjahr = ls_head-gjahr.

    " Determinar pedido de referencia
    FIELD-SYMBOLS <ls_pos> TYPE vbrpvb.
    SORT lt_positions BY posnr.
    LOOP AT lt_positions ASSIGNING <ls_pos>.
      IF <ls_pos>-aubel IS NOT INITIAL.
        ev_pedido = <ls_pos>-aubel.
        EXIT.
      ENDIF.
    ENDLOOP.
    IF ev_pedido IS INITIAL.
      READ TABLE lt_positions ASSIGNING <ls_pos> INDEX 1.
      IF sy-subrc = 0 AND <ls_pos>-aubel IS NOT INITIAL.
        ev_pedido = <ls_pos>-aubel.
      ENDIF.
    ENDIF.

    " Construir DET CFDI
    LOOP AT lt_positions ASSIGNING <ls_pos>.
      DATA(ls_kposn) = <ls_pos>-posnr.
      DATA ls_kmwst TYPE komv.
      READ TABLE lt_pos_price_cond INTO ls_kmwst
           WITH KEY kposn = ls_kposn kschl = 'MWST'.

      ls_det = VALUE ty_det(
        cantidad           = <ls_pos>-fkimg
        importe            = <ls_pos>-netwr
        valor_unitario     = COND #( WHEN <ls_pos>-fkimg IS NOT INITIAL
                                     THEN <ls_pos>-netwr / <ls_pos>-fkimg
                                     ELSE 0 )
        descripcion        = <ls_pos>-arktx
        clave_unidad       = ''
        clave_prodserv     = ''
        base_impuesto      = <ls_pos>-netwr
        importe_impuesto   = COND #( WHEN sy-subrc = 0 AND <ls_pos>-netwr > 0
                                     THEN ls_kmwst-kwert
                                     ELSE 0 )
        tasa_iva           = COND #( WHEN sy-subrc = 0 AND <ls_pos>-netwr > 0
                                     THEN ls_kmwst-kwert / <ls_pos>-netwr
                                     ELSE 0 )
        tipo_impuesto      = '002'
        tipo_factor        = 'Tasa' ).

      APPEND ls_det TO ct_det_cfdi.
    ENDLOOP.

  ENDMETHOD.

* <SIGNATURE>---------------------------------------------------------------------------------------+
* | Instance Public Method ZCL_ZMFCFDI40_MGS->LLENA_TABLA_PRINCIPAL_FM
* +-------------------------------------------------------------------------------------------------+
* | [--->] IV_VBELN                       TYPE        VBELN
* | [<---] ET_HEAD_CFDI                   TYPE        TT_HEAD
* | [<---] ET_DET_CFDI                    TYPE        TT_DET
* | [<-->] EV_ERROR                       TYPE        STRING
* | [<-->] EV_VGJAHR                      TYPE        GJAHR
* | [<-->] EV_PEDIDO                      TYPE        VBELN
* +--------------------------------------------------------------------------------------</SIGNATURE>
  METHOD llena_tabla_principal_fm.

    " Declaración de estructuras locales
    DATA: ls_vbrk           TYPE vbrk,
          lt_header         TYPE TABLE OF vbrkvb,
          lt_positions      TYPE TABLE OF vbrpvb,
          lt_pos_price_cond TYPE TABLE OF komv,
          lt_partners       TYPE TABLE OF vbpavb,
          ls_header         TYPE vbrkvb.
    DATA ev_first_sales_order TYPE vbeln.
    DATA ev_collect_positions TYPE mrnkz.
    DATA ev_purchase_no TYPE bstkd.
    FIELD-SYMBOLS <ls_pos> TYPE vbrpvb.

    " Inicialización
    CLEAR: ev_first_sales_order, ev_collect_positions, ev_purchase_no, ev_error.
    REFRESH: lt_header, lt_positions, lt_pos_price_cond, lt_partners.

    " Asignar número de factura
    ls_vbrk-vbeln = iv_vbeln.

    " Leer documento de facturación
    CALL FUNCTION 'RV_INVOICE_DOCUMENT_READ'
      EXPORTING
        konv_read = abap_true
        no_nast   = space
        vbrk_i    = ls_vbrk
      TABLES
        xvbrk     = lt_header
        xvbrp     = lt_positions
        xkomv     = lt_pos_price_cond
        xvbpa     = lt_partners.

    " Validar lectura de cabecera
    READ TABLE lt_header INTO ls_header WITH KEY vbeln = iv_vbeln.
    IF sy-subrc <> 0.
      ev_error = |No se encontró cabecera de factura { iv_vbeln }|.
      RETURN.
    ENDIF.

    " Buscar primer pedido de ventas relacionado
    SORT lt_positions BY posnr.
    LOOP AT lt_positions ASSIGNING <ls_pos>.
      IF <ls_pos>-aubel IS NOT INITIAL.
        ev_first_sales_order = <ls_pos>-aubel.
        EXIT.
      ENDIF.
    ENDLOOP.

    " Si no se encontró, tomar el primero disponible
    IF ev_first_sales_order IS INITIAL.
      READ TABLE lt_positions ASSIGNING <ls_pos> INDEX 1.
      IF sy-subrc = 0 AND <ls_pos>-aubel IS NOT INITIAL.
        ev_first_sales_order = <ls_pos>-aubel.
      ENDIF.
    ENDIF.

    " Obtener datos del pedido de ventas
    IF ev_first_sales_order IS NOT INITIAL.
      SELECT SINGLE mrnkz bstkd
        INTO (ev_collect_positions, ev_purchase_no)
        FROM vbkd
        WHERE vbeln = ev_first_sales_order.
    ENDIF.

  ENDMETHOD.

* <SIGNATURE>---------------------------------------------------------------------------------------+
* | Instance Public Method ZCL_ZMFCFDI40_MGS->PROCESAR_SALIDA_MENSAJE
* +-------------------------------------------------------------------------------------------------+
* | [--->] IV_OBJKEY                      TYPE        NAST-OBJKY
* | [--->] IV_NACHA                       TYPE        NAST-NACHA
* | [--->] IV_SPRAS                       TYPE        SPRAS
* | [<---] EV_ERROR                       TYPE        STRING
* +--------------------------------------------------------------------------------------</SIGNATURE>
METHOD procesar_salida_mensaje.

  " 🔹 Inicializar estado
  me->mv_success = abap_false.

  DATA: lv_error     TYPE string,
        lo_logger    TYPE REF TO zcl_logger,
        lx_root      TYPE REF TO cx_root,
        lv_success   TYPE abap_bool.

  CONSTANTS: c_log_objt   TYPE balobj_d   VALUE 'ZCFDI',
             c_log_subobj TYPE balsubobj  VALUE 'TIMBRADO'.

  " 🔹 Crear logger
  TRY.
      CREATE OBJECT lo_logger
        EXPORTING
          iv_extnumber = iv_objkey
          iv_object    = c_log_objt
          iv_subobject = c_log_subobj
          iv_repid     = sy-repid.

      lo_logger->add_text( EXPORTING iv_type = 'I' iv_text = |Inicio de procesamiento para { iv_objkey }| ).
      lo_logger->flush_to_slg1( ).

    CATCH cx_root INTO lx_root.
      " Si falla el logger, continuar sin trazabilidad
  ENDTRY.

  " 🔹 Validaciones previas
  IF iv_objkey IS INITIAL.
    lv_error = 'Clave de objeto vacía. No se puede continuar.'.
    ev_error = lv_error.
    me->mv_success = abap_false. " 🔸 Asegurar estado de fallo
    IF lo_logger IS BOUND.
      lo_logger->add_text( EXPORTING iv_type = 'E' iv_text = lv_error ).
      lo_logger->flush_to_slg1( ).
    ENDIF.
    RETURN.
  ENDIF.

  " 🔹 Procesamiento principal
  TRY.
      " Aquí iría la lógica real de timbrado
      IF iv_objkey CP '000*'.
        RAISE EXCEPTION TYPE cx_sy_no_handler
          EXPORTING textid = 'Error funcional simulado'.
      ENDIF.

      " Simulación de éxito
      lv_success = abap_true.
      me->mv_success = abap_true. " 🔸 Marcar éxito

    CATCH cx_root INTO lx_root.
      lv_error = lx_root->get_text( ).
      ev_error = lv_error.
      me->mv_success = abap_false. " 🔸 Marcar fallo
      IF lo_logger IS BOUND.
        lo_logger->add_text( EXPORTING iv_type = 'E' iv_text = |Excepción atrapada: { lv_error }| ).
        lo_logger->flush_to_slg1( ).
      ENDIF.
      RETURN.
  ENDTRY.

  " 🔹 Evaluación final
  IF me->mv_success = abap_true.
    ev_error = ''.
    IF lo_logger IS BOUND.
      lo_logger->add_text( EXPORTING iv_type = 'S' iv_text = |Procesamiento exitoso.| ).
      lo_logger->flush_to_slg1( ).
    ENDIF.
  ELSE.
    IF ev_error IS INITIAL.
      lv_error = 'Proceso finalizó sin éxito, sin excepción ni error explícito.'.
      ev_error = lv_error.
    ENDIF.
    IF lo_logger IS BOUND.
      lo_logger->add_text( EXPORTING iv_type = 'E' iv_text = |Error funcional: { ev_error }| ).
      lo_logger->flush_to_slg1( ).
    ENDIF.
  ENDIF.

  " 🔹 Finalizar
  IF lo_logger IS BOUND.
    lo_logger->add_text( EXPORTING iv_type = 'I' iv_text = |Fin de procesamiento para { iv_objkey }| ).
    lo_logger->flush_to_slg1( ).
  ENDIF.

ENDMETHOD.

* <SIGNATURE>---------------------------------------------------------------------------------------+
* | Instance Public Method ZCL_ZMFCFDI40_MGS->TIMBRAR
* +-------------------------------------------------------------------------------------------------+
* | [--->] T_HEAD_CFDI                    TYPE        TT_HEAD
* | [--->] T_DET_CFDI                     TYPE        TT_DET
* | [--->] I_ADDENDA                      TYPE        STRING
* | [--->] I_PEDIDO                       TYPE        VBELN
* | [--->] IV_FIX_IMPORTES                TYPE        ABAP_BOOL
* | [--->] IV_FIX_FECHA                   TYPE        ABAP_BOOL
* | [<---] E_UUID                         TYPE        STRING
* | [<---] E_URL                          TYPE        STRING
* | [<---] E_ERROR                        TYPE        STRING
* | [<-->] PO_LOGGER                      TYPE REF TO ZCL_LOGGER
* +--------------------------------------------------------------------------------------</SIGNATURE>
METHOD timbrar.

  DATA: ls_head       TYPE LINE OF tt_head,
        ls_head40       TYPE LINE OF tt_head40,
        lt_det        TYPE tt_det,
        lv_subtot     TYPE ty_p14,
        lv_dom_string TYPE string,
        lv_dat_string TYPE string,
        lv_xml_string TYPE string,
        lv_uso_cfdi   TYPE ty_char3,
        lv_pac_out    TYPE string,
        lv_err_s      TYPE string,
        lv_is_prod    TYPE abap_bool.

  CLEAR: e_uuid, e_url, e_error.

  " Validar encabezado CFDI
  READ TABLE t_head_cfdi INTO ls_head INDEX 1.
  IF sy-subrc <> 0.
    e_error = 'T_HEAD_CFDI vacío'.
    RETURN.
  ENDIF.

  lt_det = t_det_cfdi.

  " Inicializar motor XML
  ixml_init( ).

  " Construir nodos CFDI
  build_header(
    EXPORTING is_head = ls_head40
              iv_fix_fecha = iv_fix_fecha
    CHANGING  cv_uso_cfdi = lv_uso_cfdi ).

  build_emisor( EXPORTING is_head = ls_head40 ).
  build_receptor( EXPORTING is_head = ls_head40 ).

  CLEAR lv_subtot.
  build_conceptos(
    EXPORTING is_head        = ls_head40
              it_det         = lt_det
              iv_ver         = abap_true
              iv_fix_import  = iv_fix_importes
    CHANGING  cv_subtotal    = lv_subtot ).

  build_impuestos(
    EXPORTING is_head     = ls_head40
              iv_subtotal = lv_subtot
              it_det      = lt_det ).

  build_custom_data(
    EXPORTING is_head       = ls_head40
              it_det        = lt_det
    IMPORTING ev_dom_string = lv_dom_string
              ev_dat_string = lv_dat_string ).

  maybe_addenda(
    EXPORTING is_head  = ls_head40
              i_addenda = i_addenda ).

  render_to_string(
    EXPORTING io_doc = mr_doc
    IMPORTING ev_xml = lv_xml_string ).

  " Validar XML generado
  IF lv_xml_string IS INITIAL.
    e_error = 'No se generó XML CFDI válido'.
    RETURN.
  ENDIF.

  " Enviar a PAC
  send_to_pac(
    EXPORTING is_head = ls_head40
              iv_xml  = lv_xml_string
              iv_dom  = lv_dom_string
              iv_dat  = lv_dat_string
    IMPORTING es_output   = lv_pac_out
              ev_error_s  = lv_err_s
              ev_is_prod  = lv_is_prod ).

  " Validar respuesta del PAC
  IF lv_err_s IS NOT INITIAL.
    e_error = |Error al timbrar: { lv_err_s }|.
    RETURN.
  ENDIF.

  " Asignar resultados
  e_uuid = lv_pac_out.
  e_url  = |https://cfdi.sat.gob.mx/xml/{ lv_pac_out }.xml|.

  " Registrar en logger si está disponible
  IF po_logger IS BOUND.
    po_logger->add_text( EXPORTING iv_type = 'S' iv_text = |UUID generado: { e_uuid }| ).
    po_logger->add_text( EXPORTING iv_type = 'S' iv_text = |URL XML: { e_url }| ).
    po_logger->add_text( EXPORTING iv_type = 'I' iv_text = |XML generado: { lv_xml_string }| ).
  ENDIF.

ENDMETHOD.


* <SIGNATURE>---------------------------------------------------------------------------------------+
* | Instance Public Method ZCL_ZMFCFDI40_MGS->TO_STR
* +-------------------------------------------------------------------------------------------------+
* | [--->] IV_VAL                         TYPE        TY_P14
* | [--->] IV_DEC                         TYPE        I
* | [<-()] RV_STR                         TYPE        STRING
* +--------------------------------------------------------------------------------------</SIGNATURE>
  METHOD to_str.

    DATA: lv_d   TYPE i,
          lv_num TYPE ty_p14,
          lv_buf TYPE c LENGTH 120,
          lv_str TYPE string.

    lv_d = iv_dec.
    IF lv_d < 0.
      lv_d = 0.
    ELSEIF lv_d > 14.
      lv_d = 14.
    ENDIF.

    lv_num = iv_val.
    CLEAR lv_buf.
    WRITE lv_num TO lv_buf DECIMALS lv_d NO-GROUPING.
    lv_str = lv_buf.

    SHIFT lv_str LEFT  DELETING LEADING space.
    SHIFT lv_str RIGHT DELETING TRAILING space.
    REPLACE ALL OCCURRENCES OF ',' IN lv_str WITH '.'.
    REPLACE ALL OCCURRENCES OF ' ' IN lv_str WITH ''.

    rv_str = lv_str.

  ENDMETHOD.

    DATA: lw_header    TYPE thead,
          lw_line      TYPE tline,
          lt_lines     TYPE STANDARD TABLE OF tline,
          lv_documento TYPE tdobname,
          lv_subrc     TYPE sy-subrc,
          lv_line      TYPE string.

    CLEAR: ev_success, ev_message.
    ev_success = abap_true.



    " Validación de entrada
    IF iv_bukrs IS INITIAL OR iv_belnr IS INITIAL OR iv_gjahr IS INITIAL OR iv_uuid IS INITIAL.
      ev_message = 'Error: parámetros obligatorios vacíos (BUKRS, BELNR, GJAHR, UUID)'.
      RETURN.
    ENDIF.

    lv_documento = |{ iv_bukrs }{ iv_belnr }{ iv_gjahr }|.

    " Encabezado de texto
    IF iv_tdobject IS INITIAL.
      lw_header-tdobject = 'BELEG'.
      lw_header-tdid     = 'YUUD'.
      lw_header-tdname   = lv_documento.
    ELSE.
      lw_header-tdobject = iv_tdobject.
      lw_header-tdid     = 'ZUUI'.
      lw_header-tdname   = iv_belnr.
    ENDIF.
    lw_header-tdspras = sy-langu.

    " Línea de texto
    lw_line-tdformat = 'ST'.
    lw_line-tdline   = iv_uuid.
    APPEND lw_line TO lt_lines.

    " Guardar texto
    CALL FUNCTION 'SAVE_TEXT'
      EXPORTING
        client          = sy-mandt
        header          = lw_header
        insert          = 'X'
        savemode_direct = 'X'
      TABLES
        lines           = lt_lines
      EXCEPTIONS
        id              = 1
                          language= 2
        name            = 3
        object          = 4
        OTHERS          = 5.

    lv_subrc = sy-subrc.
    lv_line = |IV_BELNR: { iv_belnr }|.

    IF lv_subrc = 0.
      CALL FUNCTION 'COMMIT_TEXT'.
      ev_success = abap_true.
      ev_message = |Texto guardado:{ cl_abap_char_utilities=>newline }{ lv_line }|.
    ELSE.
      CASE lv_subrc.
        WHEN 1. ev_message = |Error: ID de texto no válido.|.
        WHEN 2. ev_message = |Error: idioma no válido.|.
        WHEN 3. ev_message = |Error: nombre de texto no válido.|.
        WHEN 4. ev_message = |Error: objeto de texto no válido.|.
        WHEN OTHERS. ev_message = |Error desconocido al guardar el texto.|.
      ENDCASE.

      ev_message = ev_message &&
                   cl_abap_char_utilities=>newline && lv_line &&
                   cl_abap_char_utilities=>newline && |Año: { iv_gjahr }| &&
                   cl_abap_char_utilities=>newline && |Bukrs: { iv_bukrs }| &&
                   cl_abap_char_utilities=>newline && |UUID: { iv_uuid }|.
    ENDIF.

  ENDMETHOD.
ENDCLASS.

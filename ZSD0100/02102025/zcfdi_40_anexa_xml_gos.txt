FUNCTION zcfdi_40_anexa_xml_gos.
*"----------------------------------------------------------------------
*"  IMPORTING
*"     VALUE(iv_bukrs)       TYPE bukrs
*"     VALUE(iv_belnr)       TYPE vbrk-vbeln
*"     VALUE(iv_gjahr)       TYPE gjahr
*"     VALUE(iv_tipo)        TYPE char2          " 'TI' 'RQ' 'ER' 'AL'
*"     VALUE(iv_descripcion) TYPE so_obj_des
*"     VALUE(iv_xml_str)     TYPE string OPTIONAL " XML timbrado ya armado (STRING)
*"  EXPORTING
"*     VALUE(ev_success)     TYPE flag
"*     VALUE(ev_message)     TYPE string
*"----------------------------------------------------------------------

  DATA: lv_vbeln10    TYPE vbrk-vbeln,
        lv_tipo       TYPE char2,
        lv_desc_base  TYPE so_obj_des,
        lv_any_done   TYPE abap_bool VALUE abap_false.

  " --- SAPoffice / GOS
  DATA: ls_fol_id   TYPE soodk,
        ls_obj_id   TYPE soodk,
        ls_obj_data TYPE sood1,
        lt_objhead  TYPE soli_tab,
        lt_soli     TYPE soli_tab,
        lt_solix    TYPE solix_tab,
        ls_object   TYPE borident,
        ls_note     TYPE borident,
        ls_folmem_k TYPE sofmk,
        lv_ep_note  TYPE borident-objkey.

  " --- Tablas Z
  DATA: ls_xml40 TYPE zcfdi_40_xml,
        lt_log   TYPE STANDARD TABLE OF zcfdi_xml_log,
        ls_log   TYPE zcfdi_xml_log,
        ls_head  TYPE zcfdi_head_40.

  FIELD-SYMBOLS: <fs_any>   TYPE any,
                 <ls_soli>  TYPE soli.

  CLEAR: ev_success, ev_message.

  "--------------------------
  " Validaciones básicas
  "--------------------------
  IF iv_bukrs IS INITIAL OR iv_belnr IS INITIAL OR iv_gjahr IS INITIAL.
    ev_message = 'Parámetros obligatorios vacíos (BUKRS/BELNR/GJAHR).'.
    ev_success = abap_false.
    RETURN.
  ENDIF.

  " VBELN a 10 (ALPHA IN)
  lv_vbeln10 = iv_belnr.
  CALL FUNCTION 'CONVERSION_EXIT_ALPHA_INPUT'
    EXPORTING  input  = lv_vbeln10
    IMPORTING  output = lv_vbeln10.

  " Normaliza tipo
  lv_tipo = iv_tipo.
  TRANSLATE lv_tipo TO UPPER CASE.
  IF lv_tipo IS INITIAL.
    lv_tipo = 'AL'.  " adjunta el primero que encuentre (TI->RQ->ER)
  ENDIF.

  " Descripción base
  IF iv_descripcion IS INITIAL.
    CONCATENATE 'XML CFDI' lv_vbeln10 INTO lv_desc_base SEPARATED BY space.
  ELSE.
    lv_desc_base = iv_descripcion.
  ENDIF.

  " Carpeta SAPoffice (B = Private)
  CALL FUNCTION 'SO_FOLDER_ROOT_ID_GET'
    EXPORTING  region    = 'B'
    IMPORTING  folder_id = ls_fol_id.
  IF sy-subrc <> 0 OR ls_fol_id-objtp IS INITIAL.
    ev_message = 'No se obtuvo carpeta SAPoffice (SO_FOLDER_ROOT_ID_GET).'.
    ev_success = abap_false.
    RETURN.
  ENDIF.

  "--------------------------
  " 1) Construir contenido (lt_soli)
  "    Preferencia: IV_XML_STR; si vacío, BD según IV_TIPO
  "--------------------------
  REFRESH lt_soli.

  " A) Si viene STRING por parámetro, úsalo tal cual
  IF iv_xml_str IS NOT INITIAL.
    CALL FUNCTION 'SO_STRING_TO_TAB'
      EXPORTING  content_str = iv_xml_str
      TABLES     content_tab = lt_soli.
    IF lt_soli IS NOT INITIAL.
      lv_any_done = abap_true.
    ENDIF.
  ENDIF.

  " B) TI (Timbrado) si lo pide el tipo o 'AL'
  IF lv_any_done = abap_false AND ( lv_tipo = 'TI' OR lv_tipo = 'AL' ).
    " ZCFDI_40_XML (STRING)
    CLEAR ls_xml40.
    SELECT SINGLE * INTO ls_xml40 FROM zcfdi_40_xml WHERE factura = lv_vbeln10.
    IF sy-subrc = 0 AND ls_xml40-xml IS NOT INITIAL.
      CALL FUNCTION 'SO_STRING_TO_TAB'
        EXPORTING  content_str = ls_xml40-xml
        TABLES     content_tab = lt_soli.
      IF lt_soli IS NOT INITIAL.
        lv_any_done = abap_true.
      ENDIF.
    ENDIF.

    " Fallback: ZCFDI_XML_LOG tipo 'TI' (XSTRING)
    IF lv_any_done = abap_false.
      REFRESH lt_log.
      SELECT * INTO TABLE lt_log
        FROM zcfdi_xml_log
        WHERE bukrs = iv_bukrs
          AND belnr = lv_vbeln10
          AND gjahr = iv_gjahr
          AND tipo_doc = 'TI'.
      IF sy-subrc = 0.
        SORT lt_log BY created_on DESCENDING created_at DESCENDING.
        READ TABLE lt_log INTO ls_log INDEX 1.
        IF sy-subrc = 0 AND ls_log-xml_req IS NOT INITIAL.
          REFRESH lt_solix.
          CALL FUNCTION 'SCMS_XSTRING_TO_BINARY'
            EXPORTING  buffer     = ls_log-xml_req
            TABLES     binary_tab = lt_solix.
          CALL FUNCTION 'SO_SOLIXTAB_TO_SOLITAB'
            EXPORTING  ip_solixtab = lt_solix
            IMPORTING  ep_solitab  = lt_soli.
          IF lt_soli IS NOT INITIAL.
            lv_any_done = abap_true.
          ENDIF.
        ENDIF.
      ENDIF.
    ENDIF.

    " Fallback: ZCFDI_HEAD_40 (prueba campos típicos como string/xstring)
    IF lv_any_done = abap_false.
      CLEAR ls_head.
      SELECT SINGLE * INTO ls_head
        FROM zcfdi_head_40
        WHERE bukrs = iv_bukrs
          AND belnr = lv_vbeln10
          AND gjahr = iv_gjahr.
      IF sy-subrc = 0.
        " Intento 1: XML_TIMB
        ASSIGN COMPONENT 'XML_TIMB' OF STRUCTURE ls_head TO <fs_any>.
        IF <fs_any> IS ASSIGNED AND <fs_any> IS NOT INITIAL.
          DATA(lv_try_s1) = ``.
          TRY. lv_try_s1 = CONV string( <fs_any> ). CATCH cx_sy_move_cast_error cx_sy_conversion_error. ENDTRY.
          IF lv_try_s1 IS NOT INITIAL.
            CALL FUNCTION 'SO_STRING_TO_TAB' EXPORTING content_str = lv_try_s1 TABLES content_tab = lt_soli.
          ENDIF.
          IF lt_soli IS INITIAL.
            DATA(lv_try_x1) = CONV xstring( '' ).
            TRY. lv_try_x1 = CONV xstring( <fs_any> ). CATCH cx_sy_move_cast_error cx_sy_conversion_error. ENDTRY.
            IF lv_try_x1 IS NOT INITIAL.
              CALL FUNCTION 'SCMS_XSTRING_TO_BINARY' EXPORTING buffer = lv_try_x1 TABLES binary_tab = lt_solix.
              CALL FUNCTION 'SO_SOLIXTAB_TO_SOLITAB' EXPORTING ip_solixtab = lt_solix IMPORTING ep_solitab = lt_soli.
            ENDIF.
          ENDIF.
        ENDIF.

        " Intento 2: XML_TIMBRADO
        IF lt_soli IS INITIAL.
          ASSIGN COMPONENT 'XML_TIMBRADO' OF STRUCTURE ls_head TO <fs_any>.
          IF <fs_any> IS ASSIGNED AND <fs_any> IS NOT INITIAL.
            DATA(lv_try_s2) = ``.
            TRY. lv_try_s2 = CONV string( <fs_any> ). CATCH cx_sy_move_cast_error cx_sy_conversion_error. ENDTRY.
            IF lv_try_s2 IS NOT INITIAL.
              CALL FUNCTION 'SO_STRING_TO_TAB' EXPORTING content_str = lv_try_s2 TABLES content_tab = lt_soli.
            ENDIF.
            IF lt_soli IS INITIAL.
              DATA(lv_try_x2) = CONV xstring( '' ).
              TRY. lv_try_x2 = CONV xstring( <fs_any> ). CATCH cx_sy_move_cast_error cx_sy_conversion_error. ENDTRY.
              IF lv_try_x2 IS NOT INITIAL.
                CALL FUNCTION 'SCMS_XSTRING_TO_BINARY' EXPORTING buffer = lv_try_x2 TABLES binary_tab = lt_solix.
                CALL FUNCTION 'SO_SOLIXTAB_TO_SOLITAB' EXPORTING ip_solixtab = lt_solix IMPORTING ep_solitab = lt_soli.
              ENDIF.
            ENDIF.
          ENDIF.
        ENDIF.

        " Intento 3: XML_TFD
        IF lt_soli IS INITIAL.
          ASSIGN COMPONENT 'XML_TFD' OF STRUCTURE ls_head TO <fs_any>.
          IF <fs_any> IS ASSIGNED AND <fs_any> IS NOT INITIAL.
            DATA(lv_try_s3) = ``.
            TRY. lv_try_s3 = CONV string( <fs_any> ). CATCH cx_sy_move_cast_error cx_sy_conversion_error. ENDTRY.
            IF lv_try_s3 IS NOT INITIAL.
              CALL FUNCTION 'SO_STRING_TO_TAB' EXPORTING content_str = lv_try_s3 TABLES content_tab = lt_soli.
            ENDIF.
            IF lt_soli IS INITIAL.
              DATA(lv_try_x3) = CONV xstring( '' ).
              TRY. lv_try_x3 = CONV xstring( <fs_any> ). CATCH cx_sy_move_cast_error cx_sy_conversion_error. ENDTRY.
              IF lv_try_x3 IS NOT INITIAL.
                CALL FUNCTION 'SCMS_XSTRING_TO_BINARY' EXPORTING buffer = lv_try_x3 TABLES binary_tab = lt_solix.
                CALL FUNCTION 'SO_SOLIXTAB_TO_SOLITAB' EXPORTING ip_solixtab = lt_solix IMPORTING ep_solitab = lt_soli.
              ENDIF.
            ENDIF.
          ENDIF.
        ENDIF.

        " Intento 4: XML
        IF lt_soli IS INITIAL.
          ASSIGN COMPONENT 'XML' OF STRUCTURE ls_head TO <fs_any>.
          IF <fs_any> IS ASSIGNED AND <fs_any> IS NOT INITIAL.
            DATA(lv_try_s4) = ``.
            TRY. lv_try_s4 = CONV string( <fs_any> ). CATCH cx_sy_move_cast_error cx_sy_conversion_error. ENDTRY.
            IF lv_try_s4 IS NOT INITIAL.
              CALL FUNCTION 'SO_STRING_TO_TAB' EXPORTING content_str = lv_try_s4 TABLES content_tab = lt_soli.
            ENDIF.
            IF lt_soli IS INITIAL.
              DATA(lv_try_x4) = CONV xstring( '' ).
              TRY. lv_try_x4 = CONV xstring( <fs_any> ). CATCH cx_sy_move_cast_error cx_sy_conversion_error. ENDTRY.
              IF lv_try_x4 IS NOT INITIAL.
                CALL FUNCTION 'SCMS_XSTRING_TO_BINARY' EXPORTING buffer = lv_try_x4 TABLES binary_tab = lt_solix.
                CALL FUNCTION 'SO_SOLIXTAB_TO_SOLITAB' EXPORTING ip_solixtab = lt_solix IMPORTING ep_solitab = lt_soli.
              ENDIF.
            ENDIF.
          ENDIF.
        ENDIF.

        IF lt_soli IS NOT INITIAL.
          lv_any_done = abap_true.
        ENDIF.
      ENDIF.
    ENDIF.
  ENDIF.

  " C) RQ
  IF lv_any_done = abap_false AND ( lv_tipo = 'RQ' OR lv_tipo = 'AL' ).
    REFRESH lt_log.
    SELECT * INTO TABLE lt_log
      FROM zcfdi_xml_log
      WHERE bukrs = iv_bukrs
        AND belnr = lv_vbeln10
        AND gjahr = iv_gjahr
        AND tipo_doc = 'RQ'.
    IF sy-subrc = 0.
      SORT lt_log BY created_on DESCENDING created_at DESCENDING.
      READ TABLE lt_log INTO ls_log INDEX 1.
      IF sy-subrc = 0 AND ls_log-xml_req IS NOT INITIAL.
        REFRESH lt_solix.
        CALL FUNCTION 'SCMS_XSTRING_TO_BINARY'
          EXPORTING  buffer     = ls_log-xml_req
          TABLES     binary_tab = lt_solix.
        CALL FUNCTION 'SO_SOLIXTAB_TO_SOLITAB'
          EXPORTING  ip_solixtab = lt_solix
          IMPORTING  ep_solitab  = lt_soli.
        IF lt_soli IS NOT INITIAL.
          lv_any_done = abap_true.
        ENDIF.
      ENDIF.
    ENDIF.
  ENDIF.

  " D) ER
  IF lv_any_done = abap_false AND ( lv_tipo = 'ER' OR lv_tipo = 'AL' ).
    REFRESH lt_log.
    SELECT * INTO TABLE lt_log
      FROM zcfdi_xml_log
      WHERE bukrs = iv_bukrs
        AND belnr = lv_vbeln10
        AND gjahr = iv_gjahr
        AND tipo_doc = 'ER'.
    IF sy-subrc = 0.
      SORT lt_log BY created_on DESCENDING created_at DESCENDING.
      READ TABLE lt_log INTO ls_log INDEX 1.
      IF sy-subrc = 0 AND ls_log-xml_req IS NOT INITIAL.
        REFRESH lt_solix.
        CALL FUNCTION 'SCMS_XSTRING_TO_BINARY'
          EXPORTING  buffer     = ls_log-xml_req
          TABLES     binary_tab = lt_solix.
        CALL FUNCTION 'SO_SOLIXTAB_TO_SOLITAB'
          EXPORTING  ip_solixtab = lt_solix
          IMPORTING  ep_solitab  = lt_soli.
        IF lt_soli IS NOT INITIAL.
          lv_any_done = abap_true.
        ENDIF.
      ENDIF.
    ENDIF.
  ENDIF.

  IF lt_soli IS INITIAL.
    ev_success = abap_false.
    ev_message = |No hay XML para adjuntar (tipo={ lv_tipo }).|.
    RETURN.
  ENDIF.

  "--------------------------
  " 2) Insertar en SO + vincular a BUS2081 (SIN COMMIT)
  "--------------------------
  CLEAR ls_obj_data.
  ls_obj_data-objsns   = 'O'.
  ls_obj_data-objla    = sy-langu.
  ls_obj_data-objdes   = lv_desc_base.
  ls_obj_data-file_ext = 'XML'.

  " Tamaño real en bytes (Unicode-safe)
  DATA(lv_bytes) = 0.
  LOOP AT lt_soli ASSIGNING <ls_soli>.
    lv_bytes = lv_bytes + strlen( <ls_soli>-line ).
  ENDLOOP.
  ls_obj_data-objlen = lv_bytes.

  CALL FUNCTION 'SO_OBJECT_INSERT'
    EXPORTING
      folder_id        = ls_fol_id
      object_type      = 'RAW'
      object_hd_change = ls_obj_data
    IMPORTING
      object_id        = ls_obj_id
    TABLES
      objhead          = lt_objhead
      objcont          = lt_soli.
  IF sy-subrc <> 0.
    ev_success = abap_false.
    ev_message = 'Error al crear objeto SAPoffice (SO_OBJECT_INSERT).'.
    RETURN.
  ENDIF.

  " Mantener VBRK
  ls_object-objtype = 'VBRK'.
  ls_object-objkey  = lv_vbeln10.

  ls_folmem_k-foltp = ls_fol_id-objtp.
  ls_folmem_k-folyr = ls_fol_id-objyr.
  ls_folmem_k-folno = ls_fol_id-objno.
  ls_folmem_k-doctp = ls_obj_id-objtp.
  ls_folmem_k-docyr = ls_obj_id-objyr.
  ls_folmem_k-docno = ls_obj_id-objno.
  lv_ep_note        = ls_folmem_k.

  ls_note-objtype   = 'MESSAGE'.
  ls_note-objkey    = lv_ep_note.

  " SIN COMMIT aquí: que lo haga el llamador
  CALL FUNCTION 'BINARY_RELATION_CREATE'
    EXPORTING
      obj_rolea    = ls_object
      obj_roleb    = ls_note
      relationtype = 'ATTA'
    EXCEPTIONS
      no_model       = 1
      internal_error = 2
      unknown        = 3
      OTHERS         = 4.

  IF sy-subrc = 0.
    ev_success = abap_true.
    ev_message = |Adjunto GOS creado para { lv_vbeln10 } (tipo={ lv_tipo }).|.
  ELSE.
    ev_success = abap_false.
    ev_message = |Documento creado, pero falló relación GOS (ATTA).|.
  ENDIF.

ENDFUNCTION.

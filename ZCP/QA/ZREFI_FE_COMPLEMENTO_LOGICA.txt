FORM f_set_from_sources
  USING    iv_target_name   TYPE clike
           iv_source_names  TYPE clike
           cs_source        TYPE any
  CHANGING cs_target        TYPE any.

  FIELD-SYMBOLS: <ft> TYPE any, <fs> TYPE any.
  DATA: lt_names TYPE STANDARD TABLE OF string,
        lv_name  TYPE string.

  ASSIGN COMPONENT iv_target_name OF STRUCTURE cs_target TO <ft>.
  IF <ft> IS ASSIGNED.
    SPLIT iv_source_names AT '|' INTO TABLE lt_names.
    LOOP AT lt_names INTO lv_name.
      IF lv_name IS INITIAL. CONTINUE. ENDIF.
      ASSIGN COMPONENT lv_name OF STRUCTURE cs_source TO <fs>.
      IF sy-subrc = 0 AND <fs> IS ASSIGNED AND <fs> IS NOT INITIAL.
        <ft> = <fs>.
        EXIT.
      ENDIF.
    ENDLOOP.
  ENDIF.
ENDFORM.

FORM f_copy_if_new
  USING    iv_field_name TYPE clike
           cs_source     TYPE any
  CHANGING cs_target     TYPE any.

  FIELD-SYMBOLS: <f_t> TYPE any, <f_s> TYPE any.
  ASSIGN COMPONENT iv_field_name OF STRUCTURE cs_target TO <f_t>.
  ASSIGN COMPONENT iv_field_name OF STRUCTURE cs_source TO <f_s>.
  IF sy-subrc = 0 AND <f_t> IS ASSIGNED AND <f_s> IS ASSIGNED.
    IF <f_s> IS NOT INITIAL AND <f_t> <> <f_s>.
      <f_t> = <f_s>.
    ENDIF.
  ENDIF.
ENDFORM.

  DATA: lo_conv TYPE REF TO cl_abap_conv_out_ce,

FORM f_fill_text_est
  USING    iv_stat       TYPE zta1_cp1-estatus
           iv_text_stat  TYPE clike
  CHANGING cv_text_est   TYPE zta1_cp1-text_est.

  DATA: lv_txt     TYPE string,
        lv_stat_up TYPE string.

  CLEAR cv_text_est.

  IF iv_stat = gc_icon_green.
    cv_text_est = 'Timbrado'.
    RETURN.
  ENDIF.

  lv_txt = iv_text_stat.
  REPLACE ALL OCCURRENCES OF cl_abap_char_utilities=>cr_lf   IN lv_txt WITH space.
  REPLACE ALL OCCURRENCES OF cl_abap_char_utilities=>newline IN lv_txt WITH space.
  CONDENSE lv_txt.

  lv_stat_up = iv_stat.
  TRANSLATE lv_stat_up TO UPPER CASE.

  IF lv_stat_up CS 'VERDE'.
    IF lv_txt IS INITIAL.
      cv_text_est = 'Timbrado'.
    ELSE.
      cv_text_est = lv_txt.
    ENDIF.
  ELSE.
    IF lv_txt IS INITIAL.
      cv_text_est = 'Error timbrado (PAC)'.
    ELSE.
      CONCATENATE 'Error PAC:' lv_txt INTO cv_text_est SEPARATED BY space.
    ENDIF.
  ENDIF.
ENDFORM.

FORM f_build_success_template
  USING    it_succ TYPE STANDARD TABLE
  CHANGING ct_text TYPE soli_tab.

  DATA: ls_text     TYPE soli,
        lv_line     TYPE string,
        lv_sep      TYPE string,
        lv_avail    TYPE string,
        lv_pol      TYPE char35,
        lv_estok    TYPE string,
        lv_xml_path TYPE string,
        lv_pdf_path TYPE string.

  CLEAR ct_text.
  CLEAR lv_sep.
  DO 78 TIMES.
    CONCATENATE lv_sep '=' INTO lv_sep.
  ENDDO.

  ls_text-line = lv_sep.                                                   APPEND ls_text TO ct_text.
  ls_text-line = |  COMPLEMENTO DE PAGO TIMBRADO  |.                       APPEND ls_text TO ct_text.
  ls_text-line = lv_sep.                                                   APPEND ls_text TO ct_text.
  ls_text-line = | |.                                                      APPEND ls_text TO ct_text.
  ls_text-line = |Detalle por documento:|.                                 APPEND ls_text TO ct_text.
  ls_text-line = | |.                                                      APPEND ls_text TO ct_text.

  FIELD-SYMBOLS: <ls> TYPE ty_cp1_success.
  LOOP AT it_succ ASSIGNING <ls>.
    lv_line = |- Sociedad: { <ls>-bukrs }  Cliente: { <ls>-kunnr }  NumComp: { <ls>-numcomp }|.
    ls_text-line = lv_line.                                                APPEND ls_text TO ct_text.

    lv_line = |  Estatus: { <ls>-estatus }  UUID: { <ls>-uuid_comp }  Fecha Timb.: { <ls>-ztfetimb }|.
    ls_text-line = lv_line.                                                APPEND ls_text TO ct_text.

    IF <ls>-estatus = gc_icon_green.
      lv_estok = 'OK'.
    ELSE.
      lv_estok = 'ERROR'.
    ENDIF.
    ls_text-line = |  Estatus Timbrado: { lv_estok }|.                     APPEND ls_text TO ct_text.

    IF <ls>-xml_dispon = abap_true.  lv_avail = 'Disponible'. ELSE. lv_avail = 'No disponible'. ENDIF.
    ls_text-line = |  XML: { lv_avail }|.                                  APPEND ls_text TO ct_text.

    CLEAR: lv_xml_path, lv_pdf_path.
    lv_xml_path = <ls>-ruta_xml.
    lv_pdf_path = <ls>-ruta_pdf.
    IF lv_xml_path IS NOT INITIAL.
      PERFORM f_ensure_extension USING 'xml' CHANGING lv_xml_path.
    ENDIF.
    IF lv_pdf_path IS NOT INITIAL.
      PERFORM f_ensure_extension USING 'pdf' CHANGING lv_pdf_path.
    ENDIF.

    IF lv_xml_path IS INITIAL AND lv_pdf_path IS INITIAL.
      ls_text-line = |  Rutas -> XML: (N/A)  PDF: (N/A)|.
    ELSE.
      lv_line = |  Rutas -> XML: { lv_xml_path }  PDF: { lv_pdf_path }|.
      ls_text-line = lv_line.
    ENDIF.
    APPEND ls_text TO ct_text.

    PERFORM f_cp_pol_get USING <ls>-numcomp CHANGING lv_pol.
    IF lv_pol IS NOT INITIAL.
      ls_text-line = |  Actualización en Póliza: { lv_pol }|.              APPEND ls_text TO ct_text.
    ENDIF.

    ls_text-line = | |.                                                    APPEND ls_text TO ct_text.
  ENDLOOP.

  ls_text-line = lv_sep.                                                   APPEND ls_text TO ct_text.
  ls_text-line = |Nota: Puede abrir los PDF ahora o más tarde desde la opción de visualización.|. APPEND ls_text TO ct_text.
  ls_text-line = lv_sep.                                                   APPEND ls_text TO ct_text.
ENDFORM.

FORM f_parse_cfdi_fields_str
  USING    iv_xml_str     TYPE string
  CHANGING cv_uuid        TYPE string
           cv_fecha_timb  TYPE string
           cv_forma_pago  TYPE string
           cv_metodo_pago TYPE string
           cv_rfc_ord     TYPE string
           cv_no_cert     TYPE string.

  DATA lv_xml TYPE string.
  lv_xml = iv_xml_str.
  IF lv_xml IS INITIAL. RETURN. ENDIF.

  FIND REGEX 'UUID="([^"]+)"'            IN lv_xml SUBMATCHES cv_uuid.
  FIND REGEX 'FechaTimbrado="([^"]+)"'   IN lv_xml SUBMATCHES cv_fecha_timb.
  FIND REGEX 'FormaPago="([^"]+)"'       IN lv_xml SUBMATCHES cv_forma_pago.
  FIND REGEX 'MetodoPago="([^"]+)"'      IN lv_xml SUBMATCHES cv_metodo_pago.
  FIND REGEX 'RfcEmisorCtaOrd="([^"]+)"' IN lv_xml SUBMATCHES cv_rfc_ord.
  IF cv_rfc_ord IS INITIAL.
    FIND REGEX 'RfcEmisorCtaBen="([^"]+)"' IN lv_xml SUBMATCHES cv_rfc_ord.
  ENDIF.
  FIND REGEX 'NoCertificado="?([^"> ]+)"?' IN lv_xml SUBMATCHES cv_no_cert.
ENDFORM.

FORM f_parse_cfdi_fields_xstr
  USING    iv_xml_xstr    TYPE xstring
  CHANGING cv_uuid        TYPE string
           cv_fecha_timb  TYPE string
           cv_forma_pago  TYPE string
           cv_metodo_pago TYPE string
           cv_rfc_ord     TYPE string
           cv_no_cert     TYPE string.

  DATA lv_xml_str TYPE string.
  IF iv_xml_xstr IS INITIAL. RETURN. ENDIF.

  PERFORM zconv_xstr_to_str USING iv_xml_xstr CHANGING lv_xml_str.
  IF lv_xml_str IS INITIAL. RETURN. ENDIF.

  PERFORM f_parse_cfdi_fields_str
    USING lv_xml_str
    CHANGING cv_uuid cv_fecha_timb cv_forma_pago cv_metodo_pago cv_rfc_ord cv_no_cert.
ENDFORM.

FORM f_fill_from_xml_if_missing
  CHANGING cs_cp1 TYPE zta1_cp1.

  DATA: lv_uuid    TYPE string,
        lv_fec     TYPE string,
        lv_forma   TYPE string,
        lv_metodo  TYPE string,
        lv_rfc_ord TYPE string,
        lv_no_cert TYPE string.

  DATA: lo_desc TYPE REF TO cl_abap_typedescr.
  FIELD-SYMBOLS: <f_xml_any> TYPE any.

  ASSIGN COMPONENT 'XML_TIMB' OF STRUCTURE cs_cp1 TO <f_xml_any>.
  IF sy-subrc = 0 AND <f_xml_any> IS ASSIGNED AND <f_xml_any> IS NOT INITIAL.
    lo_desc = cl_abap_typedescr=>describe_by_data( <f_xml_any> ).
    CASE lo_desc->type_kind.
      WHEN cl_abap_typedescr=>typekind_string.
        PERFORM f_parse_cfdi_fields_str
          USING    <f_xml_any>
          CHANGING lv_uuid lv_fec lv_forma lv_metodo lv_rfc_ord lv_no_cert.
      WHEN cl_abap_typedescr=>typekind_xstring.
        PERFORM f_parse_cfdi_fields_xstr
          USING    <f_xml_any>
          CHANGING lv_uuid lv_fec lv_forma lv_metodo lv_rfc_ord lv_no_cert.
    ENDCASE.
  ENDIF.

  IF cs_cp1-uuid_comp IS INITIAL AND lv_uuid IS NOT INITIAL.
    cs_cp1-uuid_comp = lv_uuid.
  ENDIF.

  FIELD-SYMBOLS <f_fet> TYPE any.
  ASSIGN COMPONENT 'ZTFETIMB' OF STRUCTURE cs_cp1 TO <f_fet>.
  IF <f_fet> IS ASSIGNED AND <f_fet> IS INITIAL AND lv_fec IS NOT INITIAL.
    <f_fet> = lv_fec.
  ENDIF.

  PERFORM f_apply_forma_pago USING lv_forma lv_metodo CHANGING cs_cp1.

  FIELD-SYMBOLS <f_rfc_ord> TYPE any.
  ASSIGN COMPONENT 'RFC_ORD' OF STRUCTURE cs_cp1 TO <f_rfc_ord>.
  IF <f_rfc_ord> IS ASSIGNED AND <f_rfc_ord> IS INITIAL AND lv_rfc_ord IS NOT INITIAL.
    <f_rfc_ord> = lv_rfc_ord.
  ENDIF.

  FIELD-SYMBOLS: <f_no_cert>   TYPE any,
                 <f_cert_pago> TYPE any.
  IF lv_no_cert IS NOT INITIAL.
    ASSIGN COMPONENT 'NO_CERT'   OF STRUCTURE cs_cp1 TO <f_no_cert>.
    IF <f_no_cert> IS ASSIGNED AND <f_no_cert> IS INITIAL.
      <f_no_cert> = lv_no_cert.
    ENDIF.
    ASSIGN COMPONENT 'CERT_PAGO' OF STRUCTURE cs_cp1 TO <f_cert_pago>.
    IF <f_cert_pago> IS ASSIGNED AND <f_cert_pago> IS INITIAL.
      <f_cert_pago> = lv_no_cert.
    ENDIF.
  ENDIF.
ENDFORM.

FORM f_replicar_alv_en_cp1
  USING    is_src            TYPE zsa1_cp1
           iv_stat           TYPE zta1_cp1-estatus
           iv_uuid           TYPE zta1_cp1-uuid_comp
           iv_fechatimbrado  TYPE clike
           iv_certificado    TYPE clike
           iv_ruta_pdf       TYPE clike
           iv_ruta_xml       TYPE clike
           iv_xml_timbrado   TYPE xstring
           iv_text_stat      TYPE clike.

  DATA: lwa_cp1       TYPE zta1_cp1,
        lv_kunnr_norm TYPE kunnr,
        lv_text_calc  TYPE zta1_cp1-text_est,
        lv_text_in    TYPE string,
        lv_xml_str    TYPE string,
        lv_pdf_tmp    TYPE string,
        lv_xml_tmp    TYPE string.

  MOVE-CORRESPONDING is_src TO lwa_cp1.

  IF is_src-kunnr IS NOT INITIAL.
    lv_kunnr_norm = is_src-kunnr.
    CALL FUNCTION 'CONVERSION_EXIT_ALPHA_INPUT'
      EXPORTING
        input  = lv_kunnr_norm
      IMPORTING
        output = lv_kunnr_norm.
    lwa_cp1-kunnr = lv_kunnr_norm.
  ENDIF.

  lwa_cp1-numcomp = is_src-numcomp.
  lwa_cp1-estatus = iv_stat.

  lv_text_in = iv_text_stat.
  PERFORM f_fill_text_est USING    lwa_cp1-estatus
                                   lv_text_in
                          CHANGING lv_text_calc.
  IF lv_text_calc IS NOT INITIAL.
    lwa_cp1-text_est = lv_text_calc.
  ENDIF.

  IF iv_stat = gc_icon_green AND ( lwa_cp1-text_est IS INITIAL OR lwa_cp1-text_est CS 'Cancel' ).
    lwa_cp1-text_est = 'Timbrado'.
  ENDIF.

  lwa_cp1-uuid_comp = iv_uuid.

  FIELD-SYMBOLS <f_fet> TYPE any.
  ASSIGN COMPONENT 'ZTFETIMB' OF STRUCTURE lwa_cp1 TO <f_fet>.
  IF <f_fet> IS ASSIGNED AND iv_fechatimbrado IS NOT INITIAL.
    <f_fet> = iv_fechatimbrado.

  USING    iv_belnr     TYPE zfid_noper
           iv_gjahr     TYPE gjahr

FORM f_ensure_extension
  USING    iv_ext  TYPE clike
  CHANGING cv_path TYPE string.

  DATA: lv_path TYPE string,
        lv_ext  TYPE string.

  lv_path = cv_path.
  CONDENSE lv_path NO-GAPS.
  IF lv_path IS INITIAL.
    RETURN.
  ENDIF.

  lv_ext = iv_ext.
  TRANSLATE lv_ext TO LOWER CASE.
  IF lv_ext IS INITIAL.
    RETURN.
  ENDIF.
  IF lv_ext(1) <> '.'.
    CONCATENATE '.' lv_ext INTO lv_ext.
  ENDIF.

  FIND REGEX '\.[^\\/]+$' IN lv_path.
  IF sy-subrc <> 0.
    CONCATENATE lv_path lv_ext INTO lv_path.
  ENDIF.

  cv_path = lv_path.
ENDFORM.

FORM f_xstr_save_bin
  USING    iv_xstr      TYPE xstring
           iv_fullpath  TYPE string
  CHANGING ev_ok        TYPE abap_bool
           ev_message   TYPE string.

  DATA: lt_bin  TYPE solix_tab,
        lv_len  TYPE i,
        lv_path TYPE string.

  CLEAR: ev_ok, ev_message.
  lv_path = iv_fullpath.
  CONDENSE lv_path NO-GAPS.

  IF iv_xstr IS INITIAL.
    ev_message = 'Contenido vacío (XSTRING).'.
    RETURN.
  ENDIF.
  IF lv_path IS INITIAL.
    ev_message = 'Ruta de archivo vacía (FILENAME).'.
    RETURN.
  ENDIF.
  IF lv_path CP '*\' OR lv_path CP '*/'.
    ev_message = 'Ruta inválida: falta nombre de archivo.'.
    RETURN.
  ENDIF.

  CALL FUNCTION 'SCMS_XSTRING_TO_BINARY'
    EXPORTING
      buffer        = iv_xstr
    IMPORTING
      output_length = lv_len
    TABLES
      binary_tab    = lt_bin
    EXCEPTIONS
      failed        = 1
      OTHERS        = 2.
  IF sy-subrc <> 0.
    ev_message = 'Error al convertir XSTRING a BINARIO.'.
    RETURN.
  ENDIF.

  TRY.
      cl_gui_frontend_services=>gui_download(
        EXPORTING
          filename     = lv_path
          filetype     = 'BIN'
          bin_filesize = lv_len
        CHANGING
          data_tab     = lt_bin
      ).
      ev_ok = abap_true.
    CATCH cx_root INTO DATA(lx).
      ev_message = lx->get_text( ).
  ENDTRY.
ENDFORM.

      " Opcional: guarda request/response del PAC
      CLEAR lv_request.
      CALL FUNCTION 'SCMS_STRING_TO_XSTRING'
        EXPORTING
          text   = vxml
        IMPORTING
          buffer = lv_request
        EXCEPTIONS
          failed = 1
          OTHERS = 2.
      PERFORM update_req_response_pac_logica
              USING
                lwa_clientes
                lv_request       " request
                lv_response.     " response
    ENDIF.

    v_log = c_x.

    "---- Error: mostrar detalle inmediato
    IF lv_stat IS INITIAL OR lv_stat = gc_icon_red.

      DATA(lv_msgid_now) = sy-msgid.
      DATA(lv_msgno_now) = sy-msgno.
      DATA(lv_msgv1_now) = sy-msgv1.
      DATA(lv_msgv2_now) = sy-msgv2.
      DATA(lv_msgv3_now) = sy-msgv3.
      DATA(lv_msgv4_now) = sy-msgv4.

      IF lv_resstr IS NOT INITIAL.
        PERFORM f_show_error_resstr USING lwa_clientes lv_resstr 'Error de timbrado'.
      ENDIF.

      lv_any_error = abap_true.
      CONTINUE.
    ENDIF.

    "---- Éxito: reflejar en buffers / ALV y actualizar textos
    IF lv_stat = gc_icon_green.

      IF lv_pathpdf IS NOT INITIAL.
        PERFORM f_ensure_extension USING 'pdf' CHANGING lv_pathpdf.
      ENDIF.

      PERFORM f_replicar_alv_en_cp1
        USING lwa_clientes
              lv_stat
              lv_uuid
              lv_fechatimbrado
              lv_certificado
              lv_pathpdf
              space
              lv_request

        USING lwa_clientes
              lv_stat
              lv_uuid
              lv_fechatimbrado
              lv_text_stat.

      lv_any_success = abap_true.

      IF lv_name2 IS NOT INITIAL.
        PERFORM f_cp_pol_set USING lwa_clientes-numcomp lv_name2.
      ENDIF.

      " Actualiza texto del documento con el UUID (FI)
      DATA: lv_ok        TYPE flag,
            lv_msg       TYPE string,
            lv_belnr_upd TYPE zfid_noper.

      IF lwa_clientes-ztpago IS NOT INITIAL.
        lv_belnr_upd = lwa_clientes-ztpago.
      ELSEIF lwa_clientes-ztfactrel IS NOT INITIAL.
        lv_belnr_upd = lwa_clientes-ztfactrel.
      ENDIF.

      IF lv_belnr_upd IS NOT INITIAL AND lv_uuid IS NOT INITIAL.
        CALL FUNCTION 'CONVERSION_EXIT_ALPHA_INPUT'
          EXPORTING
            input  = lv_belnr_upd
          IMPORTING
            output = lv_belnr_upd.

        PERFORM f_update_text USING    lv_belnr_upd
                                      lwa_clientes-anio_pago
                                      lwa_clientes-bukrs
                                      lv_uuid
                                      space
                             CHANGING lv_ok
                                      lv_msg.
      ENDIF.

      " >>> Generar y UPSERT de CODE_QR en ZCP_SOVOS (CFDI 4.0)
      CLEAR: lv_qr_ok, lv_qr_msg.
      PERFORM f_upsert_zcp_sovos_qr
        USING    lwa_clientes
                 lv_uuid
                 lv_xml_timbrado
        CHANGING lv_qr_ok
                 lv_qr_msg.
      IF lv_qr_ok IS INITIAL AND lv_qr_msg IS NOT INITIAL.
        MESSAGE w398(00) WITH lv_qr_msg.
      ENDIF.

    ENDIF.

    REFRESH t_datos_mail[].
    REFRESH t_xml[].

  ENDLOOP.

  " Guardar log al final
  IF lo_logger IS BOUND.
    lo_logger->save( ).  " BAL_DB_SAVE + refresh
  ENDIF.

  "======== UPSERT y, si hubo éxito, avisar y refrescar ALV
  IF lv_any_success = abap_true AND t_zta1_cp1_2 IS NOT INITIAL.
    DATA lv_default_estatus TYPE zta1_cp1-estatus.
    lv_default_estatus = gc_icon_green.

    " Inserta/actualiza en ZTA1_CP1 usando el buffer temporal
    PERFORM f_upsert_zta1_cp1_batch USING 'T_ZTA1_CP1_2' lv_default_estatus.

    " Flujo simple de éxito: mensaje + abrir TEMP + ALV
    PERFORM f_exito_cp_timbre_simple
      USING 'T_ZTA1_CP1_2'
            'Complemento de pago timbrado correctamente.'
            lc_gui_log.

    " Limpia el buffer
    REFRESH t_zta1_cp1_2.
  ENDIF.

  " Si por alguna razón llegamos aquí con último doc en error (respaldo)
  IF lv_stat IS INITIAL OR lv_stat = gc_icon_red.
    RETURN.
  ENDIF.

  PERFORM f_alv_el USING lc_gui_log.

FORM f_upsert_zta1_cp1_batch
  USING
    iv_tabname         TYPE string
    iv_default_estatus TYPE zta1_cp1-estatus.

  DATA: lv_inserted   TYPE i VALUE 0,
        lv_updated    TYPE i VALUE 0,
        lv_skipped    TYPE i VALUE 0,
        lv_failed     TYPE i VALUE 0,
        lv_viewed     TYPE i VALUE 0,
        lv_kunnr_norm TYPE kunnr.

  DATA: lwa_new TYPE zta1_cp1,
        lwa_db  TYPE zta1_cp1.

  DATA: lt_success TYPE STANDARD TABLE OF ty_cp1_success,
        ls_success TYPE ty_cp1_success.

  FIELD-SYMBOLS: <lt_any>   TYPE STANDARD TABLE,
                 <ls_any>   TYPE any,
                 <fs_kunnr> TYPE any.

  DATA: lv_doc   TYPE string,
        lv_s_pdf TYPE string,
        lv_s_xml TYPE string.

  ASSIGN (iv_tabname) TO <lt_any>.
  IF <lt_any> IS NOT ASSIGNED.
    MESSAGE e004(00) WITH |Tabla interna { iv_tabname } no asignada|.
    RETURN.
  ENDIF.

  LOOP AT <lt_any> ASSIGNING <ls_any>.
    CLEAR: lwa_new, lwa_db.
    MOVE-CORRESPONDING <ls_any> TO lwa_new.

    ASSIGN COMPONENT 'KUNNR' OF STRUCTURE <ls_any> TO <fs_kunnr>.
    IF <fs_kunnr> IS ASSIGNED AND <fs_kunnr> IS NOT INITIAL.
      lv_kunnr_norm = <fs_kunnr>.
      CALL FUNCTION 'CONVERSION_EXIT_ALPHA_INPUT'
        EXPORTING
          input  = lv_kunnr_norm
        IMPORTING
          output = lv_kunnr_norm.
      lwa_new-kunnr = lv_kunnr_norm.
    ENDIF.

    IF lwa_new-estatus IS INITIAL AND iv_default_estatus IS NOT INITIAL.
      lwa_new-estatus = iv_default_estatus.
    ENDIF.

    PERFORM f_set_from_sources USING 'UUID_COMP'     'UUID_COMP|UUID|UUID_CFDI'      <ls_any> CHANGING lwa_new.
    PERFORM f_set_from_sources USING 'ZTFETIMB'      'ZTFETIMB|FECHA_TIMBRADO|F_TIMBRADO|ZTFETIMBRADO' <ls_any> CHANGING lwa_new.
    PERFORM f_set_from_sources USING 'ZTFORMAPAGO'   'ZTFORMAPAGO|MET_PAGODR|FORMA_PAGO|FORMADEPAGO'    <ls_any> CHANGING lwa_new.
    PERFORM f_set_from_sources USING 'MET_PAGODR'    'ZTFORMAPAGO|MET_PAGODR|FORMA_PAGO|FORMADEPAGO'    <ls_any> CHANGING lwa_new.
    PERFORM f_set_from_sources USING 'RFC_ORD'       'RFC_ORD|RFC_ORDENANTE|RFCORD' <ls_any> CHANGING lwa_new.
    PERFORM f_set_from_sources USING 'NO_CERT'       'NO_CERT|CERT_PAGO|NUM_CERT|NOCERT' <ls_any> CHANGING lwa_new.
    PERFORM f_set_from_sources USING 'CERT_PAGO'     'NO_CERT|CERT_PAGO|NUM_CERT|NOCERT' <ls_any> CHANGING lwa_new.
    PERFORM f_set_from_sources USING 'RUTA_XML'      'RUTA_XML|RUTAXML|PATH_XML|XML_PATH|FILE_XML' <ls_any> CHANGING lwa_new.
    PERFORM f_set_from_sources USING 'RUTA_PDF'      'RUTA_PDF|RUTAPDF|PATH_PDF|PDF_PATH|FILE_PDF' <ls_any> CHANGING lwa_new.

    PERFORM f_fill_from_xml_if_missing CHANGING lwa_new.

    IF lwa_new-bukrs     IS INITIAL OR
       lwa_new-kunnr     IS INITIAL OR
       lwa_new-ztpago    IS INITIAL OR
       lwa_new-anio_pago IS INITIAL OR
       lwa_new-ztfactrel IS INITIAL OR
       lwa_new-anio_fac  IS INITIAL.
      ADD 1 TO lv_skipped.
      CONTINUE.
    ENDIF.

    SELECT SINGLE * INTO lwa_db FROM zta1_cp1
      WHERE bukrs     = lwa_new-bukrs
        AND kunnr     = lwa_new-kunnr
        AND ztpago    = lwa_new-ztpago
        AND anio_pago = lwa_new-anio_pago
        AND ztfactrel = lwa_new-ztfactrel
        AND anio_fac  = lwa_new-anio_fac.

    IF sy-subrc <> 0.
      INSERT zta1_cp1 FROM lwa_new.
      IF sy-subrc = 0.
        ADD 1 TO lv_inserted.
        CLEAR ls_success.
        ls_success-bukrs     = lwa_new-bukrs.
        ls_success-kunnr     = lwa_new-kunnr.
        ls_success-ztpago    = lwa_new-ztpago.
        ls_success-anio_pago = lwa_new-anio_pago.
        ls_success-ztfactrel = lwa_new-ztfactrel.
        ls_success-anio_fac  = lwa_new-anio_fac.
        ls_success-numcomp   = lwa_new-numcomp.
        ls_success-estatus   = lwa_new-estatus.
        ls_success-uuid_comp = lwa_new-uuid_comp.
        ls_success-ztfetimb  = lwa_new-ztfetimb.

        CLEAR: lv_s_pdf, lv_s_xml.
        PERFORM f_get_field_dyn USING lwa_new 'RUTA_PDF|RUTAPDF|PATH_PDF|PDF_PATH|FILE_PDF'
                          CHANGING lv_s_pdf.
        PERFORM f_get_field_dyn USING lwa_new 'RUTA_XML|RUTAXML|PATH_XML|XML_PATH|FILE_XML'
                          CHANGING lv_s_xml.
        ls_success-ruta_pdf = lv_s_pdf.
        ls_success-ruta_xml = lv_s_xml.

        IF lwa_new-xml_timb IS NOT INITIAL OR ls_success-ruta_xml IS NOT INITIAL.
          ls_success-xml_dispon = abap_true.
        ELSE.
          ls_success-xml_dispon = abap_false.
        ENDIF.
        APPEND ls_success TO lt_success.
      ELSE.
        ADD 1 TO lv_failed.
      ENDIF.
    ELSE.
      PERFORM f_merge_cp1 USING lwa_new CHANGING lwa_db.
      UPDATE zta1_cp1 FROM lwa_db.
      IF sy-subrc = 0.
        ADD 1 TO lv_updated.
        CLEAR ls_success.
        ls_success-bukrs     = lwa_db-bukrs.
        ls_success-kunnr     = lwa_db-kunnr.
        ls_success-ztpago    = lwa_db-ztpago.
        ls_success-anio_pago = lwa_db-anio_pago.
        ls_success-ztfactrel = lwa_db-ztfactrel.
        ls_success-anio_fac  = lwa_db-anio_fac.
        ls_success-numcomp   = lwa_db-numcomp.
        ls_success-estatus   = lwa_db-estatus.
        ls_success-uuid_comp = lwa_db-uuid_comp.
        ls_success-ztfetimb  = lwa_db-ztfetimb.

        CLEAR: lv_s_pdf, lv_s_xml.
        PERFORM f_get_field_dyn USING lwa_db 'RUTA_PDF|RUTAPDF|PATH_PDF|PDF_PATH|FILE_PDF'
                          CHANGING lv_s_pdf.
        PERFORM f_get_field_dyn USING lwa_db 'RUTA_XML|RUTAXML|PATH_XML|XML_PATH|FILE_XML'
                          CHANGING lv_s_xml.
        ls_success-ruta_pdf = lv_s_pdf.
        ls_success-ruta_xml = lv_s_xml.

        IF lwa_db-xml_timb IS NOT INITIAL OR ls_success-ruta_xml IS NOT INITIAL.
          ls_success-xml_dispon = abap_true.
        ELSE.
          ls_success-xml_dispon = abap_false.
        ENDIF.
        APPEND ls_success TO lt_success.
      ELSE.
        ADD 1 TO lv_failed.
      ENDIF.
    ENDIF.
  ENDLOOP.

  IF lv_inserted > 0 OR lv_updated > 0.
    COMMIT WORK AND WAIT.
  ENDIF.

  IF lt_success IS NOT INITIAL.
    DATA lt_text TYPE soli_tab.
    PERFORM f_build_success_template USING lt_success CHANGING lt_text.
    PERFORM f_show_success_template USING lt_text.
  ENDIF.

*  DATA lv_send_now TYPE abap_bool.
*  PERFORM f_prompt_send_decision CHANGING lv_send_now.
*
*  "=== CAMBIO: generar y mostrar PDF (SmartForm) en lugar de abrir ruta_pdf

*    lv_viewed = lines( lt_success ).  "conteo aproximado de PDFs mostrados
*  ENDIF.

  WRITE: / |ZTA1_CP1 -> Insertados: { lv_inserted }  Actualizados: { lv_updated }  PDF abiertos: { lv_viewed }  Saltados: { lv_skipped }  Fallidos: { lv_failed }|.

ENDFORM.

FORM f_get_next_numcomp
  USING    iv_object   TYPE c
           iv_interval TYPE c
  CHANGING cv_number   TYPE char35.

  DATA: lv_num1 TYPE n LENGTH 10,
        lv_num2 TYPE n LENGTH 10.

  TRY.

        EXPORTING
          nr_range_nr             = iv_interval
          object                  = iv_object
        IMPORTING
          number                  = lv_num1
        EXCEPTIONS
          interval_not_found      = 1
          number_range_not_intern = 2
          object_not_found        = 3
          quantity_is_0           = 4

          interval_overflow       = 6
          buffer_overflow         = 7

      IF sy-subrc = 0.
        cv_number = lv_num1.

      ENDIF.
    CATCH cx_sy_dyn_call_param_not_found.

      nr_range_nr             = iv_interval

    IMPORTING
      tonumber                = lv_num2
    EXCEPTIONS
      interval_not_found      = 1

      quantity_is_0           = 4
      quantity_is_not_1       = 5
      interval_overflow       = 6
      buffer_overflow         = 7
      OTHERS                  = 8.

  IF sy-subrc = 0.
    cv_number = lv_num2.
  ELSE.
    CLEAR cv_number.
  ENDIF.
ENDFORM.

* f_genera_xml_cpago: usa zcl_cpago_xml_builder_cp20 y regresa el XML *
*---------------------------------------------------------------------*
FORM f_genera_xml_cpago
  USING    is_cliente     TYPE zsa1_cp1
           it_pagos       TYPE ANY TABLE
           it_fac         TYPE ANY TABLE
           is_fac010      TYPE ty_factura_010
           is_fac012      TYPE ty_factura_012
           is_fac017      TYPE ty_factura_017
           is_vbrk        TYPE ty_vbrk
           is_t001        TYPE ty_t001
           is_t001z       TYPE ty_t001z
           is_kna1        TYPE ty_kna1
           is_adrc        TYPE ty_adrc
  CHANGING cv_xml         TYPE string
           ct_xml         TYPE ANY TABLE
           ct_datos_mail  TYPE ANY TABLE.

  " Adaptadores de tablas genéricas
  TYPES: tt_zsa1_cp1 TYPE STANDARD TABLE OF zsa1_cp1 WITH EMPTY KEY.
  DATA: lt_pagos       TYPE tt_zsa1_cp1,
        lt_fac         TYPE tt_zsa1_cp1,
        lwa_datos_mail TYPE ty_datos_mail.

  FIELD-SYMBOLS: <ftab> TYPE ANY TABLE.

  ASSIGN it_pagos TO <ftab>.
  IF <ftab> IS ASSIGNED. lt_pagos[] = <ftab>[]. ENDIF.
  ASSIGN it_fac   TO <ftab>.
  IF <ftab> IS ASSIGNED. lt_fac[]   = <ftab>[]. ENDIF.

  " Instanciar la clase pedida
  DATA lo TYPE REF TO zcl_cpago_xml_builder_cp20.
  TRY.
      CREATE OBJECT lo.
    CATCH cx_root.
      CLEAR cv_xml.
      REFRESH ct_xml.
      REFRESH ct_datos_mail.
      RETURN.
  ENDTRY.

  " Limpiar buffers y llamar al generador
  CLEAR t_xml.   " (o REFRESH t_xml[] si lo prefieres)
  REFRESH t_datos_mail.

  TRY.
      CALL METHOD lo->genera_xml
        EXPORTING
          pwa_clientes    = is_cliente
          pt_pagos        = lt_pagos
          pt_fac          = lt_fac
          pwa_factura_010 = is_fac010
          pwa_factura_012 = is_fac012
          pwa_factura_017 = is_fac017
          pwa_vbrk        = is_vbrk
          pwa_t001        = is_t001
          pwa_t001z       = is_t001z
          pwa_kna1        = is_kna1
          pwa_adrc        = is_adrc
          pztmonp         = is_cliente-ztmonp
          ptcambio        = is_cliente-tcambio
        CHANGING
          pwa_datos_mail  = lwa_datos_mail
          ct_xml          = t_xml.
    CATCH cx_root.
      CLEAR cv_xml.
      REFRESH ct_xml.
      REFRESH ct_datos_mail.
      RETURN.
  ENDTRY.

  " Entregar buffers al llamador
  ct_xml        = t_xml.
  ct_datos_mail = t_datos_mail.

  " Construir el string XML desde la tabla t_xml (campo LINEA/LINE/TEXT/TDLINE)
  CLEAR cv_xml.
  FIELD-SYMBOLS: <row> TYPE any,
                 <f>   TYPE any.

  LOOP AT t_xml ASSIGNING <row>.
    ASSIGN COMPONENT 'LINEA'  OF STRUCTURE <row> TO <f>.
    IF sy-subrc <> 0 OR <f> IS INITIAL.
      ASSIGN COMPONENT 'LINE'   OF STRUCTURE <row> TO <f>.
    ENDIF.
    IF sy-subrc <> 0 OR <f> IS INITIAL.
      ASSIGN COMPONENT 'TEXT'   OF STRUCTURE <row> TO <f>.
    ENDIF.
    IF sy-subrc <> 0 OR <f> IS INITIAL.
      ASSIGN COMPONENT 'TDLINE' OF STRUCTURE <row> TO <f>.
    ENDIF.

    IF sy-subrc = 0 AND <f> IS ASSIGNED AND <f> IS NOT INITIAL.
      IF cv_xml IS INITIAL.
        cv_xml = <f>.
      ELSE.
        CONCATENATE cv_xml <f> INTO cv_xml
          SEPARATED BY cl_abap_char_utilities=>newline.
      ENDIF.
    ENDIF.
  ENDLOOP.

ENDFORM.

*---------------------------------------------------------------------*
* Toma el mejor texto de error disponible (PAC -> texto -> mensaje)   *
*---------------------------------------------------------------------*
*---------------------------------------------------------------------*
* Toma el mejor texto de error disponible (PAC -> texto -> mensaje)   *
*---------------------------------------------------------------------*
FORM f_pick_error_text
  USING    iv_resp_xstr  TYPE xstring   " Respuesta PAC (xstring)
           iv_text_stat  TYPE clike     " Texto interno/estatus
  CHANGING cv_strmsg     TYPE string.   " Mensaje elegido (string)

  DATA: lv_full  TYPE string,
        lv_extr  TYPE string,
        lv_dummy TYPE string.

  CLEAR cv_strmsg.

  " 1) PAC -> xstring a string
  IF iv_resp_xstr IS NOT INITIAL.
    PERFORM zconv_xstr_to_str USING iv_resp_xstr CHANGING lv_full.

    " 1a) Atributo tipo  key="valor"
    "     p.ej. Descripcion="CFDI inválido" o faultstring="Detail ..."
    FIND REGEX '(Descripcion|Descripción|Mensaje|message|faultstring)\s*=\s*"([^"]+)"'
      IN lv_full IGNORING CASE
      SUBMATCHES lv_dummy lv_extr.
    IF sy-subrc = 0 AND lv_extr IS NOT INITIAL.
      cv_strmsg = lv_extr.
    ENDIF.

    " 1b) Etiqueta XML <key>valor</key>
    IF cv_strmsg IS INITIAL.
      FIND REGEX '<(Descripcion|Descripción|Mensaje|message|faultstring)[^>]*>([^<]+)<'
        IN lv_full IGNORING CASE
        SUBMATCHES lv_dummy lv_extr.
      IF sy-subrc = 0 AND lv_extr IS NOT INITIAL.
        cv_strmsg = lv_extr.
      ENDIF.
    ENDIF.

    " 1c) JSON {"message":"..."} o {"descripcion":"..."}
    IF cv_strmsg IS INITIAL.
      FIND REGEX '"(message|descripcion|descripción)"\s*:\s*"([^"]+)"'
        IN lv_full IGNORING CASE
        SUBMATCHES lv_dummy lv_extr.
      IF sy-subrc = 0 AND lv_extr IS NOT INITIAL.
        cv_strmsg = lv_extr.
      ENDIF.
    ENDIF.

    " 1d) Genérico <error>...</error>
    IF cv_strmsg IS INITIAL.
      FIND REGEX '<error[^>]*>([^<]+)<'
        IN lv_full IGNORING CASE
        SUBMATCHES lv_extr.
      IF sy-subrc = 0 AND lv_extr IS NOT INITIAL.
        cv_strmsg = lv_extr.
      ENDIF.
    ENDIF.

    " 1e) Sin extracción “limpia”: corta el cuerpo
    IF cv_strmsg IS INITIAL AND lv_full IS NOT INITIAL.
      cv_strmsg = lv_full.
      IF strlen( cv_strmsg ) > 1000.
        cv_strmsg = cv_strmsg(1000) && ' ...'.
      ENDIF.
    ENDIF.
  ENDIF.

  " 2) Texto interno del proceso
  IF cv_strmsg IS INITIAL AND iv_text_stat IS NOT INITIAL.
    cv_strmsg = iv_text_stat.
  ENDIF.

  " 3) Último mensaje ABAP (si existe)
  IF cv_strmsg IS INITIAL.
    DATA lv_msgtxt TYPE string.
    CALL FUNCTION 'MESSAGE_TEXT_BUILD'
      EXPORTING
        msgid               = sy-msgid
        msgnr               = sy-msgno
        msgv1               = sy-msgv1
        msgv2               = sy-msgv2
        msgv3               = sy-msgv3
        msgv4               = sy-msgv4
      IMPORTING
        message_text_output = lv_msgtxt
      EXCEPTIONS
        OTHERS              = 1.
    IF sy-subrc = 0 AND lv_msgtxt IS NOT INITIAL.
      cv_strmsg = lv_msgtxt.
    ENDIF.
  ENDIF.

  IF cv_strmsg IS INITIAL.
    cv_strmsg = 'Error de timbrado sin detalle devuelto por el PAC.'.
  ENDIF.

  CONDENSE cv_strmsg.
ENDFORM.

FORM f_sync_alv_line
  USING    is_cli          TYPE zsa1_cp1
           iv_stat         TYPE zta1_cp1-estatus
           iv_uuid         TYPE zta1_cp1-uuid_comp
           iv_fechatimbr   TYPE clike

        lv_xml     TYPE string,
        lv_numcomp TYPE zta1_cp1-numcomp.

  READ TABLE t_zta1_2 ASSIGNING <fs>
       WITH KEY bukrs     = is_cli-bukrs
                kunnr     = is_cli-kunnr
                ztpago    = is_cli-ztpago
                anio_pago = is_cli-anio_pago
                ztfactrel = is_cli-ztfactrel
                anio_fac  = is_cli-anio_fac.

                                   iv_text_stat
                          CHANGING lv_txt.

    lv_txt = 'Timbrado'.

  <fs>-uuid_comp = iv_uuid.
  <fs>-ztfetimb  = iv_fechatimbr.

  ELSE.
    lv_numcomp = <fs>-numcomp.
  ENDIF.

  " Forzar rutas = NUMCOMP con extensión
  lv_pdf = lv_numcomp.
  lv_xml = lv_numcomp.
  PERFORM f_ensure_extension USING 'pdf' CHANGING lv_pdf.
  PERFORM f_ensure_extension USING 'xml' CHANGING lv_xml.

  " CONCATENATE 'C:\Facturas\' lv_pdf INTO lv_pdf.
  " CONCATENATE 'C:\Facturas\' lv_xml INTO lv_xml.

  <fs>-ruta_pdf = lv_pdf.
  <fs>-ruta_xml = lv_xml.

  " Guardar cambios en la tabla
  MODIFY t_zta1_2 FROM <fs> INDEX sy-tabix.

ENDFORM.

* Lee T001 portable (sin asumir STCD1). Devuelve RFC, BUTXT, LAND1.   *

FORM f_read_t001_portable
  USING    iv_bukrs      TYPE bukrs
  CHANGING cv_rfc        TYPE string
           cv_butxt      TYPE string
           cv_land1      TYPE land1.

  DATA: ls_t001 TYPE t001.
  FIELD-SYMBOLS <f_any> TYPE any.

  CLEAR: cv_rfc, cv_butxt, cv_land1.
  IF iv_bukrs IS INITIAL.
    RETURN.
  ENDIF.

  SELECT SINGLE *
    INTO @ls_t001
    FROM t001
    WHERE bukrs = @iv_bukrs.

  IF sy-subrc <> 0.
    RETURN.
  ENDIF.

  cv_butxt = ls_t001-butxt.
  cv_land1 = ls_t001-land1.

  " RFC: intenta STCD1, si no existe/está vacío intenta STCEG
  ASSIGN COMPONENT 'STCD1' OF STRUCTURE ls_t001 TO <f_any>.
  IF sy-subrc = 0 AND <f_any> IS ASSIGNED AND <f_any> IS NOT INITIAL.
    cv_rfc = <f_any>.
  ELSE.
    ASSIGN COMPONENT 'STCEG' OF STRUCTURE ls_t001 TO <f_any>.
    IF sy-subrc = 0 AND <f_any> IS ASSIGNED AND <f_any> IS NOT INITIAL.
      cv_rfc = <f_any>.
    ENDIF.
  ENDIF.
ENDFORM.

      IF sy-subrc = 0 AND lv_extr IS NOT INITIAL.
        cv_strmsg = lv_extr.
      ENDIF.
    ENDIF.

    " Genérico <error>...</error>
    IF cv_strmsg IS INITIAL.
      FIND REGEX '<error[^>]*>([^<]+)<'
        IN lv_full IGNORING CASE SUBMATCHES lv_extr.
      IF sy-subrc = 0 AND lv_extr IS NOT INITIAL.
        cv_strmsg = lv_extr.
      ENDIF.
    ENDIF.

    " Si no hubo extracción limpia, corta cuerpo para no saturar
    IF cv_strmsg IS INITIAL AND lv_full IS NOT INITIAL.

        msgnr               = iv_msgno

  IF cv_strmsg IS INITIAL.
    cv_strmsg = 'Error de timbrado sin detalle devuelto por el PAC.'.
  ENDIF.

*----------------------------------------------------------------------
* Recupera Emisor/Receptor de forma robusta (sin listas de campos)
*----------------------------------------------------------------------
FORM f_resolver_emisor_receptor
  USING    i_cliente       TYPE zsa1_cp1
  CHANGING cv_rfc_emisor   TYPE paval
           cv_nom_emisor   TYPE string
           cv_rfc_receptor TYPE string
           cv_nom_receptor TYPE string.

  FIELD-SYMBOLS: <fs_any> TYPE any, <fs_cmp> TYPE any, <fv> TYPE any.
  ASSIGN i_cliente TO <fs_any>.

  "---------------- Receptor desde la estructura ----------------
  IF cv_rfc_receptor IS INITIAL.
    ASSIGN COMPONENT 'RFC_CL'        OF STRUCTURE <fs_any> TO <fs_cmp>.
    IF sy-subrc = 0 AND <fs_cmp> IS ASSIGNED AND <fs_cmp> IS NOT INITIAL.
      cv_rfc_receptor = <fs_cmp>.
    ENDIF.
  ENDIF.
  IF cv_rfc_receptor IS INITIAL.
    ASSIGN COMPONENT 'RFC_RECEPTOR'  OF STRUCTURE <fs_any> TO <fs_cmp>.
    IF sy-subrc = 0 AND <fs_cmp> IS ASSIGNED AND <fs_cmp> IS NOT INITIAL.
      cv_rfc_receptor = <fs_cmp>.
    ENDIF.
  ENDIF.

  IF cv_nom_receptor IS INITIAL.
    ASSIGN COMPONENT 'NOMBRE'        OF STRUCTURE <fs_any> TO <fs_cmp>.
    IF sy-subrc = 0 AND <fs_cmp> IS ASSIGNED AND <fs_cmp> IS NOT INITIAL.
      cv_nom_receptor = <fs_cmp>.
    ENDIF.
  ENDIF.
  IF cv_nom_receptor IS INITIAL.
    ASSIGN COMPONENT 'NOMRECEPTOR'   OF STRUCTURE <fs_any> TO <fs_cmp>.
    IF sy-subrc = 0 AND <fs_cmp> IS ASSIGNED AND <fs_cmp> IS NOT INITIAL.
      cv_nom_receptor = <fs_cmp>.
    ENDIF.
  ENDIF.

  "---------------- Receptor desde KNA1 (SELECT * seguro) ----------------
  IF cv_rfc_receptor IS INITIAL OR cv_nom_receptor IS INITIAL.
    DATA lv_kunnr TYPE kunnr.
    ASSIGN COMPONENT 'KUNNR' OF STRUCTURE <fs_any> TO <fs_cmp>.
    IF sy-subrc = 0 AND <fs_cmp> IS ASSIGNED. lv_kunnr = <fs_cmp>. ENDIF.

    IF lv_kunnr IS NOT INITIAL.
      DATA ls_kna1 TYPE kna1.
      SELECT SINGLE *
        FROM kna1
        INTO @ls_kna1
       WHERE kunnr = @lv_kunnr.

      IF sy-subrc = 0.
        " Nombre (estático)
        IF cv_nom_receptor IS INITIAL AND ls_kna1-name1 IS NOT INITIAL.
          cv_nom_receptor = ls_kna1-name1.
        ENDIF.

        " RFC receptor: intentar STCD1 de forma DINÁMICA (no falla si no existe)
        IF cv_rfc_receptor IS INITIAL.
          ASSIGN COMPONENT 'STCD1' OF STRUCTURE ls_kna1 TO <fv>.
          IF sy-subrc = 0 AND <fv> IS ASSIGNED AND <fv> IS NOT INITIAL.
            cv_rfc_receptor = <fv>.
          ENDIF.
        ENDIF.
      ENDIF.
    ENDIF.
  ENDIF.

  "---------------- Emisor desde T001/T001Z (SELECT * seguro) --------------
  DATA lv_bukrs TYPE bukrs.
  ASSIGN COMPONENT 'BUKRS' OF STRUCTURE <fs_any> TO <fs_cmp>.
  IF sy-subrc = 0 AND <fs_cmp> IS ASSIGNED. lv_bukrs = <fs_cmp>. ENDIF.

  IF lv_bukrs IS NOT INITIAL.
    DATA ls_t001 TYPE t001.
    SELECT SINGLE *
      FROM t001
      INTO @ls_t001
     WHERE bukrs = @lv_bukrs.

    IF sy-subrc = 0.
      " Nombre emisor
      IF cv_nom_emisor IS INITIAL AND ls_t001-butxt IS NOT INITIAL.
        cv_nom_emisor = ls_t001-butxt.
      ENDIF.

      " RFC emisor preferente: T001Z-PAVAL
      IF cv_rfc_emisor IS INITIAL.
        SELECT SINGLE paval
          FROM t001z
          INTO @cv_rfc_emisor
         WHERE bukrs = @ls_t001-bukrs.
      ENDIF.

      " Respaldo: intenta STCEG en T001 (si existe) de forma DINÁMICA
      IF cv_rfc_emisor IS INITIAL.
        ASSIGN COMPONENT 'STCEG' OF STRUCTURE ls_t001 TO <fv>.
        IF sy-subrc = 0 AND <fv> IS ASSIGNED AND <fv> IS NOT INITIAL.
          cv_rfc_emisor = <fv>.
        ENDIF.
      ENDIF.
    ENDIF.
  ENDIF.

  "---------------- Defaults para no dejar vacíos ----------------
  IF cv_nom_receptor IS INITIAL. cv_nom_receptor = 'N/A'.           ENDIF.
  IF cv_rfc_receptor IS INITIAL. cv_rfc_receptor = 'XAXX010101000'. ENDIF.
  IF cv_nom_emisor   IS INITIAL. cv_nom_emisor   = 'N/A'.           ENDIF.
  IF cv_rfc_emisor   IS INITIAL. cv_rfc_emisor   = 'XAXX010101000'. ENDIF.

ENDFORM.

*----------------------------------------------------------------------
*  Helper: cargar tablas del SmartForm (sin conflictos de tipos)
*----------------------------------------------------------------------
FORM f_cargar_tablas_sf
  USING    i_cliente TYPE zsa1_cp1
  CHANGING ct_pagos  TYPE ztta1_cp1
           ct_fac    TYPE ztta1_cp1
           ct_fac2   TYPE ztta1_cp1.

  FIELD-SYMBOLS: <fs_any> TYPE any, <fs_cmp> TYPE any.
  ASSIGN i_cliente TO <fs_any>.

  " Variables del MISMO tipo de línea que cada tabla
  DATA: ls_pagos_line LIKE LINE OF ct_pagos,
        ls_fac_line   LIKE LINE OF ct_fac,
        ls_fac2_line  LIKE LINE OF ct_fac2.

  " 1) Asegurar FAC2 con el registro actual, pero convirtiendo el tipo
  CLEAR ls_fac2_line.
  MOVE-CORRESPONDING i_cliente TO ls_fac2_line.
  IF ls_fac2_line IS NOT INITIAL.
    APPEND ls_fac2_line TO ct_fac2.
  ENDIF.

  " 2) Cargar más líneas desde ZTA1_CP1 con el mismo NUMCOMP
  DATA lv_numcomp TYPE string.
  ASSIGN COMPONENT 'NUMCOMP' OF STRUCTURE <fs_any> TO <fs_cmp>.
  IF sy-subrc = 0 AND <fs_cmp> IS ASSIGNED AND <fs_cmp> IS NOT INITIAL.
    lv_numcomp = <fs_cmp>.
  ENDIF.
  IF lv_numcomp IS INITIAL.
    RETURN.
  ENDIF.

  " New Open SQL: host vars con @ y comas
  SELECT *
    FROM zta1_cp1
    INTO TABLE @DATA(lt_zta1)
   WHERE numcomp = @lv_numcomp.

  LOOP AT lt_zta1 ASSIGNING FIELD-SYMBOL(<ls_zta1>).
    FIELD-SYMBOLS <f_pago>  TYPE any.
    FIELD-SYMBOLS <f_total> TYPE any.

    ASSIGN COMPONENT 'IMP_PAGO' OF STRUCTURE <ls_zta1> TO <f_pago>.
    ASSIGN COMPONENT 'TOTAL'    OF STRUCTURE <ls_zta1> TO <f_total>.

    IF <f_pago> IS ASSIGNED AND <f_pago> IS NOT INITIAL.
      CLEAR ls_pagos_line.
      MOVE-CORRESPONDING <ls_zta1> TO ls_pagos_line.
      APPEND ls_pagos_line TO ct_pagos.
      CONTINUE.
    ENDIF.

    IF <f_total> IS ASSIGNED AND <f_total> IS NOT INITIAL.
      CLEAR ls_fac_line.
      MOVE-CORRESPONDING <ls_zta1> TO ls_fac_line.
      APPEND ls_fac_line TO ct_fac.
      CONTINUE.
    ENDIF.

    CLEAR ls_fac2_line.
    MOVE-CORRESPONDING <ls_zta1> TO ls_fac2_line.
    APPEND ls_fac2_line TO ct_fac2.
  ENDLOOP.

ENDFORM.


*----------------------------------------------------------------------
*  Helper: nombre/ruta segura en carpeta temporal del frontend
*----------------------------------------------------------------------
*----------------------------------------------------------------------
*  Helper: nombre/ruta segura en carpeta temporal del frontend
*----------------------------------------------------------------------
*---------------------------------------------------------------------*
*  Helper: nombre/ruta segura en carpeta temporal del frontend
*---------------------------------------------------------------------*
FORM f_build_pdf_filename
  USING    i_cliente TYPE zsa1_cp1
           i_fname   TYPE string
  CHANGING cv_full   TYPE string.

  DATA: lv_name TYPE string,
        lv_sep  TYPE c LENGTH 1,
        lv_len  TYPE i,
        lv_off  TYPE i,
        lv_last TYPE c LENGTH 1.

  " Nombre sugerido
  IF i_fname IS NOT INITIAL.
    lv_name = i_fname.
  ELSE.
    FIELD-SYMBOLS: <fs_any> TYPE any, <fs_cmp> TYPE any.
    ASSIGN i_cliente TO <fs_any>.
    ASSIGN COMPONENT 'NUMCOMP' OF STRUCTURE <fs_any> TO <fs_cmp>.
    IF sy-subrc = 0 AND <fs_cmp> IS ASSIGNED AND <fs_cmp> IS NOT INITIAL.
      lv_name = <fs_cmp>.
    ELSE.
      lv_name = 'smartform'.
    ENDIF.
  ENDIF.

  " Sanea y asegura .pdf
  REPLACE ALL OCCURRENCES OF REGEX '[\\/:*?"<>|]' IN lv_name WITH '_'.
  IF lv_name NP '*.pdf'.
    CONCATENATE lv_name '.pdf' INTO lv_name.
  ENDIF.

  " ¡IMPORTANTE!: usar variable GLOBAL para el CHANGING
  CLEAR gv_temp_dir.
  TRY.
      cl_gui_frontend_services=>get_temp_directory(
        CHANGING temp_dir = gv_temp_dir ).
      cl_gui_cfw=>flush( ).  " asegura el retorno antes de salir
    CATCH cx_root.
      CLEAR gv_temp_dir.
  ENDTRY.

  " Si no hay GUI o falló, usa solo el nombre
  IF gv_temp_dir IS INITIAL.
    cv_full = lv_name.
    RETURN.
  ENDIF.

  " Separador según la ruta devuelta
  lv_sep = '/'.
  IF gv_temp_dir CS '\'.
    lv_sep = '\'.
  ENDIF.

  lv_len = strlen( gv_temp_dir ).
  IF lv_len > 0.
    lv_off  = lv_len - 1.
    lv_last = gv_temp_dir+lv_off(1).
  ENDIF.

  IF lv_last = lv_sep.
    CONCATENATE gv_temp_dir lv_name INTO cv_full.
  ELSE.
    CONCATENATE gv_temp_dir lv_sep lv_name INTO cv_full.
  ENDIF.
ENDFORM.

           iv_numcomp  TYPE znumcomp
           iv_title    TYPE string.

  DATA: lv_in_str     TYPE string,
        lv_xml_err    TYPE string,
        lv_xml_pretty TYPE string,
        lv_nl         TYPE string VALUE cl_abap_char_utilities=>newline,
        lv_end_tag    TYPE string VALUE '</InnerErrors>',
        lv_off_start  TYPE i,
        lv_off_end    TYPE i,
        lv_end_len    TYPE i,
        lv_len        TYPE i.

  DATA: lv_dir     TYPE string,
        lv_file    TYPE string,
        lv_base    TYPE string,
        lv_upper   TYPE string,
        lv_exec_rc TYPE i.

  " 0) Validar entrada
  lv_in_str = iv_resstr.
  IF lv_in_str IS INITIAL.
    MESSAGE w398(00) WITH 'Cadena de respuesta vacía; no hay XML para mostrar.'.
    RETURN.
  ENDIF.

  " 1) Tomar específicamente <InnerErrors>... si existe
  CLEAR: lv_xml_err, lv_off_start, lv_off_end.
  lv_off_start = -1.
  FIND REGEX '<[Ii]nner[Ee]rrors' IN lv_in_str MATCH OFFSET lv_off_start.
  IF sy-subrc = 0.
    FIND REGEX '</[Ii]nner[Ee]rrors>' IN lv_in_str MATCH OFFSET lv_off_end.
    IF sy-subrc = 0 AND lv_off_end > lv_off_start.
      lv_end_len = strlen( lv_end_tag ).
      lv_len     = lv_off_end - lv_off_start + lv_end_len.
      lv_xml_err = lv_in_str+lv_off_start(lv_len).
    ELSE.
      lv_xml_err = lv_in_str+lv_off_start.
    ENDIF.
  ELSE.
    FIND FIRST OCCURRENCE OF '<' IN lv_in_str MATCH OFFSET lv_off_start.
    IF sy-subrc = 0.
      lv_xml_err = lv_in_str+lv_off_start.
    ELSE.
      lv_xml_err = lv_in_str.
    ENDIF.
  ENDIF.

  " 2) Pretty print + normalizar saltos
  PERFORM f_pretty_xml_from_text USING lv_xml_err CHANGING lv_xml_pretty.
  IF lv_xml_pretty IS INITIAL.
    lv_xml_pretty = lv_xml_err.
  ENDIF.
  REPLACE ALL OCCURRENCES OF cl_abap_char_utilities=>cr_lf IN lv_xml_pretty WITH lv_nl.

  " 3) Carpeta destino "como éxito" (se asegura existencia)
  PERFORM f_get_success_like_dir CHANGING lv_dir.
  IF lv_dir IS INITIAL.
    MESSAGE e398(00) WITH 'No se pudo determinar/crear la carpeta destino.'.
    RETURN.
  ENDIF.

  " 4) Base CP<NUMCOMP> sin duplicar 'CP'
  lv_base = iv_numcomp.
  lv_upper = lv_base. TRANSLATE lv_upper TO UPPER CASE.
  IF strlen( lv_upper ) < 2 OR lv_upper(2) <> 'CP'.
    CONCATENATE 'CP' lv_base INTO lv_base.
  ENDIF.
  CONCATENATE lv_dir '\' lv_base '_ERROR.xml' INTO lv_file.

  " 5) Convertir a UTF-8 (xstring) y descargar BIN (portable)
  DATA: lv_xdata TYPE xstring,
        lt_bin   TYPE solix_tab,
        lv_size  TYPE i.

  DATA(lo_conv) = cl_abap_conv_out_ce=>create( encoding = 'UTF-8' ).
  lo_conv->convert( EXPORTING data = lv_xml_pretty IMPORTING buffer = lv_xdata ).

  CALL FUNCTION 'SCMS_XSTRING_TO_BINARY'
    EXPORTING
      buffer        = lv_xdata
    IMPORTING
      output_length = lv_size
    TABLES
      binary_tab    = lt_bin
    EXCEPTIONS
      OTHERS        = 1.

  CALL METHOD cl_gui_frontend_services=>gui_download
    EXPORTING
      filename     = lv_file
      filetype     = 'BIN'
      bin_filesize = lv_size
    CHANGING
      data_tab     = lt_bin
    EXCEPTIONS
      OTHERS       = 1.

  IF sy-subrc <> 0.
    MESSAGE e398(00) WITH |No se pudo crear { lv_file } (SUBRC={ sy-subrc }).|.
    RETURN.
  ENDIF.

  CALL FUNCTION 'SAPGUI_PROGRESS_INDICATOR'
    EXPORTING
      text = CONV string( iv_title ).

  " 6) Abrir con RC obligatorio en tu release
  CALL METHOD cl_gui_frontend_services=>execute
    EXPORTING
      document  = lv_file
      operation = 'OPEN'
*    CHANGING
*     rc        = lv_exec_rc
    EXCEPTIONS
      OTHERS    = 1.

  IF lv_exec_rc IS NOT INITIAL.
    MESSAGE w398(00) WITH |Windows devolvió RC={ lv_exec_rc } al abrir { lv_file }.|.
  ENDIF.

ENDFORM.

*---------------------------------------------------------------------*
* FORM f_get_sapgui_tmp_dir  ->  devuelve el directorio tmp de SAP GUI
*---------------------------------------------------------------------*
FORM f_get_sapgui_tmp_dir
  CHANGING cv_dir TYPE string.

  CLEAR cv_dir.
  " 1) Carpeta de trabajo de SAP GUI (la del CP00000xxxx.pdf)
  CALL METHOD cl_gui_frontend_services=>get_sapgui_workdir
    CHANGING
      sapworkdir = cv_dir
    EXCEPTIONS
      OTHERS     = 1.

  " 2) Respaldo: temp del SO
  IF cv_dir IS INITIAL.
    CALL METHOD cl_gui_frontend_services=>get_temp_directory
      CHANGING
        temp_dir = cv_dir
      EXCEPTIONS
        OTHERS   = 1.
  ENDIF.

  " 3) Último fallback
  IF cv_dir IS INITIAL.
    cv_dir = 'C:\Users\Public\Documents'.
  ENDIF.
ENDFORM.



*---------------------------------------------------------------------*
* Carpeta "como éxito"
* - Prioriza: SAP GUI WORKDIR\tmp  >  SAP GUI WORKDIR  >  TEMP  >  SYSTEM
* - Crea la carpeta si no existe (DIRECTORY_CREATE con RC)
*---------------------------------------------------------------------*

*---------------------------------------------------------------------*
* Carpeta "como éxito"
* - Prioriza: SAP GUI WORKDIR\tmp  >  SAP GUI WORKDIR  >  TEMP  >  SYSTEM
* - Crea la carpeta si no existe (DIRECTORY_CREATE con RC)
*---------------------------------------------------------------------*
FORM f_get_success_like_dir CHANGING cv_dir TYPE string.

  DATA: lv_ok      TYPE abap_bool,
        lv_rc      TYPE i,
        lv_work    TYPE string,
        lv_tmpcand TYPE string.

  CLEAR: cv_dir, lv_work, lv_tmpcand.

  " 1) Workdir de SAP GUI
  CALL METHOD cl_gui_frontend_services=>get_sapgui_workdir
    CHANGING
      sapworkdir = lv_work
    EXCEPTIONS
      OTHERS     = 1.

  IF lv_work IS NOT INITIAL.
    " Preferir subcarpeta \tmp si existe
    CONCATENATE lv_work '\tmp' INTO lv_tmpcand.
    CALL METHOD cl_gui_frontend_services=>directory_exist
      EXPORTING
        directory = lv_tmpcand
      RECEIVING
        result    = lv_ok
      EXCEPTIONS
        OTHERS    = 1.
    IF lv_ok = abap_true.
      cv_dir = lv_tmpcand.
    ELSE.
      cv_dir = lv_work.
    ENDIF.
  ENDIF.

  " 2) Fallback: TEMP del usuario
  IF cv_dir IS INITIAL OR sy-subrc <> 0.
    CLEAR cv_dir.
    CALL METHOD cl_gui_frontend_services=>get_temp_directory
      CHANGING
        temp_dir = cv_dir
      EXCEPTIONS
        OTHERS   = 1.
  ENDIF.

  " 3) Fallback extra: carpeta del sistema
  IF cv_dir IS INITIAL OR sy-subrc <> 0.
    CLEAR cv_dir.
    CALL METHOD cl_gui_frontend_services=>get_system_directory
      CHANGING
        system_directory = cv_dir
      EXCEPTIONS
        OTHERS           = 1.
  ENDIF.

  " 4) Si no logramos ninguna ruta, salir
  IF cv_dir IS INITIAL.
    RETURN.
  ENDIF.

  " 5) Asegurar que exista (tu release exige RC en DIRECTORY_CREATE)
  CALL METHOD cl_gui_frontend_services=>directory_exist
    EXPORTING
      directory = cv_dir
    RECEIVING
      result    = lv_ok
    EXCEPTIONS
      OTHERS    = 1.

  IF lv_ok IS INITIAL.
    CALL METHOD cl_gui_frontend_services=>directory_create
      EXPORTING
        directory = cv_dir
      CHANGING
        rc        = lv_rc
      EXCEPTIONS
        OTHERS    = 1.
    IF sy-subrc <> 0 OR lv_rc <> 0.
      " Si no se pudo crear, devolver vacío para que el caller avise
      CLEAR cv_dir.
    ENDIF.
  ENDIF.

ENDFORM.

FORM f_upsert_zcp_sovos_qr
  USING    is_cli          TYPE zsa1_cp1
           iv_uuid         TYPE zfid_uuid_cfdi
           iv_xml_timbrado TYPE xstring
  CHANGING ev_ok           TYPE abap_bool
           ev_msg          TYPE string.

  DATA: ls_sovos     TYPE zcp_sovos,
        lv_found     TYPE abap_bool,
        lv_qr        TYPE zqr40,
        lv_total     TYPE char18 VALUE '0.00',
        lv_sello     TYPE string.

  " RFCs: tomar como string, normalizar a CHAR13
  DATA: lv_rfc_em_s      TYPE string,
        lv_rfc_re_s      TYPE string,
        lv_rfc_em_c13    TYPE char13,
        lv_rfc_re_c13    TYPE char13,
        lv_extranjero    TYPE char1 VALUE space.

  " UUID: adaptar a dominio ZUUID (32 o 36)
  DATA: lv_puuid         TYPE zuuid,
        lv_puuid_len     TYPE i,
        lv_uuid_c32      TYPE char32,
        lv_uuid_c36      TYPE char36.

  CLEAR: ev_ok, ev_msg, lv_puuid, lv_uuid_c32, lv_uuid_c36, lv_found.

  "--- RFC receptor desde cliente
  lv_rfc_re_s = is_cli-rfc_cl.

  "--- RFC emisor robusto (T001Z-PAVAL, fallback T001-STCEG)
  DATA: lv_butxt TYPE string,
        lv_land1 TYPE land1.
  PERFORM f_read_t001_portable
    USING    is_cli-bukrs
    CHANGING lv_rfc_em_s lv_butxt lv_land1.

  "--- Normalizar RFCs a CHAR13 (uppercase, sin espacios)
  lv_rfc_em_c13 = lv_rfc_em_s.
  TRANSLATE lv_rfc_em_c13 TO UPPER CASE.
  CONDENSE  lv_rfc_em_c13 NO-GAPS.
  IF strlen( lv_rfc_em_c13 ) > 13. lv_rfc_em_c13 = lv_rfc_em_c13(13). ENDIF.

  lv_rfc_re_c13 = lv_rfc_re_s.
  TRANSLATE lv_rfc_re_c13 TO UPPER CASE.
  CONDENSE  lv_rfc_re_c13 NO-GAPS.
  IF strlen( lv_rfc_re_c13 ) > 13. lv_rfc_re_c13 = lv_rfc_re_c13(13). ENDIF.

  IF lv_rfc_re_c13 = 'XEXX010101000'.
    lv_extranjero = 'X'.
  ENDIF.

  "--- SelloCFD del XML (o del registro existente)
  PERFORM f_extract_sello_from_xml_xstr USING iv_xml_timbrado CHANGING lv_sello.

  SELECT SINGLE * INTO @ls_sovos
    FROM zcp_sovos
    WHERE bukrs   = @is_cli-bukrs
      AND numcomp = @is_cli-numcomp.
  IF sy-subrc = 0.
    lv_found = abap_true.
    IF lv_sello IS INITIAL AND ls_sovos-sellocfd IS NOT INITIAL.
      lv_sello = ls_sovos-sellocfd.
    ENDIF.
  ENDIF.

  "=== UUID → ZUUID (32 o 36) ========================================
  DATA(lo_desc) = cl_abap_typedescr=>describe_by_data( iv_uuid ).
  IF lo_desc->type_kind = cl_abap_typedescr=>typekind_hex
  OR lo_desc->type_kind = cl_abap_typedescr=>typekind_xstring.
    CALL FUNCTION 'GUID_CONVERT'
      EXPORTING  guid_16 = iv_uuid
      IMPORTING  guid_32 = lv_uuid_c32
      EXCEPTIONS OTHERS  = 1.
    IF lv_uuid_c32 IS NOT INITIAL.
      lv_uuid_c36 = |{ lv_uuid_c32(8) }-{ lv_uuid_c32+8(4) }-{ lv_uuid_c32+12(4) }-{ lv_uuid_c32+16(4) }-{ lv_uuid_c32+20(12) }|.
    ENDIF.
  ELSE.
    DATA(lv_uuid_str) = |{ iv_uuid }|.
    TRANSLATE lv_uuid_str TO UPPER CASE.
    REPLACE ALL OCCURRENCES OF '-' IN lv_uuid_str WITH ''.
    CONDENSE lv_uuid_str NO-GAPS.
    IF strlen( lv_uuid_str ) = 32.
      lv_uuid_c32 = lv_uuid_str.
      lv_uuid_c36 = |{ lv_uuid_c32(8) }-{ lv_uuid_c32+8(4) }-{ lv_uuid_c32+12(4) }-{ lv_uuid_c32+16(4) }-{ lv_uuid_c32+20(12) }|.
    ELSEIF strlen( lv_uuid_str ) = 36.
      lv_uuid_c36 = lv_uuid_str.
      DATA(lv_c32_tmp) = lv_uuid_str.
      REPLACE ALL OCCURRENCES OF '-' IN lv_c32_tmp WITH ''.
      lv_uuid_c32 = lv_c32_tmp.
    ELSE.
      ev_msg = |UUID con longitud inesperada: "{ lv_uuid_str }"|.
    ENDIF.
  ENDIF.

  IF lv_uuid_c36 IS INITIAL AND lv_uuid_c32 IS INITIAL.
    ev_msg = COND string( WHEN ev_msg IS INITIAL THEN 'No fue posible convertir UUID.' ELSE ev_msg ).
    RETURN.
  ENDIF.

  DESCRIBE FIELD lv_puuid LENGTH lv_puuid_len IN CHARACTER MODE.
  IF lv_puuid_len = 32 AND lv_uuid_c32 IS NOT INITIAL.
    lv_puuid = lv_uuid_c32.
  ELSE.
    lv_puuid = lv_uuid_c36.
  ENDIF.

  "=== Construir URL QR (TIPOS ESTRICTOS) =============================
  CALL FUNCTION 'ZMFCFDI40_QR'
    EXPORTING
      puuid        = lv_puuid           " ZUUID (C32 o C36)
      rfc_emisor   = lv_rfc_em_c13      " *** CHAR13 ***
      rfc_receptor = lv_rfc_re_c13      " *** CHAR13 ***
      pextranjero  = lv_extranjero      " CHAR1
      ptotal       = lv_total           " CHAR18
      pselloemisor = lv_sello           " STRING
    IMPORTING
      purl         = lv_qr
    EXCEPTIONS
      OTHERS       = 1.

  IF lv_qr IS INITIAL.
    ev_msg = 'No se pudo construir CODE_QR (ZMFCFDI40_QR devolvió vacío).'.
    RETURN.
  ENDIF.

  "=== UPSERT en ZCP_SOVOS ============================================
  IF lv_found = abap_true.
    ls_sovos-code_qr = lv_qr.
    IF ls_sovos-uuid     IS INITIAL. ls_sovos-uuid     = lv_puuid. ENDIF.
    IF ls_sovos-sellocfd IS INITIAL. ls_sovos-sellocfd = lv_sello. ENDIF.
    UPDATE zcp_sovos FROM ls_sovos.
    IF sy-subrc = 0. ev_ok = abap_true. ELSE. ev_msg = 'UPDATE ZCP_SOVOS falló.'.
    ENDIF.
  ELSE.
    CLEAR ls_sovos.
    ls_sovos-bukrs    = is_cli-bukrs.
    ls_sovos-numcomp  = is_cli-numcomp.
    ls_sovos-uuid     = lv_puuid.
    ls_sovos-code_qr  = lv_qr.
    IF lv_sello IS NOT INITIAL.
      ls_sovos-sellocfd = lv_sello.
    ENDIF.
    INSERT zcp_sovos FROM ls_sovos.
    IF sy-subrc = 0. ev_ok = abap_true. ELSE. ev_msg = 'INSERT ZCP_SOVOS falló.'.
    ENDIF.
  ENDIF.

ENDFORM.

*---------------------------------------------------------------------*
* Extrae SelloCFD o Sello (8 últimos los recorta la FM) desde XML
*---------------------------------------------------------------------*
FORM f_extract_sello_from_xml_xstr
  USING    iv_xml_xstr TYPE xstring
  CHANGING cv_sello    TYPE string.

  DATA lv_str TYPE string.
  CLEAR cv_sello.
  IF iv_xml_xstr IS INITIAL.
    RETURN.
  ENDIF.

  PERFORM zconv_xstr_to_str USING iv_xml_xstr CHANGING lv_str.
  IF lv_str IS INITIAL.
    RETURN.
  ENDIF.

  " Busca SelloCFD="..." o Sello="..."
  FIND REGEX 'SelloCFD="([^"]+)"' IN lv_str SUBMATCHES cv_sello.
  IF cv_sello IS INITIAL.
    FIND REGEX 'Sello="([^"]+)"' IN lv_str SUBMATCHES cv_sello.
  ENDIF.

ENDFORM.

  "---------------------------------------------------------------
  " FORM SPLIT_70X4 - Parte STRING en 4 líneas (compat sin offsets)
  "---------------------------------------------------------------
  FORM split_70x4 USING    iv_text TYPE string
                  CHANGING ev_l1   TYPE ty_c70_x
                           ev_l2   TYPE ty_c70_x
                           ev_l3   TYPE ty_c70_x
                           ev_l4   TYPE ty_c70_x.

    DATA: lv_work TYPE string.

    CLEAR: ev_l1 , ev_l2 , ev_l3 , ev_l4.
    lv_work = iv_text.

    IF lv_work IS NOT INITIAL.
      ev_l1 = lv_work.   " truncado automático a 70
    ENDIF.

    IF strlen( lv_work ) > 70.
      SHIFT lv_work LEFT BY 70 PLACES. ev_l2 = lv_work.
    ELSE.
      RETURN.
    ENDIF.

    IF strlen( lv_work ) > 70.
      SHIFT lv_work LEFT BY 70 PLACES. ev_l3 = lv_work.
    ELSE.
      RETURN.
    ENDIF.

    IF strlen( lv_work ) > 70.
      SHIFT lv_work LEFT BY 70 PLACES. ev_l4 = lv_work.
    ENDIF.

  ENDFORM.

*  ---------------------------------------------------------------------*
*   Convierte un STRING a TLINE(132) – VERSIÓN TABLES (header line)
*  ---------------------------------------------------------------------*
  FORM string_to_tline
    TABLES ct_tline STRUCTURE tline    " <-- requiere header line
    USING  iv_text  TYPE string.


    DATA: lv_len  TYPE i,
          lv_off  TYPE i,
          lv_take TYPE i.

    " Compatible con releases antiguos
    REFRESH ct_tline.    " borra la tabla (NO usar ct_tline[])
    CLEAR   ct_tline.    " limpia la header line

    lv_len = strlen( iv_text ).
    lv_off = 0.

    WHILE lv_off < lv_len.
      CLEAR ct_tline.

      lv_take = lv_len - lv_off.
      IF lv_take > 132.
        lv_take = 132.
      ENDIF.

      ct_tline-tdformat = '*'.
      ct_tline-tdline   = iv_text+lv_off(lv_take).
      APPEND ct_tline.              " usa la header line

      lv_off = lv_off + 132.
    ENDWHILE.

  ENDFORM.

      IF strlen( lw ) > 70. SHIFT lw LEFT BY 70 PLACES. l3 = lw. ENDIF.
      IF strlen( lw ) > 70. SHIFT lw LEFT BY 70 PLACES. l4 = lw. ENDIF.

      CALL FUNCTION 'POPUP_TO_INFORM'
        EXPORTING

*  ---------------------------------------------------------------------*
*   Helper: obtener OBJKEY de MESSAGE de forma robusta
*  ---------------------------------------------------------------------*
  FORM sofo_objkey_from_info
    USING    is_info    TYPE sofolenti1
    CHANGING ev_objkey  TYPE swo_typeid
             ev_docid   TYPE string.

    FIELD-SYMBOLS: <f_any> TYPE any.
    DATA: lv_objtp TYPE string,
          lv_objyr TYPE string,
          lv_objno TYPE string.

    CLEAR: ev_objkey, ev_docid.

    ASSIGN COMPONENT 'OBJECT_ID' OF STRUCTURE is_info TO <f_any>.
    IF sy-subrc = 0 AND <f_any> IS ASSIGNED AND <f_any> IS NOT INITIAL.
      ev_objkey = CONV swo_typeid( <f_any> ).
      ev_docid  = CONV string( <f_any> ).
      RETURN.
    ENDIF.

    ASSIGN COMPONENT 'OBJTP'     OF STRUCTURE is_info TO <f_any>.
    IF sy-subrc = 0 AND <f_any> IS ASSIGNED. lv_objtp = <f_any>. ENDIF.
    IF lv_objtp IS INITIAL.
      ASSIGN COMPONENT 'OBJ_TYPE' OF STRUCTURE is_info TO <f_any>.
      IF sy-subrc = 0 AND <f_any> IS ASSIGNED. lv_objtp = <f_any>. ENDIF.
    ENDIF.

    ASSIGN COMPONENT 'OBJYR'     OF STRUCTURE is_info TO <f_any>.
    IF sy-subrc = 0 AND <f_any> IS ASSIGNED. lv_objyr = <f_any>. ENDIF.
    IF lv_objyr IS INITIAL.
      ASSIGN COMPONENT 'OBJ_YEAR' OF STRUCTURE is_info TO <f_any>.
      IF sy-subrc = 0 AND <f_any> IS ASSIGNED. lv_objyr = <f_any>. ENDIF.
    ENDIF.

    ASSIGN COMPONENT 'OBJNO'     OF STRUCTURE is_info TO <f_any>.
    IF sy-subrc = 0 AND <f_any> IS ASSIGNED. lv_objno = <f_any>. ENDIF.
    IF lv_objno IS INITIAL.
      ASSIGN COMPONENT 'OBJ_NO'   OF STRUCTURE is_info TO <f_any>.
      IF sy-subrc = 0 AND <f_any> IS ASSIGNED. lv_objno = <f_any>. ENDIF.
    ENDIF.

    IF lv_objtp IS NOT INITIAL AND lv_objyr IS NOT INITIAL AND lv_objno IS NOT INITIAL.
      CONCATENATE lv_objtp lv_objyr lv_objno INTO ev_objkey.
      ev_docid = lv_objno.
    ELSE.
      CLEAR: ev_objkey, ev_docid.
    ENDIF.

  ENDFORM.

  FORM gos_attach_xml_pac
    USING
             iv_vbeln TYPE vbrk-vbeln     " Nº factura
             iv_xml   TYPE string         " XML
             iv_tipo  TYPE char2          " 'RQ' / 'ER' / 'TI'
    CHANGING cv_docid TYPE string         " Devuelve ID del adjunto (opcional)

    DATA: ls_obj     TYPE borident,
          lv_vbeln10 TYPE vbrk-vbeln,
          lv_fname   TYPE string,
          lv_descr   TYPE so_obj_des,
          lt_hex     TYPE solix_tab,
          lv_size    TYPE i.

    DATA lv_enc TYPE abap_encoding.
    lv_enc = CONV abap_encoding( '4110' ).   " UTF-8

    CLEAR: cv_docid, cv_ok.

    " 1) Normalizar clave BUS2081
    lv_vbeln10 = iv_vbeln.

    ENDCASE.

  FORM anexa_xml_gos
    USING
      p_bukrs     TYPE bukrs
      p_belnr     TYPE vbrk-vbeln
      p_gjahr     TYPE gjahr
      p_tipo      TYPE char2        " 'TI' 'RQ' 'ER' 'AL'
      p_desc      TYPE so_obj_des
      p_xml_str   TYPE string       " opcional: XML timbrado como STRING
    CHANGING
      p_success   TYPE flag
      p_message   TYPE string.

    "--------------------------
    " Inicialización / helpers
    "--------------------------
    DATA: lv_vbeln10   TYPE vbrk-vbeln,
          lv_desc_base TYPE so_obj_des,
          lv_tipo      TYPE char2,
          lv_lines     TYPE i.

    DATA: ls_fol_id   TYPE soodk,
          ls_obj_id   TYPE soodk,
          ls_obj_data TYPE sood1,
          lt_objhead  TYPE soli_tab,
          lt_soli     TYPE soli_tab,
          lt_solix    TYPE solix_tab,
          ls_object   TYPE borident,
          ls_note     TYPE borident,
          ls_folmem_k TYPE sofmk,
          lv_ep_note  TYPE borident-objkey.

    DATA: ls_xml40    TYPE zcfdi_40_xml,
          lt_log      TYPE STANDARD TABLE OF zcfdi_xml_log,
          ls_log      TYPE zcfdi_xml_log,
          ls_head     TYPE zcfdi_head_40,
          lv_any_done TYPE abap_bool VALUE abap_false.

    FIELD-SYMBOLS: <fs_any> TYPE any.

    CLEAR: p_success, p_message.

    "--------------------------
    " Validaciones básicas
    "--------------------------
    IF p_bukrs IS INITIAL OR p_belnr IS INITIAL OR p_gjahr IS INITIAL.
      p_message = 'Parámetros obligatorios vacíos (BUKRS/BELNR/GJAHR).'.
      p_success = abap_false.
      EXIT.
    ENDIF.

    " VBELN a 10 (ALPHA)
    lv_vbeln10 = p_belnr.
    CALL FUNCTION 'CONVERSION_EXIT_ALPHA_INPUT'
      EXPORTING
        input  = lv_vbeln10
      IMPORTING
        output = lv_vbeln10.

    " Normaliza tipo
    lv_tipo = p_tipo.
    TRANSLATE lv_tipo TO UPPER CASE.
    IF lv_tipo IS INITIAL.
      lv_tipo = 'AL'.
    ENDIF.

    " Descripción base
    IF p_desc IS INITIAL.
      CONCATENATE 'XML CFDI' lv_vbeln10 INTO lv_desc_base SEPARATED BY space.
    ELSE.
      lv_desc_base = p_desc.
    ENDIF.

    " Carpeta SAPoffice (B = privado)
    CALL FUNCTION 'SO_FOLDER_ROOT_ID_GET'
      EXPORTING
        region    = 'B'
      IMPORTING
        folder_id = ls_fol_id.
    IF sy-subrc <> 0 OR ls_fol_id-objtp IS INITIAL.
      p_message = 'No se obtuvo carpeta SAPoffice (SO_FOLDER_ROOT_ID_GET).'.
      p_success = abap_false.
      EXIT.
    ENDIF.

    "--------------------------
    " 1) Construir contenido lt_soli
    "    Preferencia: p_xml_str (STRING) -> BD
    "--------------------------
    REFRESH lt_soli.

    " A) Si viene STRING por parámetro, úsalo tal cual
    IF p_xml_str IS NOT INITIAL.
      CALL FUNCTION 'SO_STRING_TO_TAB'
        EXPORTING
          content_str = p_xml_str
        TABLES
          content_tab = lt_soli.
      IF lt_soli IS NOT INITIAL.
        lv_any_done = abap_true.
      ENDIF.
    ENDIF.

    " B) Si aún no hay contenido y el TIPO lo pide, busca en tablas Z
    IF lv_any_done = abap_false AND ( lv_tipo = 'TI' OR lv_tipo = 'AL' ).
      " ZCFDI_40_XML (STRING)
      CLEAR ls_xml40.
      SELECT SINGLE * INTO ls_xml40 FROM zcfdi_40_xml WHERE factura = lv_vbeln10.
      IF sy-subrc = 0 AND ls_xml40-xml IS NOT INITIAL.
        CALL FUNCTION 'SO_STRING_TO_TAB'
          EXPORTING
            content_str = ls_xml40-xml
          TABLES
            content_tab = lt_soli.
        IF lt_soli IS NOT INITIAL.
          lv_any_done = abap_true.
        ENDIF.
      ENDIF.

      " Fallback: ZCFDI_XML_LOG tipo 'TI' (XSTRING)
      IF lv_any_done = abap_false.
        REFRESH lt_log.
        SELECT * INTO TABLE lt_log
          FROM zcfdi_xml_log
          WHERE bukrs = p_bukrs
            AND belnr = lv_vbeln10
            AND gjahr = p_gjahr
            AND tipo_doc = 'TI'.
        IF sy-subrc = 0.
          SORT lt_log BY created_on DESCENDING created_at DESCENDING.
          READ TABLE lt_log INTO ls_log INDEX 1.
          IF sy-subrc = 0 AND ls_log-xml_req IS NOT INITIAL.
            REFRESH lt_solix.
            CALL FUNCTION 'SCMS_XSTRING_TO_BINARY'
              EXPORTING
                buffer     = ls_log-xml_req
              TABLES
                binary_tab = lt_solix.
            CALL FUNCTION 'SO_SOLIXTAB_TO_SOLITAB'
              EXPORTING
                ip_solixtab = lt_solix
              IMPORTING
                ep_solitab  = lt_soli.
            IF lt_soli IS NOT INITIAL.
              lv_any_done = abap_true.
            ENDIF.
          ENDIF.
        ENDIF.
      ENDIF.

      " Fallback: ZCFDI_HEAD_40 (intenta campos típicos, string/xstring)
      IF lv_any_done = abap_false.
        CLEAR ls_head.
        SELECT SINGLE * INTO ls_head
          FROM zcfdi_head_40
          WHERE bukrs = p_bukrs
            AND belnr = lv_vbeln10
            AND gjahr = p_gjahr.
        IF sy-subrc = 0.
          DATA lt_try_ti TYPE STANDARD TABLE OF string WITH DEFAULT KEY.
          APPEND 'XML_TIMB'     TO lt_try_ti.
          APPEND 'XML_TIMBRADO' TO lt_try_ti.
          APPEND 'XML_TFD'      TO lt_try_ti.
          APPEND 'XML'          TO lt_try_ti.
          LOOP AT lt_try_ti INTO DATA(lv_name).
            ASSIGN COMPONENT lv_name OF STRUCTURE ls_head TO <fs_any>.
            IF sy-subrc = 0 AND <fs_any> IS ASSIGNED AND <fs_any> IS NOT INITIAL.
              " Primero como STRING
              DATA(lv_try_s) = ``.
              TRY.
                  lv_try_s = CONV string( <fs_any> ).
                CATCH cx_sy_move_cast_error cx_sy_conversion_error.
              ENDTRY.
              IF lv_try_s IS NOT INITIAL.
                REFRESH lt_soli.
                CALL FUNCTION 'SO_STRING_TO_TAB'
                  EXPORTING
                    content_str = lv_try_s
                  TABLES
                    content_tab = lt_soli.
              ENDIF.
              " Si no, como XSTRING
              IF lt_soli IS INITIAL.
                DATA(lv_try_x) = CONV xstring( '' ).
                TRY.
                    lv_try_x = CONV xstring( <fs_any> ).
                  CATCH cx_sy_move_cast_error cx_sy_conversion_error.
                ENDTRY.
                IF lv_try_x IS NOT INITIAL.
                  REFRESH lt_solix.
                  CALL FUNCTION 'SCMS_XSTRING_TO_BINARY'
                    EXPORTING
                      buffer     = lv_try_x
                    TABLES
                      binary_tab = lt_solix.
                  CALL FUNCTION 'SO_SOLIXTAB_TO_SOLITAB'
                    EXPORTING
                      ip_solixtab = lt_solix
                    IMPORTING
                      ep_solitab  = lt_soli.
                ENDIF.
              ENDIF.
              IF lt_soli IS NOT INITIAL.
                lv_any_done = abap_true.
                EXIT.
              ENDIF.
            ENDIF.
          ENDLOOP.
        ENDIF.
      ENDIF.
    ENDIF.

    " RQ (si se pidió explícito o AL y aún no adjuntaste nada)
    IF lv_any_done = abap_false AND ( lv_tipo = 'RQ' OR lv_tipo = 'AL' ).
      REFRESH lt_log.
      SELECT * INTO TABLE lt_log
        FROM zcfdi_xml_log
        WHERE bukrs = p_bukrs
          AND belnr = lv_vbeln10
          AND gjahr = p_gjahr
          AND tipo_doc = 'RQ'.
      IF sy-subrc = 0.
        SORT lt_log BY created_on DESCENDING created_at DESCENDING.

          CALL FUNCTION 'SCMS_XSTRING_TO_BINARY'
            EXPORTING
              buffer     = ls_log-xml_req
            TABLES
              binary_tab = lt_solix.
          CALL FUNCTION 'SO_SOLIXTAB_TO_SOLITAB'
            EXPORTING
              ip_solixtab = lt_solix
            IMPORTING
              ep_solitab  = lt_soli.
          IF lt_soli IS NOT INITIAL.
            lv_any_done = abap_true.
          ENDIF.
        ENDIF.
      ENDIF.
    ENDIF.

    " ER (si se pidió explícito o AL y aún no adjuntaste nada)
    IF lv_any_done = abap_false AND ( lv_tipo = 'ER' OR lv_tipo = 'AL' ).
      REFRESH lt_log.
      SELECT * INTO TABLE lt_log
        FROM zcfdi_xml_log
        WHERE bukrs = p_bukrs
          AND belnr = lv_vbeln10
          AND gjahr = p_gjahr
          AND tipo_doc = 'ER'.
      IF sy-subrc = 0.
        SORT lt_log BY created_on DESCENDING created_at DESCENDING.
        READ TABLE lt_log INTO ls_log INDEX 1.
        IF sy-subrc = 0 AND ls_log-xml_req IS NOT INITIAL.
          REFRESH lt_solix.
          CALL FUNCTION 'SCMS_XSTRING_TO_BINARY'
            EXPORTING
              buffer     = ls_log-xml_req
            TABLES
              binary_tab = lt_solix.
          CALL FUNCTION 'SO_SOLIXTAB_TO_SOLITAB'
            EXPORTING
              ip_solixtab = lt_solix
            IMPORTING
              ep_solitab  = lt_soli.
          IF lt_soli IS NOT INITIAL.
            lv_any_done = abap_true.
          ENDIF.
        ENDIF.
      ENDIF.
    ENDIF.

    IF lt_soli IS INITIAL.
      p_message = |No hay XML para adjuntar (tipo={ lv_tipo }).|.
      p_success = abap_false.
      EXIT.
    ENDIF.

    "--------------------------
    " 2) Insertar en SO + vincular a BUS2081 (sin COMMIT)
    "--------------------------
    CLEAR ls_obj_data.
    ls_obj_data-objsns   = 'O'.
    ls_obj_data-objla    = sy-langu.
    ls_obj_data-objdes   = lv_desc_base.
    ls_obj_data-file_ext = 'XML'.
    DESCRIBE TABLE lt_soli LINES lv_lines.
    ls_obj_data-objlen   = lv_lines * 510.

    CALL FUNCTION 'SO_OBJECT_INSERT'
      EXPORTING
        folder_id        = ls_fol_id
        object_type      = 'RAW'
        object_hd_change = ls_obj_data
      IMPORTING
        object_id        = ls_obj_id
      TABLES
        objhead          = lt_objhead
        objcont          = lt_soli.
    IF sy-subrc <> 0.
      p_message = 'Error al crear objeto SAPoffice (SO_OBJECT_INSERT).'.
      p_success = abap_false.
      EXIT.
    ENDIF.

    ls_object-objtype = 'BUS2081'.
    ls_object-objkey  = lv_vbeln10.

    ls_folmem_k-foltp = ls_fol_id-objtp.
    ls_folmem_k-folyr = ls_fol_id-objyr.
    ls_folmem_k-folno = ls_fol_id-objno.
    ls_folmem_k-doctp = ls_obj_id-objtp.
    ls_folmem_k-docyr = ls_obj_id-objyr.
    ls_folmem_k-docno = ls_obj_id-objno.
    lv_ep_note        = ls_folmem_k.

    ls_note-objtype   = 'MESSAGE'.
    ls_note-objkey    = lv_ep_note.

    CALL FUNCTION 'BINARY_RELATION_CREATE'
      EXPORTING
        obj_rolea      = ls_object
        obj_roleb      = ls_note
        relationtype   = 'ATTA'
      EXCEPTIONS
        no_model       = 1
        internal_error = 2
        unknown        = 3
        OTHERS         = 4.

    IF sy-subrc = 0.
      p_success = abap_true.
      p_message = |Adjunto GOS creado para { lv_vbeln10 } (tipo={ lv_tipo }).|.
    ELSE.
      p_success = abap_false.
      p_message = |Objeto creado, pero falló relación GOS (ATTA) para { lv_vbeln10 } (tipo={ lv_tipo }).|.
    ENDIF.

  ENDFORM.                    " anexa_xml_gos

*   CHANGING

*     ev_cerrado TYPE abap_bool

  FORM validar_mes_cerrado_fi
    USING    iv_bukrs   TYPE bukrs
             iv_budat   TYPE sy-datum
             iv_idmsg   TYPE symsgid
    CHANGING cv_errmsg  TYPE string
             ev_cerrado TYPE abap_bool.

    DATA: lv_gjahr TYPE gjahr,
          lv_poper TYPE poper.   " I_MONAT de FI_PERIOD_CHECK es POPER (NUMC3)

    CLEAR: ev_cerrado, cv_errmsg.

    " Validación mínima de entrada
    IF iv_bukrs IS INITIAL OR iv_budat IS INITIAL.
      ev_cerrado = abap_true.
      cv_errmsg  = |Faltan datos: sociedad o fecha.|.
      RETURN.
    ENDIF.

    " Derivar año/período desde la fecha (POPER = 001..012)
    lv_gjahr = CONV gjahr( iv_budat(4) ).
    lv_poper = CONV poper( iv_budat+4(2) ).  " '03' -> '003'

    " Validación estándar OB52
    CALL FUNCTION 'FI_PERIOD_CHECK'
      EXPORTING
        i_bukrs      = iv_bukrs
        i_gjahr      = lv_gjahr
        i_monat      = lv_poper      " usar POPER (NUMC3)
        i_koart      = '+'           " validación general (todas las clases)
      EXCEPTIONS
        error_period = 1
        OTHERS       = 2.

    IF sy-subrc = 0.
      ev_cerrado = abap_false.   " ABIERTO
      RETURN.
    ENDIF.

    " CERRADO / no permitido
    ev_cerrado = abap_true.
    cv_errmsg  = |Período { lv_poper }/{ lv_gjahr } cerrado o no válido para la sociedad { iv_bukrs }.|.

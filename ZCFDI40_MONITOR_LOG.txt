FORM user_command USING p_ucomm TYPE sy-ucomm
                        rs_self TYPE slis_selfield.

  DATA: lv_idx    TYPE sy-tabix,
        lv_vbeln  TYPE vbrk-vbeln,
        ls_out    TYPE ty_out,
        lv_result TYPE c.            " 'T' timbrado, 'E' error, 'N' sin cambios

  CASE p_ucomm.

    WHEN '&F03' OR '&F12' OR '&F15' OR 'BACK' OR 'EXIT' OR 'CANCEL'.
      rs_self-exit = 'X'.
      RETURN.

    WHEN '&IC1'.
      READ TABLE gt_out INDEX rs_self-tabindex INTO ls_out.
      IF sy-subrc <> 0.
        RETURN.
      ENDIF.

      CASE rs_self-fieldname.

        WHEN 'FACTURA'.
          lv_vbeln = ls_out-factura.
          CALL FUNCTION 'CONVERSION_EXIT_ALPHA_INPUT'
            EXPORTING
              input  = lv_vbeln
            IMPORTING
              output = lv_vbeln.
          IF lv_vbeln IS INITIAL.
            MESSAGE 'Número de factura vacío.' TYPE 'S' DISPLAY LIKE 'E'.
            RETURN.
          ENDIF.
          SET PARAMETER ID 'VF' FIELD lv_vbeln.
          CALL TRANSACTION 'VF03' AND SKIP FIRST SCREEN.
          rs_self-refresh = 'X'.



        WHEN 'REQ_ICON' OR 'STATUS_ICON'.
          " ⬇️ abrir XML (prioridad TI→RQ→ER); 4º parámetro opcional
          PERFORM open_xml_from_log    USING ls_out-factura ls_out-bukrs ls_out-gjahr.


        WHEN 'ERROR_TIM_ICON' OR 'ERROR_ICON'.
          " ⬇️ abrir XML de error (ER)
          PERFORM open_xml_from_error  USING ls_out-factura ls_out-bukrs ls_out-gjahr.


        WHEN 'TIMB_ICON'.
          PERFORM open_xml_timb
                  USING ls_out-factura ls_out-bukrs ls_out-gjahr.

        WHEN 'FACT_ICO'. " Ejecutar timbrado solo si NO está timbrada
          IF ls_out-estado = 'Timbrada'.
            MESSAGE 'La factura ya está timbrada (TI).' TYPE 'S'.
            RETURN.
          ENDIF.

          CALL FUNCTION 'SAPGUI_PROGRESS_INDICATOR'
            EXPORTING
              percentage = 10
              text       = |Timbrando { ls_out-factura }...|.

          CLEAR lv_result.
          PERFORM pacsovos_fac USING    ls_out-factura
                                        ls_out-bukrs
                                        ls_out-gjahr
                               CHANGING lv_result.       "<<< devuelve T / E / N

          "---- Refresh inteligente ---------------------------------
          CASE lv_result.
            WHEN 'T'.   " Timbró → recargar datos (pero conservando filtros de barra)
              PERFORM get_data.
              PERFORM apply_current_filter.
              rs_self-refresh = 'X'.

            WHEN 'E'.   " Error → NO recargar; solo actualizar la fila
              PERFORM mark_row_error USING rs_self-tabindex.

            WHEN OTHERS. " N/sin cambios → no hagas nada
          ENDCASE.
          "-----------------------------------------------------------

      ENDCASE.

    WHEN 'LOG'.
      IF rs_self-tabindex IS INITIAL.
        GET CURSOR LINE lv_idx.
      ELSE.
        lv_idx = rs_self-tabindex.
      ENDIF.
      IF lv_idx > 0.
        READ TABLE gt_out INDEX lv_idx INTO ls_out.
        IF sy-subrc = 0.
          PERFORM show_log USING ls_out-factura ls_out-bukrs ls_out-gjahr.
        ENDIF.
      ENDIF.

    WHEN 'GRN'.
      gv_last_filter = 'GRN'.
      REFRESH gt_out.
      LOOP AT gt_out_all INTO ls_out
        WHERE estado = 'Timbrada'
           OR estado = 'Éxito (RQ)'.
        APPEND ls_out TO gt_out.
      ENDLOOP.
      rs_self-refresh = 'X'.

    WHEN 'YEL'.
      gv_last_filter = 'YEL'.
      REFRESH gt_out.
      LOOP AT gt_out_all INTO ls_out
        WHERE estado = 'En proceso'
           OR estado = 'Sin solicitud'.
        APPEND ls_out TO gt_out.
      ENDLOOP.
      rs_self-refresh = 'X'.

    WHEN 'RED'.
      gv_last_filter = 'RED'.
      REFRESH gt_out.
      LOOP AT gt_out_all INTO ls_out
        WHERE estado = 'Error'.
        APPEND ls_out TO gt_out.
      ENDLOOP.
      rs_self-refresh = 'X'.

    WHEN 'ALL'.
      gv_last_filter = 'ALL'.
      gt_out[] = gt_out_all[].
      rs_self-refresh = 'X'.

  ENDCASE.
ENDFORM.

        lr_bukrs TYPE RANGE OF vbrk-bukrs,
        lr_fech  TYPE RANGE OF vbrk-fkdat,
        lr_kunnr TYPE RANGE OF vbrk-kunrg,
        lr_mon   TYPE RANGE OF vbrk-waerk.

  DATA: ls_rvbeln LIKE LINE OF lr_vbeln,
        ls_rbukrs LIKE LINE OF lr_bukrs,
        ls_rfech  LIKE LINE OF lr_fech,
        ls_rkunnr LIKE LINE OF lr_kunnr,
        ls_rmon   LIKE LINE OF lr_mon,
        ls_smon   LIKE LINE OF s_mon.

  CLEAR: gt_out, gt_out_all.

  " Copiar select-options a rangos locales
  lr_vbeln[] = s_vbeln[].
  lr_bukrs[] = s_bukrs[].
  lr_fech[]  = s_fech[].
  lr_kunnr[] = s_kunnr[].

  " Moneda: normalizar a mayúsculas
  REFRESH lr_mon.
  LOOP AT s_mon INTO ls_smon.
    CLEAR ls_rmon.
    ls_rmon-sign   = ls_smon-sign.
    ls_rmon-option = ls_smon-option.
    ls_rmon-low    = ls_smon-low.
    ls_rmon-high   = ls_smon-high.
    TRANSLATE ls_rmon-low  TO UPPER CASE.
    TRANSLATE ls_rmon-high TO UPPER CASE.
    APPEND ls_rmon TO lr_mon.
  ENDLOOP.

  " Si algún rango está vacío, darle intervalo "todo"
  IF lr_vbeln[] IS INITIAL.
    CLEAR ls_rvbeln. ls_rvbeln-sign = 'I'. ls_rvbeln-option = 'BT'.
    ls_rvbeln-low  = '0000000000'. ls_rvbeln-high = '9999999999'.
    APPEND ls_rvbeln TO lr_vbeln.
  ENDIF.

  IF lr_bukrs[] IS INITIAL.
    CLEAR ls_rbukrs. ls_rbukrs-sign = 'I'. ls_rbukrs-option = 'BT'.
    ls_rbukrs-low  = '0000'. ls_rbukrs-high = 'ZZZZ'.
    APPEND ls_rbukrs TO lr_bukrs.
  ENDIF.

  IF lr_fech[] IS INITIAL.
    CLEAR ls_rfech. ls_rfech-sign = 'I'. ls_rfech-option = 'BT'.
    ls_rfech-low  = '00000000'. ls_rfech-high = '99991231'.
    APPEND ls_rfech TO lr_fech.
  ENDIF.

  IF lr_kunnr[] IS INITIAL.
    CLEAR ls_rkunnr. ls_rkunnr-sign = 'I'. ls_rkunnr-option = 'BT'.
    ls_rkunnr-low  = '0000000000'. ls_rkunnr-high = '9999999999'.
    APPEND ls_rkunnr TO lr_kunnr.
  ENDIF.

  IF lr_mon[] IS INITIAL.
    CLEAR ls_rmon. ls_rmon-sign = 'I'. ls_rmon-option = 'BT'.
    ls_rmon-low  = ' '. ls_rmon-high = 'ZZZZZ'.
    APPEND ls_rmon TO lr_mon.
  ENDIF.

  "--------------- 1) Facturas base (VBRK) ---------------
  PERFORM prepare_fech_db.
  SELECT vbeln bukrs gjahr fkdat kunrg waerk netwr vkorg vtweg spart
    INTO TABLE lt_vbrk
    FROM vbrk
    WHERE vbeln IN lr_vbeln
      AND bukrs IN lr_bukrs
      AND fkdat IN lr_fech_db
      AND kunrg IN lr_kunnr
      AND waerk IN lr_mon.

  IF lt_vbrk IS INITIAL.
    EXIT.
  ENDIF.

  "--------------- 2) Llaves para HEAD y LOG ------------
  LOOP AT lt_vbrk INTO ls_vbrk.
    IF ls_vbrk-gjahr IS INITIAL.
      lv_gjahr_key = ls_vbrk-fkdat(4).
    ELSE.
      lv_gjahr_key = ls_vbrk-gjahr.
    ENDIF.

    ls_kh-factura = ls_vbrk-vbeln.
    ls_kh-bukrs   = ls_vbrk-bukrs.
    ls_kh-gjahr   = lv_gjahr_key.
    INSERT ls_kh INTO TABLE lt_keys_h.

    ls_kl-belnr = ls_vbrk-vbeln.
    ls_kl-bukrs = ls_vbrk-bukrs.
    ls_kl-gjahr = lv_gjahr_key.
    INSERT ls_kl INTO TABLE lt_keys_l.

    INSERT ls_vbrk-kunrg INTO TABLE lt_kunnr.
  ENDLOOP.

  "--------------- 3) HEAD (LA QUE MANDA) ---------------
  IF NOT lt_keys_h IS INITIAL.
    SELECT *
      INTO TABLE lt_head
      FROM zcfdi_head_40
      FOR ALL ENTRIES IN lt_keys_h
      WHERE factura = lt_keys_h-factura
        AND bukrs   = lt_keys_h-bukrs
        AND gjahr   = lt_keys_h-gjahr.
    IF NOT lt_head IS INITIAL.
      lt_head_h[] = lt_head[].
    ENDIF.
  ENDIF.

  "--------------- 4) LOGs (complemento) ---------------
  IF NOT lt_keys_l IS INITIAL.
    SELECT *
      INTO TABLE lt_log_all
      FROM zcfdi_xml_log
      FOR ALL ENTRIES IN lt_keys_l
      WHERE belnr = lt_keys_l-belnr
        AND bukrs = lt_keys_l-bukrs
        AND gjahr = lt_keys_l-gjahr.

    IF NOT lt_log_all IS INITIAL.
      SORT lt_log_all BY belnr bukrs gjahr created_on DESCENDING created_at DESCENDING.
      lt_log_last[] = lt_log_all[].
      DELETE ADJACENT DUPLICATES FROM lt_log_last COMPARING belnr bukrs gjahr.
      lt_last_h[] = lt_log_last[].

      CLEAR: lt_er_keys, lt_rq_keys, lt_ti_idx.
      LOOP AT lt_log_all INTO ls_log.
        CASE ls_log-tipo_doc.
          WHEN 'TI'.
            READ TABLE lt_ti_idx WITH TABLE KEY
                 belnr = ls_log-belnr bukrs = ls_log-bukrs gjahr = ls_log-gjahr
                 TRANSPORTING NO FIELDS.
            IF sy-subrc <> 0.
              ls_ti_idx-belnr      = ls_log-belnr.
              ls_ti_idx-bukrs      = ls_log-bukrs.
              ls_ti_idx-gjahr      = ls_log-gjahr.
              ls_ti_idx-created_on = ls_log-created_on.
              ls_ti_idx-created_at = ls_log-created_at.
              INSERT ls_ti_idx INTO TABLE lt_ti_idx.
            ENDIF.
          WHEN 'ER'.
            INSERT VALUE ty_flag_key( belnr = ls_log-belnr bukrs = ls_log-bukrs gjahr = ls_log-gjahr )
              INTO TABLE lt_er_keys.
          WHEN 'RQ'.
            INSERT VALUE ty_flag_key( belnr = ls_log-belnr bukrs = ls_log-bukrs gjahr = ls_log-gjahr )
              INTO TABLE lt_rq_keys.
        ENDCASE.
      ENDLOOP.
    ENDIF.
  ENDIF.

  "--------------- 5) Clientes -------------------
  IF NOT lt_kunnr IS INITIAL.
    SELECT kunnr name1
      INTO TABLE lt_knam
      FROM kna1
      FOR ALL ENTRIES IN lt_kunnr
      WHERE kunnr = lt_kunnr-table_line.
    SORT lt_knam BY kunnr.
  ENDIF.

  "--------------- 6) Construcción de salida -------
  DATA lv_has_er TYPE abap_bool.
  DATA lv_has_rq TYPE abap_bool.

  LOOP AT lt_vbrk INTO ls_vbrk.

    IF ls_vbrk-gjahr IS INITIAL.
      lv_gjahr = ls_vbrk-fkdat(4).
    ELSE.
      lv_gjahr = ls_vbrk-gjahr.
    ENDIF.

    CLEAR gs_out.
    gs_out-factura = ls_vbrk-vbeln.
    gs_out-bukrs   = ls_vbrk-bukrs.
    gs_out-gjahr   = lv_gjahr.
    gs_out-fecha   = ls_vbrk-fkdat.
    gs_out-kunnr   = ls_vbrk-kunrg.
    gs_out-total   = ls_vbrk-netwr.
    gs_out-moneda  = ls_vbrk-waerk.
    gs_out-vkorg   = ls_vbrk-vkorg.
    gs_out-vtweg   = ls_vbrk-vtweg.
    gs_out-spart   = ls_vbrk-spart.

    READ TABLE lt_knam INTO ls_knam WITH KEY kunnr = gs_out-kunnr BINARY SEARCH.
    IF sy-subrc = 0.
      gs_out-nombre = ls_knam-name1.
    ENDIF.

    " Default TipoCambioXML: 1 si MXN
    IF gs_out-moneda = c_mxn.
      gs_out-tipocambioxml = '1'.
    ELSE.
      CLEAR gs_out-tipocambioxml.
    ENDIF.

    CLEAR: gs_out-req_icon, gs_out-error_tim_icon, gs_out-timb_icon, gs_out-fact_ico,
           gs_out-uuid, gs_out-timb_on, gs_out-timb_at, gs_out-estado, gs_out-status_icon,
           gs_out-last_tdoc, gs_out-last_on, gs_out-last_at.

    " ===== A) HEAD MANDA: si existe en HEAD → Timbrada =====
    READ TABLE lt_head_h INTO ls_head
         WITH KEY factura = ls_vbrk-vbeln bukrs = ls_vbrk-bukrs gjahr = lv_gjahr.
    IF sy-subrc = 0.
      gs_out-estado      = 'Timbrada'.
      gs_out-status_icon = icon_led_green.
      gs_out-timb_icon   = icon_led_green.

      IF ls_head-uuid IS NOT INITIAL.
        gs_out-uuid = ls_head-uuid.
      ENDIF.
      IF ls_head-vkorg IS NOT INITIAL. gs_out-vkorg = ls_head-vkorg. ENDIF.
      IF ls_head-vtweg IS NOT INITIAL. gs_out-vtweg = ls_head-vtweg. ENDIF.
      IF ls_head-spart IS NOT INITIAL. gs_out-spart = ls_head-spart. ENDIF.

      " --- TIPOCAMBIOXML (HEAD manda; fallback XML) ---
      IF gs_out-moneda = c_mxn.
        gs_out-tipocambioxml = '1'.
      ELSEIF ls_head-tipocambioxml IS NOT INITIAL.
        gs_out-tipocambioxml = ls_head-tipocambioxml.
      ELSEIF ls_head-xml_timb IS NOT INITIAL.
        CLEAR: lv_xml_str, lv_tcxml_str, lv_pos_start, lv_pos_end, lv_len.
        CALL FUNCTION 'ECATT_CONV_XSTRING_TO_STRING'
          EXPORTING
            im_xstring  = ls_head-xml_timb
            im_encoding = 'UTF-8'
          IMPORTING
            ex_string   = lv_xml_str.
        PERFORM sanitizar_xml_final CHANGING lv_xml_str.

        IF sy-subrc = 0 AND lv_xml_str IS NOT INITIAL.
          FIND FIRST OCCURRENCE OF 'TipoCambio="' IN lv_xml_str.
          IF sy-subrc = 0.
            lv_pos_start = sy-fdpos + 12.
            lv_pos_end   = lv_pos_start.
            WHILE lv_pos_end < strlen( lv_xml_str ) AND lv_xml_str+lv_pos_end(1) <> '"'.
              lv_pos_end = lv_pos_end + 1.
            ENDWHILE.
            lv_len = lv_pos_end - lv_pos_start.
            IF lv_len > 0.
              lv_tcxml_str = lv_xml_str+lv_pos_start(lv_len).
              IF lv_tcxml_str IS NOT INITIAL.
                gs_out-tipocambioxml = lv_tcxml_str.
              ENDIF.
            ENDIF.
          ENDIF.
        ENDIF.
      ENDIF.

      " Fecha/hora timbrado como complemento desde LOG TI (si existe)
      READ TABLE lt_ti_idx INTO ls_ti_idx
           WITH TABLE KEY belnr = ls_vbrk-vbeln bukrs = ls_vbrk-bukrs gjahr = lv_gjahr.
      IF sy-subrc = 0.
        gs_out-timb_on = ls_ti_idx-created_on.
        gs_out-timb_at = ls_ti_idx-created_at.
      ENDIF.

      CLEAR: gs_out-req_icon, gs_out-error_tim_icon, gs_out-fact_ico.

    ELSE.
      " ===== B) No hay HEAD → no timbrada; evaluar LOGs =====
      lv_has_er = abap_false.
      lv_has_rq = abap_false.

      READ TABLE lt_er_keys WITH TABLE KEY
           belnr = ls_vbrk-vbeln bukrs = ls_vbrk-bukrs gjahr = lv_gjahr
           TRANSPORTING NO FIELDS.
      IF sy-subrc = 0. lv_has_er = abap_true. ENDIF.

      READ TABLE lt_rq_keys WITH TABLE KEY
           belnr = ls_vbrk-vbeln bukrs = ls_vbrk-bukrs gjahr = lv_gjahr
           TRANSPORTING NO FIELDS.
      IF sy-subrc = 0. lv_has_rq = abap_true. ENDIF.

      IF lv_has_rq = abap_true. gs_out-req_icon        = icon_led_green. ENDIF.
      IF lv_has_er = abap_true. gs_out-error_tim_icon = icon_led_red.   ENDIF.

      gs_out-fact_ico = icon_execute_object. " Ejecutar siempre en no timbradas

      IF lv_has_er = abap_true.
        gs_out-estado      = 'Error'.
        gs_out-status_icon = icon_led_red.
      ELSEIF lv_has_rq = abap_true.
        gs_out-estado      = 'Éxito (RQ)'.
        gs_out-status_icon = icon_led_green.
      ELSE.
        READ TABLE lt_last_h INTO ls_log
             WITH KEY belnr = ls_vbrk-vbeln bukrs = ls_vbrk-bukrs gjahr = lv_gjahr.
        IF sy-subrc = 0.
          gs_out-last_tdoc = ls_log-tipo_doc.
          gs_out-last_on   = ls_log-created_on.
          gs_out-last_at   = ls_log-created_at.
          gs_out-estado      = 'En proceso'.
          gs_out-status_icon = icon_led_yellow.
        ELSE.
          gs_out-estado      = 'Sin solicitud'.
          gs_out-status_icon = icon_led_yellow.
        ENDIF.
      ENDIF.
    ENDIF.

    APPEND gs_out TO gt_out.
  ENDLOOP.

  gt_out_all[] = gt_out[].

  "--------------- 7) (Opcional) Filtro por moneda salida ------------
  " Ya no es necesario porque el SELECT ya filtró con lr_mon.

  "--------------- 8) Filtro por checks de selección -----------------
  PERFORM filter_by_date.
  PERFORM filter_by_date.
  PERFORM filter_by_state.

ENDFORM.




*---------------------------------------------------------------------*
* Timbrado (PAC) con resultado: 'T' timbrado, 'E' error, 'N' sin cambios
*---------------------------------------------------------------------*
FORM pacsovos_fac  USING    p_factura TYPE vbrk-vbeln
                            p_bukrs   TYPE bukrs
                            p_gjahr   TYPE gjahr
                   CHANGING pv_result TYPE c.

  "--------------------- Declaraciones -------------------------------
  DATA lv_gjahr             TYPE gjahr.
  DATA lv_fkdat             TYPE vbrk-fkdat.
  DATA lv_docdate           TYPE sy-datum.

  DATA lv_uuid              TYPE zuuid.
  DATA lv_url               TYPE zqr40.
  DATA lv_xml_req           TYPE string.
  DATA lv_xml_timb          TYPE string.
  DATA lv_err_xml           TYPE string.
  DATA lv_err_msg           TYPE string.
  DATA lv_err_code          TYPE string.

  DATA lv_xstr              TYPE xstring.
  DATA lv_uuid_existente    TYPE zuuid.

  DATA gv_first_sales_order TYPE vbeln_va.

  DATA lt_head_cfdi         TYPE tt_head_cfdi.
  DATA lt_det_cfdi          TYPE tt_det_cfdi.
  DATA ls_head_cfdi         TYPE zcfdi_head_40.

  DATA ls_log               TYPE zcfdi_xml_log.
  DATA ls_log_er            TYPE zcfdi_xml_log.
  DATA ls_log_ti            TYPE zcfdi_xml_log.

  DATA lo_cfdi              TYPE REF TO zcl_cfdi40_nast.

  pv_result = 'N'.

  "--------------------- Año ejercicio / fecha doc -------------------
  CLEAR lv_fkdat.
  PERFORM prepare_fech_db.
  SELECT SINGLE fkdat INTO lv_fkdat FROM vbrk WHERE vbeln = p_factura.
  IF sy-subrc = 0 AND lv_fkdat IS NOT INITIAL.
    lv_gjahr = lv_fkdat(4).
  ELSEIF p_gjahr IS NOT INITIAL.
    lv_gjahr = p_gjahr.
  ELSE.
    lv_gjahr = sy-datum(4).
  ENDIF.

  lv_docdate = lv_fkdat.
  IF lv_docdate IS INITIAL.
    lv_docdate = sy-datum.
  ENDIF.

  "--------------------- Construir HEAD/DET --------------------------
  REFRESH lt_head_cfdi.
  REFRESH lt_det_cfdi.
  PERFORM getfactura USING p_factura lv_docdate
                     CHANGING lt_head_cfdi lt_det_cfdi.

  READ TABLE lt_head_cfdi INTO ls_head_cfdi INDEX 1.
  IF sy-subrc <> 0.
    MESSAGE 'No se pudo construir HEAD/DET para timbrado.' TYPE 'S' DISPLAY LIKE 'E'.
    pv_result = 'E'.
    RETURN.
  ENDIF.

  " Documento de ventas relacionado (si lo requieres)
  PERFORM llena_tabla_pricipal_fm USING p_factura CHANGING gv_first_sales_order.

  "--------------------- Ya timbrada? -------------------------------
  CLEAR lv_uuid_existente.
  SELECT SINGLE uuid INTO lv_uuid_existente
    FROM zcfdi_head_40
    WHERE belnr = ls_head_cfdi-belnr
      AND bukrs = ls_head_cfdi-bukrs
      AND gjahr = ls_head_cfdi-gjahr
      AND status = 'T'.
  IF sy-subrc = 0 AND lv_uuid_existente IS NOT INITIAL.
    MESSAGE 'La factura ya está timbrada.' TYPE 'S'.
    pv_result = 'N'.
    RETURN.
  ENDIF.

  "--------------------- Armar XML de solicitud (RQ) -----------------
  CREATE OBJECT lo_cfdi.
  lo_cfdi->set_data(
    EXPORTING
      t_head_cfdi = lt_head_cfdi
      t_det_cfdi  = lt_det_cfdi ).

  CLEAR lv_xml_req.
  lv_xml_req = lo_cfdi->build_xml( EXPORTING is_head = ls_head_cfdi ).
  lo_cfdi->sanitizar_xml_final( CHANGING ct_xml = lv_xml_req ).

  " Guardar RQ en LOG (xstring)
  CLEAR lv_xstr.
  PERFORM to_xstr_utf8_safe USING lv_xml_req CHANGING lv_xstr.

  CLEAR ls_log.
  ls_log-bukrs      = ls_head_cfdi-bukrs.
  ls_log-belnr      = ls_head_cfdi-belnr.
  ls_log-gjahr      = ls_head_cfdi-gjahr.
  ls_log-tipo_doc   = 'RQ'.
  ls_log-created_on = sy-datum.
  ls_log-created_at = sy-uzeit.
  ls_log-xml_req    = lv_xstr.
  INSERT zcfdi_xml_log FROM ls_log.
  IF sy-subrc <> 0.
    MODIFY zcfdi_xml_log FROM ls_log.
  ENDIF.

  "--------------------- Llamada al PAC ------------------------------
  CLEAR: lv_uuid, lv_url, lv_xml_timb, lv_err_xml, lv_err_msg, lv_err_code.
  TRY.
      lo_cfdi->call_timbrado(
        EXPORTING
          i_xml      = lv_xml_req
        IMPORTING
          e_uuid     = lv_uuid
          e_url      = lv_url
          e_xml_timb = lv_xml_timb
          e_err_xml  = lv_err_xml
          e_err_msg  = lv_err_msg
          e_err_code = lv_err_code ).
    CATCH cx_root INTO DATA(lx_call).
      lv_err_msg  = lx_call->get_text( ).
      CLEAR lv_uuid.
  ENDTRY.

  "--------------------- Resultado del PAC ---------------------------
  IF lv_uuid IS INITIAL.
    " ==== ERROR ====
    PERFORM popup_error_timbrado_plus
      USING p_factura lv_err_code lv_err_msg lv_err_xml.

    " Registrar ER en LOG
    DATA(lv_payload_err) = lv_err_msg.
    IF lv_err_xml IS NOT INITIAL.
      lv_payload_err = lv_err_xml.
    ENDIF.

    CLEAR lv_xstr.
    IF lv_payload_err IS NOT INITIAL.
      PERFORM to_xstr_utf8_safe USING lv_payload_err CHANGING lv_xstr.
    ENDIF.

    CLEAR ls_log_er.
    ls_log_er-bukrs      = ls_head_cfdi-bukrs.
    ls_log_er-belnr      = ls_head_cfdi-belnr.
    ls_log_er-gjahr      = ls_head_cfdi-gjahr.
    ls_log_er-tipo_doc   = 'ER'.
    ls_log_er-created_on = sy-datum.
    ls_log_er-created_at = sy-uzeit.
    ls_log_er-xml_req    = lv_xstr.
    INSERT zcfdi_xml_log FROM ls_log_er.
    IF sy-subrc <> 0.
      MODIFY zcfdi_xml_log FROM ls_log_er.
    ENDIF.

    pv_result = 'E'.
    RETURN.
  ENDIF.

  " ==== ÉXITO ====
  " Persistir en tablas Z (HEAD/DET/UUID/URL/XML_TIMBRADO)
  TRY.
      lo_cfdi->persist_zcfdi_tables(
        iv_uuid         = lv_uuid
        iv_url          = lv_url
        iv_xml_timb_str = lv_xml_timb ).
    CATCH cx_root INTO DATA(lx_pers).
      " Si fallara persistencia, lo tratamos como error lógico
      lv_err_msg = lx_pers->get_text( ).
      PERFORM popup_error_timbrado_plus
        USING p_factura 'PERSIST' lv_err_msg ''.
      pv_result = 'E'.
      RETURN.
  ENDTRY.

  " Registrar TI en LOG con XML timbrado
  CLEAR lv_xstr.
  IF lv_xml_timb IS NOT INITIAL.
    PERFORM to_xstr_utf8_safe USING lv_xml_timb CHANGING lv_xstr.
  ENDIF.

  CLEAR ls_log_ti.
  ls_log_ti-bukrs      = ls_head_cfdi-bukrs.
  ls_log_ti-belnr      = ls_head_cfdi-belnr.
  ls_log_ti-gjahr      = ls_head_cfdi-gjahr.
  ls_log_ti-tipo_doc   = 'TI'.
  ls_log_ti-created_on = sy-datum.
  ls_log_ti-created_at = sy-uzeit.
  ls_log_ti-xml_req    = lv_xstr.
  INSERT zcfdi_xml_log FROM ls_log_ti.
  IF sy-subrc <> 0.
    MODIFY zcfdi_xml_log FROM ls_log_ti.
  ENDIF.

  COMMIT WORK AND WAIT.

  pv_result = 'T'.

ENDFORM.



*---------------------------------------------------------------------*
* Obtener HEAD/DET de factura
*---------------------------------------------------------------------*
FORM getfactura USING
    pvbeln     TYPE vbrk-vbeln
    pfechadoc  TYPE sy-datum
   CHANGING
    ct_head_cfdi TYPE tt_head_cfdi
    ct_det_cfdi  TYPE tt_det_cfdi.

  CONSTANTS: c_mxn TYPE waers VALUE 'MXN'.

  DATA:
    ls_vbrk              TYPE vbrk,
    ls_vbco3             TYPE vbco3,
    gs_header_printvw    TYPE vbdkr,
    gt_positions_printvw TYPE tbl_vbdpr,
    lt_komv              TYPE komv_tab,
    lt_vbpa              TYPE vbpa_tab,
    lt_header            TYPE vbrk_tab,
    gs_header            TYPE vbrkvb,
    wa_t001              TYPE t001,
    wa_adrc              TYPE adrc,
    lv_gjahr             TYPE gjahr,
    lv_rate              TYPE bapi1093_0,
    wa_head_cfdi         TYPE zcfdi_head_40,
    wa_det_cfdi          TYPE zcfdi_det_40,
    lv_rfc_cli           TYPE stcd1,
    lv_rfc_emisor        TYPE paval.

  ls_vbco3-spras = sy-langu.
  ls_vbco3-vbeln = pvbeln.
  REFRESH gt_positions_printvw.
  CALL FUNCTION 'RV_BILLING_PRINT_VIEW'
    EXPORTING
      comwa = ls_vbco3
    IMPORTING
      kopf  = gs_header_printvw
    TABLES
      pos   = gt_positions_printvw.

  ls_vbrk-vbeln = pvbeln.
  REFRESH: lt_komv, lt_vbpa, lt_header, gt_positions.
  CALL FUNCTION 'RV_INVOICE_DOCUMENT_READ'
    EXPORTING
      konv_read = abap_true
      vbrk_i    = ls_vbrk
      no_nast   = space
    TABLES
      xkomv     = lt_komv
      xvbpa     = lt_vbpa
      xvbrk     = lt_header
      xvbrp     = gt_positions.

  SORT lt_header BY vbeln.
  READ TABLE lt_header INTO gs_header WITH KEY vbeln = pvbeln BINARY SEARCH.
  IF sy-subrc <> 0. RETURN. ENDIF.

  SELECT SINGLE * INTO wa_t001 FROM t001 WHERE bukrs = gs_header-bukrs.

FORM popup_error_timbrado_plus
  USING    pv_vbeln     TYPE vbrk-vbeln
           iv_err_code  TYPE string
           iv_err_msg   TYPE string
           iv_err_xml   TYPE string.

  DATA: lo_alv    TYPE REF TO cl_salv_table,
        lo_cols   TYPE REF TO cl_salv_columns_table,
        lo_col    TYPE REF TO cl_salv_column_table,
        lo_funcs  TYPE REF TO cl_salv_functions_list,
        lo_events TYPE REF TO cl_salv_events_table.

  TYPES: BEGIN OF ty_kv,
           campo TYPE string,
           valor TYPE string,
         END OF ty_kv.
  DATA: lt_kv TYPE STANDARD TABLE OF ty_kv,
        ls_kv TYPE ty_kv.

  " ---- Cabecera fija
  CLEAR ls_kv. ls_kv-campo = 'Factura'. ls_kv-valor = pv_vbeln.    APPEND ls_kv TO lt_kv.
  CLEAR ls_kv. ls_kv-campo = 'Código'.  ls_kv-valor = iv_err_code. APPEND ls_kv TO lt_kv.

  " ---- Mensaje con wrap manual a 120
  DATA lv_max        TYPE i VALUE 120.
  DATA lv_line       TYPE string.
  DATA lv_text       TYPE string.
  DATA lv_len        TYPE i.
  DATA lv_split      TYPE i.
  DATA lv_off        TYPE i.
  DATA lv_cut        TYPE i.
  DATA lv_msg_tagged TYPE abap_bool.

  lv_text = iv_err_msg.
  REPLACE ALL OCCURRENCES OF cl_abap_char_utilities=>cr_lf IN lv_text
          WITH cl_abap_char_utilities=>newline.

  IF lv_text IS INITIAL.
    CLEAR ls_kv.
    ls_kv-campo = 'Mensaje'.
    ls_kv-valor = '(sin detalle)'.
    APPEND ls_kv TO lt_kv.
  ELSE.
    DATA lt_parrafos TYPE STANDARD TABLE OF string.
    SPLIT lv_text AT cl_abap_char_utilities=>newline INTO TABLE lt_parrafos.

    LOOP AT lt_parrafos INTO lv_text.
      SHIFT lv_text LEFT DELETING LEADING space.

      WHILE lv_text IS NOT INITIAL.
        lv_len = strlen( lv_text ).

        IF lv_len <= lv_max.
          lv_line = lv_text.
          CLEAR lv_text.
        ELSE.
          " Buscar el último espacio dentro de los primeros 120
          lv_off = lv_max - 1.
          WHILE lv_off > 0 AND lv_text+lv_off(1) <> space.
            lv_off = lv_off - 1.
          ENDWHILE.

          IF lv_off = 0.
            " No hay espacio: corte duro en 120
            lv_split = lv_max.
            lv_cut   = lv_split.          " no incrementa (no hay espacio)
          ELSE.
            " Cortar antes del espacio
            lv_split = lv_off.
            lv_cut   = lv_split + 1.      " saltar el espacio
          ENDIF.

          lv_line = lv_text(lv_split).
          SHIFT lv_text LEFT BY lv_cut PLACES.
          SHIFT lv_text LEFT DELETING LEADING space.
        ENDIF.

        CLEAR ls_kv.
        IF lv_msg_tagged IS INITIAL.
          ls_kv-campo = 'Mensaje'.
          lv_msg_tagged = abap_true.
        ELSE.
          ls_kv-campo = ''.
        ENDIF.
        ls_kv-valor = lv_line.
        APPEND ls_kv TO lt_kv.
      ENDWHILE.
    ENDLOOP.
  ENDIF.

  " ---- SALV popup (LIST)
  cl_salv_table=>factory(
    IMPORTING r_salv_table = lo_alv
    CHANGING  t_table      = lt_kv ).

  lo_alv->set_screen_popup(
    start_column = 5  start_line = 2
    end_column   = 130 end_line  = 25 ).

  lo_alv->get_display_settings( )->set_list_header( 'Error de timbrado (PAC)' ).
  lo_alv->get_display_settings( )->set_striped_pattern( abap_true ).

  lo_cols = lo_alv->get_columns( ).
  lo_cols->set_optimize( abap_true ).
  lo_col ?= lo_cols->get_column( 'CAMPO' ). lo_col->set_short_text( 'Campo' ).
  lo_col ?= lo_cols->get_column( 'VALOR' ). lo_col->set_short_text( 'Valor' ).

  " ---- Funciones: sólo personalizadas en LIST (evitar ENABLE_FUNCTION estándar)
  lo_funcs = lo_alv->get_functions( ).
  " No usar: lo_funcs->set_all( abap_true ).

  TRY.
      lo_funcs->add_function(
        name     = 'Z_CPYMSG'                     " <— código propio
        text     = 'Copiar mensaje'
        tooltip  = 'Copiar mensaje al portapapeles'
        position = if_salv_c_function_position=>right_of_salv_functions ).

      IF iv_err_xml IS NOT INITIAL.
        lo_funcs->add_function(
          name     = 'Z_XML'                      " <— código propio
          text     = 'Guardar/Abrir XML'
          tooltip  = 'Guardar y abrir el XML de error'
          position = if_salv_c_function_position=>right_of_salv_functions ).
      ENDIF.
    CATCH cx_salv_method_not_supported.
      " En releases muy antiguos podría no permitir add_function en LIST: ignorar
  ENDTRY.

  " ---- Eventos
  lo_events = lo_alv->get_event( ).
  DATA(lo_h) = NEW lcl_pacerr_handler( ).
  lo_h->vbeln = pv_vbeln.
  lo_h->msg   = iv_err_msg.
  lo_h->xml   = iv_err_xml.
  SET HANDLER lo_h->on_added_function FOR lo_events.

  lo_alv->display( ).

ENDFORM.

           OR estado = 'Éxito (RQ)'.

    WHEN OTHERS.

  ENDCASE.

FORM filter_by_state.

  DATA lt_filtered TYPE STANDARD TABLE OF ty_out.

    ENDIF.

    IF lv_keep = abap_true.
      APPEND gs_out TO lt_filtered.
    ENDIF.
  ENDLOOP.

  gt_out = lt_filtered.

FORM last_index_of USING    p_text   TYPE string
                            p_needle TYPE string
                   CHANGING p_idx    TYPE i.

  DATA: lv_off        TYPE i VALUE 0,
        lv_total_len  TYPE i,
        lv_needle_len TYPE i,
        lv_match_off  TYPE i.

  p_idx = -1.
  lv_total_len  = strlen( p_text ).
  lv_needle_len = strlen( p_needle ).

  IF lv_total_len = 0 OR lv_needle_len = 0 OR lv_total_len < lv_needle_len.
    RETURN.
  ENDIF.

  WHILE lv_off <= lv_total_len - lv_needle_len.
    FIND p_needle IN p_text+lv_off MATCH OFFSET lv_match_off.
    IF sy-subrc <> 0.
      EXIT.
    ENDIF.
    p_idx   = lv_off + lv_match_off.     " guardar última coincidencia
    lv_off  = p_idx + 1.                 " avanzar SIEMPRE una posición
  ENDWHILE.

ENDFORM.

     AND tipo_doc = 'ER'.

  IF sy-subrc <> 0 OR lv_xstr_in IS INITIAL.
    MESSAGE 'No se encontró XML tipo ER en ZCFDI_XML_LOG' TYPE 'I'.
    RETURN.
  ENDIF.

*---------------------------------------------------------------------*
* 2) Convertir XSTRING → STRING legible (UTF-8 / UTF-16)
*---------------------------------------------------------------------*
  TRY.
      IF lv_xstr_in(2) = cl_abap_char_utilities=>byte_order_mark_big OR
         lv_xstr_in(2) = cl_abap_char_utilities=>byte_order_mark_little.
        lo_conv_in = cl_abap_conv_in_ce=>create( encoding = 'UTF-16LE' ).
      ELSE.
        lo_conv_in = cl_abap_conv_in_ce=>create( encoding = 'UTF-8' ).
      ENDIF.

      lo_conv_in->convert(
        EXPORTING input  = lv_xstr_in
        IMPORTING data   = lv_txt ).

    CATCH cx_root INTO DATA(lx_conv).
      MESSAGE lx_conv->get_text( ) TYPE 'E'.
      RETURN.
  ENDTRY.

  IF lv_txt IS INITIAL.
    MESSAGE 'El XML de error está vacío o no se pudo convertir.' TYPE 'I'.
    RETURN.
  ENDIF.

*---------------------------------------------------------------------*
* 3) Limpieza mínima / normalización
*---------------------------------------------------------------------*
  IF lv_txt CS '<InnerErrors>' OR lv_txt CS '<Error>'.
    PERFORM normalize_innererrors_xml USING lv_txt CHANGING lv_txt lv_xstr_out.
  ELSE.
    PERFORM normalize_raw_to_xml USING lv_xstr_in CHANGING lv_txt lv_xstr_out.
    PERFORM sanitizar_xml_final   CHANGING lv_txt.
    PERFORM clean_valid_xml_range CHANGING lv_txt.
  ENDIF.

  IF lv_txt(1) <> '<'.
    FIND FIRST OCCURRENCE OF '<' IN lv_txt MATCH OFFSET DATA(lv_off).
    IF sy-subrc = 0 AND lv_off GT 0.
      lv_txt = lv_txt+lv_off.
    ENDIF.
  ENDIF.

*---------------------------------------------------------------------*
* 4) Crear HTML con meta UTF-8
*---------------------------------------------------------------------*
  CONCATENATE
    '<html><head><meta http-equiv="Content-Type" content="text/html; charset=UTF-8"></head>'
    '<body><pre>'
    lv_txt
    '</pre></body></html>'
  INTO lv_html.

*---------------------------------------------------------------------*
* 5) Crear carpeta C:\Temp y guardar HTML en UTF-8 binario
*---------------------------------------------------------------------*
  CALL METHOD cl_gui_frontend_services=>directory_create
    EXPORTING
      directory = lv_workdir
    CHANGING
      rc        = lv_rc
    EXCEPTIONS
      directory_create_failed  = 1
      cntl_error               = 2
      error_no_gui             = 3
      directory_access_denied  = 4
      directory_already_exists = 5
      path_not_found           = 6
      unknown_error            = 7
      not_supported_by_gui     = 8
      wrong_parameter          = 9
      others                   = 10.

  IF sy-subrc <> 0 AND sy-subrc <> 5.
    MESSAGE |Error al crear directorio C:\\Temp (código { sy-subrc })| TYPE 'I'.
  ENDIF.

  CONCATENATE 'CFDI_Error_Timbrado_' p_belnr '.html' INTO lv_file.
  CONCATENATE lv_workdir '\' lv_file INTO lv_full.

  " --- Conversión a UTF-8 binario real ---
  CALL FUNCTION 'SCMS_STRING_TO_XSTRING'
    EXPORTING
      text     = lv_html
      mimetype = 'text/html; charset=UTF-8'
    IMPORTING
      buffer   = lv_xstr_html.

  CALL FUNCTION 'SCMS_XSTRING_TO_BINARY'
    EXPORTING
      buffer = lv_xstr_html
    TABLES
      binary_tab = lt_bin_tab.

  CALL METHOD cl_gui_frontend_services=>gui_download
    EXPORTING
      filename = lv_full
      filetype = 'BIN'
    CHANGING
      data_tab = lt_bin_tab
    EXCEPTIONS
      file_write_error        = 1
      invalid_type            = 2
      no_batch                = 3
      gui_refuse_filetransfer = 4
      others                  = 5.

  IF sy-subrc <> 0.
    MESSAGE |No se pudo crear el archivo local { lv_full }| TYPE 'E'.
    RETURN.
  ENDIF.

*---------------------------------------------------------------------*
* 6) Abrir automáticamente el archivo .html en el explorador
*---------------------------------------------------------------------*
  CALL METHOD cl_gui_frontend_services=>execute
    EXPORTING
      document  = lv_full
      operation = 'OPEN'
    EXCEPTIONS
      file_not_found = 1
      others         = 2.

  IF sy-subrc = 0.
    MESSAGE |Archivo abierto correctamente: { lv_full }| TYPE 'S'.
  ELSE.
    MESSAGE |No se pudo abrir el archivo local { lv_full }| TYPE 'I'.
  ENDIF.

ENDFORM.


*---------------------------------------------------------------------*
*   FORM NORMALIZE_INNERERRORS_XML (compatible con SAP NW 7.0x)
*---------------------------------------------------------------------*
FORM normalize_innererrors_xml USING    p_xml_txt_in   TYPE string
                               CHANGING p_xml_txt_out  TYPE string
                                         p_xstr_out    TYPE xstring.

  DATA: lv_start TYPE i,
        lv_end   TYPE i,
        lv_len   TYPE i.

  " Copiar texto original
  p_xml_txt_out = p_xml_txt_in.

  " Buscar apertura del XML válido
  FIND FIRST OCCURRENCE OF REGEX '<(InnerErrors|Error)' IN p_xml_txt_out MATCH OFFSET lv_start.
  IF sy-subrc = 0.
    p_xml_txt_out = p_xml_txt_out+lv_start.
  ENDIF.

  " Buscar cierre del XML
  FIND ALL OCCURRENCES OF REGEX '</(InnerErrors|Error)>' IN p_xml_txt_out MATCH OFFSET lv_end.
  IF sy-subrc = 0.
    lv_end = lv_end + STRLEN('</InnerErrors>').
    lv_len = lv_end.
    p_xml_txt_out = p_xml_txt_out(lv_len).
  ENDIF.

  " Convertir STRING a XSTRING (UTF-8 del sistema)
  DATA lo_conv TYPE REF TO cl_abap_conv_out_ce.
  TRY.
      lo_conv = cl_abap_conv_out_ce=>create(
                  encoding    = 'UTF-8'
                  replacement = '?'
                  ignore_cerr = abap_true ).
      lo_conv->write( data = p_xml_txt_out ).
      p_xstr_out = lo_conv->get_buffer( ).
    CATCH cx_root INTO DATA(lx_err).
      MESSAGE lx_err->get_text( ) TYPE 'E'.
  ENDTRY.

ENDFORM.



*---------------------------------------------------------------------*
*   FORM NORMALIZE_RAW_TO_XML
*---------------------------------------------------------------------*
FORM normalize_raw_to_xml USING    p_xstr_in  TYPE xstring
                          CHANGING p_xml_txt  TYPE string
                                   p_xstr_out TYPE xstring.

  CALL FUNCTION 'ECATT_CONV_XSTRING_TO_STRING'
    EXPORTING
      im_xstring  = p_xstr_in
      im_encoding = 'UTF-8'
    IMPORTING
      ex_string   = p_xml_txt.

  IF p_xml_txt IS INITIAL.
    CALL FUNCTION 'ECATT_CONV_XSTRING_TO_STRING'
      EXPORTING
        im_xstring  = p_xstr_in
        im_encoding = 'ISO-8859-1'
      IMPORTING
        ex_string   = p_xml_txt.
  ENDIF.

  PERFORM sanitizar_xml_final CHANGING p_xml_txt.

  CALL FUNCTION 'SCMS_STRING_TO_XSTRING'
    EXPORTING text = p_xml_txt
    IMPORTING buffer = p_xstr_out.

ENDFORM.
*---------------------------------------------------------------------*
*   FORM SANITIZAR_XML_FINAL (versión final sin NULL 0x00)
*---------------------------------------------------------------------*
FORM sanitizar_xml_final CHANGING p_xml_txt TYPE string.

  DATA: lv_bom_big    TYPE c LENGTH 1 VALUE cl_abap_char_utilities=>byte_order_mark_big,
        lv_bom_little TYPE c LENGTH 1 VALUE cl_abap_char_utilities=>byte_order_mark_little,
        lv_minchar    TYPE c LENGTH 1 VALUE cl_abap_char_utilities=>minchar.

  " Quitar BOM si está al inicio
  IF p_xml_txt(1) = lv_bom_big OR p_xml_txt(1) = lv_bom_little.
    p_xml_txt = p_xml_txt+1.
  ENDIF.

  " Eliminar BOM en cualquier posición
  REPLACE ALL OCCURRENCES OF lv_bom_big    IN p_xml_txt WITH ''.
  REPLACE ALL OCCURRENCES OF lv_bom_little IN p_xml_txt WITH ''.

  " Reemplazar tabs, saltos de línea y form feeds por espacios
  REPLACE ALL OCCURRENCES OF cl_abap_char_utilities=>horizontal_tab IN p_xml_txt WITH space.
  REPLACE ALL OCCURRENCES OF cl_abap_char_utilities=>newline        IN p_xml_txt WITH space.
  REPLACE ALL OCCURRENCES OF cl_abap_char_utilities=>cr_lf          IN p_xml_txt WITH space.
  REPLACE ALL OCCURRENCES OF cl_abap_char_utilities=>form_feed      IN p_xml_txt WITH space.
  REPLACE ALL OCCURRENCES OF cl_abap_char_utilities=>vertical_tab   IN p_xml_txt WITH space.

  " Eliminar caracteres NULL (0x00) o equivalentes
  REPLACE ALL OCCURRENCES OF lv_minchar IN p_xml_txt WITH ''.
  REPLACE ALL OCCURRENCES OF CONV string( '\u0000' ) IN p_xml_txt WITH ''.

  " Quitar cualquier rango de control no imprimible
  REPLACE ALL OCCURRENCES OF REGEX '[\x00-\x08]'  IN p_xml_txt WITH ''.
  REPLACE ALL OCCURRENCES OF REGEX '[\x0B-\x0C]'  IN p_xml_txt WITH ''.
  REPLACE ALL OCCURRENCES OF REGEX '[\x0E-\x1F]'  IN p_xml_txt WITH ''.

  " Quitar espacios en blanco sobrantes
  SHIFT p_xml_txt LEFT DELETING LEADING space.
  SHIFT p_xml_txt RIGHT DELETING TRAILING space.

ENDFORM.
*---------------------------------------------------------------------*
*   FORM CLEAN_VALID_XML_RANGE
*---------------------------------------------------------------------*
* Recorta el XML para eliminar contenido extra fuera del documento
*---------------------------------------------------------------------*
FORM clean_valid_xml_range CHANGING p_xml TYPE string.

  DATA: lv_start TYPE i,
        lv_end   TYPE i,
        lv_len   TYPE i.

  " Buscar inicio del primer tag raíz
  FIND FIRST OCCURRENCE OF REGEX '<[a-zA-Z0-9:_-]+' IN p_xml MATCH OFFSET lv_start.
  IF sy-subrc <> 0.
    RETURN.
  ENDIF.

  " Buscar el cierre del último tag (</...>)
  FIND ALL OCCURRENCES OF REGEX '</[a-zA-Z0-9:_-]+>' IN p_xml MATCH OFFSET lv_end.
  IF sy-subrc = 0.
    lv_end = lv_end + STRLEN( '</' ).
  ENDIF.

  " Si hay cierre válido, recortar
  IF lv_end GT lv_start.
    lv_len = lv_end - lv_start.
    p_xml = p_xml+lv_start(lv_len).
  ENDIF.

ENDFORM.
*---------------------------------------------------------------------*
*   FORM OPEN_XML_FROM_LOG
*---------------------------------------------------------------------*
*   Lee XML tipo RQ desde la tabla ZCFDI_XML_LOG, lo limpia,
*   lo guarda como UTF-8 en C:\Temp y lo abre automáticamente.
*---------------------------------------------------------------------*
FORM open_xml_from_log USING p_belnr TYPE vbrk-vbeln
                              p_bukrs TYPE bukrs
                              p_gjahr TYPE gjahr.

  DATA: lv_xstr_in   TYPE xstring,
        lv_txt       TYPE string,
        lv_workdir   TYPE string VALUE 'C:\Temp',
        lv_file      TYPE string,
        lv_full      TYPE string,
        lv_rc        TYPE i,
        lv_off       TYPE i,
        lv_first2    TYPE x LENGTH 2,
        lv_utf16     TYPE abap_bool,
        lo_conv_in   TYPE REF TO cl_abap_conv_in_ce,
        lt_xml_tab   TYPE STANDARD TABLE OF string,
        lv_clean     TYPE string.

*---------------------------------------------------------------------*
* 1) Leer XML tipo RQ desde la tabla Z
*---------------------------------------------------------------------*
  SELECT SINGLE xml_req
    INTO lv_xstr_in
    FROM zcfdi_xml_log
   WHERE bukrs    = p_bukrs
     AND belnr    = p_belnr
     AND gjahr    = p_gjahr
     AND tipo_doc = 'RQ'.

  IF sy-subrc <> 0 OR lv_xstr_in IS INITIAL.
    MESSAGE 'No se encontró XML tipo RQ en ZCFDI_XML_LOG' TYPE 'I'.
    RETURN.
  ENDIF.

*---------------------------------------------------------------------*
* 2) Detectar codificación (UTF-8 / UTF-16)
*---------------------------------------------------------------------*
  lv_first2 = lv_xstr_in(2).
  IF lv_first2 = 'FFFE' OR lv_first2 = 'FEFF'.
    lv_utf16 = abap_true.
  ELSE.
    lv_utf16 = abap_false.
  ENDIF.

*---------------------------------------------------------------------*
* 3) Convertir XSTRING → STRING legible
*---------------------------------------------------------------------*
  TRY.
      IF lv_utf16 = abap_true.
        lo_conv_in = cl_abap_conv_in_ce=>create( encoding = 'UTF-16LE' ).
      ELSE.
        lo_conv_in = cl_abap_conv_in_ce=>create( encoding = 'UTF-8' ).
      ENDIF.

      lo_conv_in->convert(
        EXPORTING input  = lv_xstr_in
        IMPORTING data   = lv_txt ).

    CATCH cx_root INTO DATA(lx_conv).
      MESSAGE lx_conv->get_text( ) TYPE 'E'.
      RETURN.
  ENDTRY.

  IF lv_txt IS INITIAL.
    MESSAGE 'El XML Request está vacío o no se pudo convertir.' TYPE 'I'.
    RETURN.
  ENDIF.

*---------------------------------------------------------------------*
* 4) Limpieza profunda de BOM, saltos y espacios invisibles
*---------------------------------------------------------------------*
  lv_clean = lv_txt.

  " Eliminar CR/LF y tabs invisibles
  REPLACE ALL OCCURRENCES OF cl_abap_char_utilities=>cr_lf IN lv_clean WITH space.
  REPLACE ALL OCCURRENCES OF cl_abap_char_utilities=>newline IN lv_clean WITH space.
  REPLACE ALL OCCURRENCES OF cl_abap_char_utilities=>horizontal_tab IN lv_clean WITH space.

  " Eliminar BOM si existe
  IF lv_clean(1) = cl_abap_char_utilities=>byte_order_mark_big OR
     lv_clean(1) = cl_abap_char_utilities=>byte_order_mark_little.
    lv_clean = lv_clean+1.
  ENDIF.

  " Quitar todo lo anterior al primer '<'
  FIND FIRST OCCURRENCE OF '<' IN lv_clean MATCH OFFSET lv_off.
  IF sy-subrc = 0 AND lv_off GT 0.
    lv_clean = lv_clean+lv_off.
  ENDIF.

  " Quitar espacios iniciales y finales
  SHIFT lv_clean LEFT DELETING LEADING space.
  SHIFT lv_clean RIGHT DELETING TRAILING space.

  lv_txt = lv_clean.

*---------------------------------------------------------------------*
* 5) Crear carpeta C:\Temp (misma lógica del HTML)
*---------------------------------------------------------------------*
  CALL METHOD cl_gui_frontend_services=>directory_create
    EXPORTING
      directory = lv_workdir
    CHANGING
      rc        = lv_rc
    EXCEPTIONS
      directory_already_exists = 1
      others                   = 2.

  " Si existe o ya se puede usar, continuar
  IF sy-subrc <> 0 AND lv_rc NE 0.
    MESSAGE 'No se pudo crear el directorio C:\Temp (verifique permisos).' TYPE 'I'.
  ENDIF.

*---------------------------------------------------------------------*
* 6) Guardar XML limpio (UTF-8 texto plano)
*---------------------------------------------------------------------*
  CONCATENATE 'CFDI_Request_' p_belnr '.xml' INTO lv_file.
  CONCATENATE lv_workdir '\' lv_file INTO lv_full.

  APPEND lv_txt TO lt_xml_tab.

  CALL METHOD cl_gui_frontend_services=>gui_download
    EXPORTING
      filename = lv_full
      filetype = 'ASC'
      codepage = '4110'    " UTF-8
    CHANGING
      data_tab = lt_xml_tab
    EXCEPTIONS
      OTHERS = 5.

  IF sy-subrc <> 0.
    MESSAGE |No se pudo crear el archivo XML local { lv_full }| TYPE 'E'.
    RETURN.
  ENDIF.

*---------------------------------------------------------------------*
* 7) Abrir automáticamente el XML en el navegador
*---------------------------------------------------------------------*
  CALL METHOD cl_gui_frontend_services=>execute
    EXPORTING
      document  = lv_full
      operation = 'OPEN'
    EXCEPTIONS
      file_not_found = 1
      others         = 2.

  IF sy-subrc = 0.
    MESSAGE |Archivo XML RQ abierto correctamente: { lv_full }| TYPE 'S'.
  ELSE.
    MESSAGE |No se pudo abrir el archivo local { lv_full }| TYPE 'I'.
  ENDIF.

ENDFORM.

*---------------------------------------------------------------------*
*   FORM OPEN_XML_TIMB
*---------------------------------------------------------------------*
*   Lee XML tipo TI (timbrado) desde ZCFDI_XML_40_XML o ZCFDI_XML_LOG,
*   corrige cabecera en memoria, valida, guarda y abre en navegador.
*---------------------------------------------------------------------*
FORM open_xml_timb USING p_belnr TYPE vbrk-vbeln
                         p_bukrs TYPE bukrs
                         p_gjahr TYPE gjahr.

  DATA: lv_xml_str   TYPE string,      " XML como STRING (tabla nueva)
        lv_xstr_in   TYPE xstring,     " XML binario (tabla antigua)
        lv_txt       TYPE string,
        lv_valido    TYPE abap_bool,
        lv_msg       TYPE string,
        lv_workdir   TYPE string VALUE 'C:\Temp',
        lv_file      TYPE string,
        lv_full      TYPE string,
        lv_rc        TYPE i,
        lv_tabla_fuente TYPE string,
        lo_conv_in   TYPE REF TO cl_abap_conv_in_ce,
        lx_conv      TYPE REF TO cx_root,
        lt_bin_tab   TYPE solix_tab,
        lv_xstr_html TYPE xstring.

*---------------------------------------------------------------------*
* 1) Leer XML tipo TI — primero de ZCFDI_XML_40_XML, luego ZCFDI_XML_LOG
*---------------------------------------------------------------------*
  CLEAR lv_tabla_fuente.

  " Buscar primero en tabla nueva ZCFDI_XML_40_XML (campo tipo STRING)
  SELECT SINGLE xml
    INTO lv_xml_str
    FROM zcfdi_40_xml
   WHERE bukrs    = p_bukrs
     AND factura     = p_belnr
     AND gjahr    = p_gjahr.


  IF sy-subrc = 0 AND lv_xml_str IS NOT INITIAL.
    lv_tabla_fuente = 'ZCFDI_XML_40_XML'.
    lv_txt = lv_xml_str.
  ELSE.
    " Buscar en tabla anterior ZCFDI_XML_LOG (campo XSTRING o STRING)
    SELECT SINGLE xml_req
      INTO lv_xstr_in
      FROM zcfdi_xml_log
     WHERE bukrs    = p_bukrs
       AND belnr    = p_belnr
       AND gjahr    = p_gjahr
       AND tipo_doc = 'TI'.

    IF sy-subrc = 0 AND lv_xstr_in IS NOT INITIAL.
      lv_tabla_fuente = 'ZCFDI_XML_LOG'.

      TRY.
          lo_conv_in = cl_abap_conv_in_ce=>create( encoding = 'UTF-8' ).
          lo_conv_in->convert(
            EXPORTING input  = lv_xstr_in
            IMPORTING data   = lv_txt ).
        CATCH cx_root INTO lx_conv.
          MESSAGE lx_conv->get_text( ) TYPE 'E'.
          RETURN.
      ENDTRY.
    ENDIF.
  ENDIF.

  IF lv_txt IS INITIAL.
    MESSAGE 'No se encontró XML tipo TI en ninguna tabla (ZCFDI_XML_40_XML / ZCFDI_XML_LOG)' TYPE 'I'.
    RETURN.
  ENDIF.

  MESSAGE |XML Timbrado recuperado correctamente desde { lv_tabla_fuente }.| TYPE 'S'.

*---------------------------------------------------------------------*
* 2) Corregir cabecera malformada en memoria
*---------------------------------------------------------------------*
  IF lv_txt CS '<?xmlversion=' OR lv_txt CS '<?xmlversion ='.
    REPLACE ALL OCCURRENCES OF '<?xmlversion' IN lv_txt WITH '<?xml version'.
  ENDIF.

  IF lv_txt CS '"1.0"encoding=' OR lv_txt CS '"1.0" encoding='.
    REPLACE ALL OCCURRENCES OF '"1.0"encoding=' IN lv_txt WITH '"1.0" encoding='.
  ENDIF.

*---------------------------------------------------------------------*
* 3) Validar XML (sintaxis y estructura básica)
*---------------------------------------------------------------------*
  PERFORM validar_xml USING lv_txt CHANGING lv_valido lv_msg.
  IF lv_valido = abap_false.
    MESSAGE lv_msg TYPE 'E'.
    RETURN.
  ENDIF.

*---------------------------------------------------------------------*
* 4) Crear directorio local C:\Temp si no existe
*---------------------------------------------------------------------*
  CALL METHOD cl_gui_frontend_services=>directory_create
    EXPORTING
      directory = lv_workdir
    CHANGING
      rc        = lv_rc
    EXCEPTIONS
      directory_create_failed  = 1
      cntl_error               = 2
      error_no_gui             = 3
      directory_access_denied  = 4
      directory_already_exists = 5
      path_not_found           = 6
      unknown_error            = 7
      not_supported_by_gui     = 8
      wrong_parameter          = 9
      OTHERS                   = 10.

  IF sy-subrc <> 0 AND sy-subrc <> 5.
    MESSAGE |No se pudo crear C:\\Temp (verifique permisos). Código: { sy-subrc }| TYPE 'I'.
    RETURN.
  ENDIF.

*---------------------------------------------------------------------*
* 5) Guardar XML limpio en UTF-8
*---------------------------------------------------------------------*
  CONCATENATE 'CFDI_Timbrado_' p_belnr '.xml' INTO lv_file.
  CONCATENATE lv_workdir '\' lv_file INTO lv_full.

  CALL FUNCTION 'SCMS_STRING_TO_XSTRING'
    EXPORTING
      text     = lv_txt
      mimetype = 'text/xml; charset=UTF-8'
    IMPORTING
      buffer   = lv_xstr_html.

  CALL FUNCTION 'SCMS_XSTRING_TO_BINARY'
    EXPORTING
      buffer = lv_xstr_html
    TABLES
      binary_tab = lt_bin_tab.

  CALL METHOD cl_gui_frontend_services=>gui_download
    EXPORTING
      filename = lv_full
      filetype = 'BIN'
    CHANGING
      data_tab = lt_bin_tab
    EXCEPTIONS
      file_write_error        = 1
      invalid_type            = 2
      no_batch                = 3
      gui_refuse_filetransfer = 4
      others                  = 5.

  IF sy-subrc <> 0.
    MESSAGE |No se pudo crear el archivo local { lv_full }| TYPE 'E'.
    RETURN.
  ENDIF.

*---------------------------------------------------------------------*
* 6) Abrir XML en navegador
*---------------------------------------------------------------------*
  CALL METHOD cl_gui_frontend_services=>execute
    EXPORTING
      document  = lv_full
      operation = 'OPEN'
    EXCEPTIONS
      file_not_found = 1
      others         = 2.

  IF sy-subrc = 0.
    MESSAGE |Archivo XML Timbrado abierto correctamente: { lv_full }| TYPE 'S'.
  ELSE.
    MESSAGE |No se pudo abrir el archivo local { lv_full }| TYPE 'I'.
  ENDIF.

ENDFORM.



*---------------------------------------------------------------------*
* FORM VALIDAR_XML
*---------------------------------------------------------------------*
* Valida sintácticamente un XML (STRING) antes de abrirlo o procesarlo.
* Compatible con SAP_BASIS 750 (sin sintaxis nueva ni expresiones inline)
*---------------------------------------------------------------------*
FORM validar_xml USING    p_xml_txt TYPE string
                 CHANGING p_valido  TYPE abap_bool
                          p_mensaje TYPE string.

  DATA: lv_len        TYPE i,
        lv_open_tag   TYPE string,
        lv_close_tag  TYPE string,
        lv_off        TYPE i,
        lv_null_char  TYPE string.

  CLEAR: p_valido, p_mensaje.
  p_valido = abap_false.

*---------------------------------------------------------------------*
* 1) Verificar que el XML no esté vacío
*---------------------------------------------------------------------*
  lv_len = STRLEN( p_xml_txt ).
  IF lv_len = 0.
    p_mensaje = 'El XML está vacío o no se pudo recuperar.'.
    RETURN.
  ENDIF.

*---------------------------------------------------------------------*
* 2) Verificar que contenga el carácter "<"
*---------------------------------------------------------------------*
  FIND FIRST OCCURRENCE OF '<' IN p_xml_txt MATCH OFFSET lv_off.
  IF sy-subrc <> 0.
    p_mensaje = 'No se encontró ningún inicio de etiqueta XML ("<").'.
    RETURN.
  ENDIF.

*---------------------------------------------------------------------*
* 3) Validar cabecera XML estándar
*---------------------------------------------------------------------*
  IF p_xml_txt CS '<?xmlversion=' OR p_xml_txt CS '<?xmlversion ='.
    p_mensaje = 'Cabecera XML malformada: falta espacio después de "xml".'.
    RETURN.
  ENDIF.

  IF p_xml_txt NS '<?xml version='.
    p_mensaje = 'Cabecera XML faltante o incorrecta.'.
    RETURN.
  ENDIF.

*---------------------------------------------------------------------*
* 4) Validar caracteres inválidos (NULL o #)
*---------------------------------------------------------------------*
  TRY.
      lv_null_char = cl_abap_conv_in_ce=>uccp( '0000' ).
    CATCH cx_root.
      CLEAR lv_null_char.
  ENDTRY.

  IF lv_null_char IS NOT INITIAL AND p_xml_txt CS lv_null_char.
    p_mensaje = 'El XML contiene caracteres NULL no válidos.'.
    RETURN.
  ENDIF.

  IF p_xml_txt CS '#'.
    p_mensaje = 'El XML contiene caracteres "#" no válidos o mal codificados.'.
    RETURN.
  ENDIF.

*---------------------------------------------------------------------*
* 5) Validar estructura mínima: etiqueta raíz de apertura/cierre
*---------------------------------------------------------------------*
  FIND FIRST OCCURRENCE OF REGEX '<([A-Za-z0-9:_\-]+)' IN p_xml_txt
       SUBMATCHES lv_open_tag.

  FIND FIRST OCCURRENCE OF REGEX '</([A-Za-z0-9:_\-]+)>' IN p_xml_txt
       SUBMATCHES lv_close_tag.

  IF lv_open_tag IS INITIAL OR lv_close_tag IS INITIAL.
    p_mensaje = 'No se encontró estructura XML con etiquetas de apertura/cierre.'.
    RETURN.
  ENDIF.

  IF lv_open_tag <> lv_close_tag.
    CONCATENATE 'Etiqueta raíz no coincide: <' lv_open_tag '> ... </'
                lv_close_tag '>.' INTO p_mensaje.
    RETURN.
  ENDIF.

*---------------------------------------------------------------------*
* 6) Validación exitosa
*---------------------------------------------------------------------*
  p_valido  = abap_true.
  p_mensaje = 'XML válido y bien formado.'.

ENDFORM.
